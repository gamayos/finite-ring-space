2"""Builds 10-dimensions-rows.ipynb: one cell per witnessed row of the paper's predicate ledger, each cell addressable
3from the ledger page by its stable id (`row-10-C5`: Colab opens the notebook at that cell with `#scrollTo=row-10-C5`).
4A row cell states the row (its text, from the site's ledger JSON) and runs `dimcommon.row("10:C5")`: the script of the

6family record citing the row is listed with its verdict. Run: python3 make_rows_notebook.py  (then execute the notebook)."""

10PKG = "10-dimensions"; NB = f"{PKG}-rows.ipynb"; PAPER = "10"

13cells = []
14def md(s, cid): cells.append({"cell_type": "markdown", "metadata": {"id": cid}, "source": s})
15def code(s, cid): cells.append({"cell_type": "code", "metadata": {"id": cid}, "execution_count": None, "outputs": [], "source": s})

17def plain(tex):
18    """The row's text for a code comment: TeX kept, whitespace collapsed, wrapped at 110 columns."""
19    words = " ".join(tex.split()).split(" "); lines, cur = [], "#"
20    for w in words:
21        if len(cur) + 1 + len(w) > 110: lines.append(cur); cur = "#"
22        cur += " " + w
23    lines.append(cur); return "\n".join(lines)
