63def markers():
64    """row label -> (script file, line) of its `# row …` marker: the line of the check that decides the row."""
65    import re
66    out = {}
67    for f in sorted(set(SCRIPT.values())):
68        for i, line in enumerate(open(os.path.join(os.path.dirname(os.path.abspath(__file__)), f + ".py"), encoding="utf-8"), 1):
69            m = re.match(r"\s*# row (.*)", line)
70            if m:
71                for lab in m.group(1).split(","): out.setdefault(lab.strip(), (f + ".py", i))
72    return out

74def row(label, lines=14):
75    """Verify one ledger row: run the script of its deciding family (once per session), print the deciding check's
76    source (from its `# row` marker) and every family record that cites the row, and return True iff all pass."""
77    fam = ROWS.get(label)
78    if fam is None:
79        print(f"{label}: no python witness (see the row's Lean witness or its source)"); return None
80    script = SCRIPT.get(fam.split(".")[0], fam)
81    for sc in (sorted(set(SCRIPT.values())) if script == "run_all" else [script]):     # the driver row runs every suite
82        if sc not in _RAN:
83            mod = __import__(sc); mod.run(); _RAN.add(sc)
84    here = os.path.dirname(os.path.abspath(__file__)); mk = markers().get(label)
85    if mk:
86        src = open(os.path.join(here, mk[0]), encoding="utf-8").read().split("\n")
87        print(f"— {mk[0]}:{mk[1]} (the check that decides {label}; family {fam})")
88        for j in range(mk[1] - 1, min(mk[1] - 1 + lines, len(src))): print(f"{j + 1:5d}  {src[j]}")
89    else:
90        print(f"— {script}.py (the whole script witnesses {label})")
91    recs = [r for r in RESULTS if label in [t.strip() for t in r["rows"].split(",")]] or [r for r in RESULTS if script == "run_all" or r["script"] == script]
92    ok = all(r["ok"] for r in recs)
93    for r in recs:
94        role = "(the script's family)" if "." not in fam else "(deciding)" if r["id"] == fam else "(corroborating)"
95        print(f"  [{'PASS' if r['ok'] else 'FAIL'}] {r['id']:8s} {role:22s} {r['detail']}")
96    print(f"{label}: {'VERIFIED' if ok and recs else 'FAILED'} — {len(recs)} family record(s)")
97    return ok

99def family(tag, fam):

102def chk(label, ok, tag=None, fam=None):
103    """A micro-check, recorded under the current (tag, family); never exits."""
104    MICRO.append((tag or _FAM[0], fam or _FAM[1], label, bool(ok)))
105    if not ok:
106        print(f"    FAIL: {label}")

108def flush(tag, order=None):
109    fams = OrderedDict()
110    for t, fam, label, ok in MICRO:
111        if t == tag:
112            fams.setdefault(fam, []).append(ok)
113    for fam in (order or fams):
114        oks = fams.get(fam, [])
115        check(f"{tag}.{fam}", ok=bool(oks) and all(oks), detail=f"{len(oks)} exact micro-checks")

117def check(pid, ok, detail="", kind="EXACT", label=None):
118    ok = bool(ok)
119    rows = LEDGER.get(pid, "")
120    label = label or LABELS.get(pid, pid)
121    RESULTS.append({"id": pid, "rows": rows, "script": SCRIPT.get(pid.split(".")[0], pid.split(".")[0]), "label": label, "ok": ok, "detail": detail, "kind": kind})
122    print(f"  [{'PASS' if ok else 'FAIL'}] {pid:8s} {kind:5s} [{rows}] {label[:110]}" + (f"  --  {detail}" if detail else ""))
123    return ok

125def summary(write=True):
126    n_ok = sum(r["ok"] for r in RESULTS)
127    print(f"\nSUMMARY: {n_ok}/{len(RESULTS)} checks passed ({len(MICRO)} exact micro-checks)" + ("" if n_ok == len(RESULTS) else "  <-- FAILURES"))
128    if write:
129        with open("results.json", "w") as f:
130            json.dump(RESULTS, f, indent=1, ensure_ascii=False)
131    return n_ok == len(RESULTS)
