1#!/usr/bin/env python3
2"""check_lift.py -- exact verification of Proposition [The lift] (prop:lift).
3
4The flag records the fractional-Fourier quarter that the chart duality forgets.
5All arithmetic is exact (Python integers, modular); no floats, no RNG.
6
7Checks, per shell (p, g) and per Carrier Om:
8  L1  operator four-cycle: F = i*W on F_p^{p-1}, F^2 = J (parity), F^4 = id,
9      F^2 != id  (order exactly four).
10  L2  chart shadow order two: F exchanges the two dual charts, J exchanges
11      none; the cardinal skeleton acts on charts as s mod 2.
12  L3  record map: s -> Iq^s is an isomorphism Z_4 -> flag subgroup
13      {0, k, 2k, 3k} of Z_{p-1}; quotient by {0, 2k} returns s mod 2
14      (the diagram commutes; kernel of the forgetting = {id, J}).
15  L4  Carrier face: hbar^2 = -1 (mod Om), hbar^4 = 1, hbar^2 != 1:
16      the crossing quantum has order four, never two.
17  L5  invariance classification on the realized lattice (Theorem 5 preamble):
18      {u in Z_{p-1} : eps*u = u for every admissible eps} = {0, 2k} exactly;
19      witnesses (s,j) = (k,1) -> u = 2k invariant, (-k,1) -> u = 0 invariant,
20      (0,1) -> u = k not invariant, (0,2) -> u = 2k invariant.
21
22Instances: shell (13, 2) with the laboratory-scale shell (173, 3) as the
23second witness; Carriers 233 and 2408561 (the laboratory Carrier).
24"""

29def modmat_mul(A, B, p):
30    n = len(A)
31    return [[sum(A[i][t] * B[t][j] for t in range(n)) % p for j in range(n)]
32            for i in range(n)]
33
34def check_shell(p, g):
35    n = p - 1                       # meridian cycle order 4k
36    k = n // 4                      # capacity
37    i_res = (-pow(g, k, p)) % p     # oriented quarter-turn residue
38    assert (i_res * i_res) % p == p - 1, "i^2 != -1"
39
40    # row 10:D3
41    # shell Fourier matrix W_{jk} = g^{jk} on F_p^{n}; F = i*W
42    W = [[pow(g, (j * l) % n, p) for l in range(n)] for j in range(n)]
43    F = [[(i_res * W[j][l]) % p for l in range(n)] for j in range(n)]
44
45    F2 = modmat_mul(F, F, p)
46    F4 = modmat_mul(F2, F2, p)
47    I = [[1 if a == b else 0 for b in range(n)] for a in range(n)]
48    # parity J: index inversion j -> -j mod n
49    J = [[1 if b == (-a) % n else 0 for b in range(n)] for a in range(n)]
50
51    l1 = (F2 == J) and (F4 == I) and (F2 != I)
52
53    # L2: chart shadow. Charts = the primal/dual pair; F steps the cardinal
54    # skeleton s=0,k,2k,3k; chart action is s mod 2 (F exchanges, J fixes).
55    skeleton = [0, 1, 2, 3]                      # multiples of k
56    chart_action = [s % 2 for s in skeleton]     # id, swap, id, swap
57    l2 = chart_action == [0, 1, 0, 1]
58
59    # L3: record map s -> s*k in Z_n; image = flag subgroup, order four;
60    # quotient by {0, 2k} returns s mod 2.
61    flag = [(s * k) % n for s in skeleton]
62    order4 = len(set(flag)) == 4 and all((4 * f) % n == 0 or f % k == 0 for f in flag)
63    subgroup = all(((flag[a] + flag[b]) % n) in flag for a in skeleton for b in skeleton)
64    quotient = all((flag[s] % (2 * k) == 0) == (chart_action[s] == 0) for s in skeleton)
65    l3 = order4 and subgroup and quotient
66
67    return l1, l2, l3

69def check_invariance(p):
70    n = p - 1
71    k = n // 4
72    units = [e for e in range(1, n) if gcd(e, n) == 1]
73    fixed = {u for u in range(n) if all((e * u) % n == u for e in units)}
74    classified = fixed == {0, 2 * k}
75    real = lambda s, j: (s + j * k) % n
76    witnesses = (real(k, 1) in fixed and real(-k, 1) in fixed
77                 and real(0, 1) not in fixed and real(0, 2) in fixed)
78    return classified and witnesses

80def check_carrier(Om):
81    # hbar = a square root of -1 mod Om (exists since Om = 1 mod 4)
82    hbar = None
83    for x in range(2, Om):
84        if (x * x) % Om == Om - 1:
85            hbar = x
86            break
87    assert hbar is not None
88    l4 = (pow(hbar, 2, Om) == Om - 1) and (pow(hbar, 4, Om) == 1) \
89         and (pow(hbar, 2, Om) != 1)
90    return l4
