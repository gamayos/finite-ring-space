53/-- 2:D6 (the Euler identity on the shell, master C14): with `e = g^i` and `π = 2κ`, `e^(iπ) = (−1)^i`
54for any integer reading `i` of the quarter-turn — `−1` exactly when that reading is odd. -/
55⧉theorem euler_identity (κ : ℕ) (hκ : Fintype.card F = 4 * κ + 1) (g : F)
56    (hg : IsPrimitiveRoot g (Fintype.card F - 1)) (i : ℕ) :
57    (g ^ i) ^ (i * (2 * κ)) = (-1 : F) ^ i := by
58  have hπ := half_period κ hκ g hg

93/-- 2:F3 (Prop. 6.3, inversion): on every shell the inverse of the Fourier matrix `W_g` (`W k j = g^{jk}`) is

95⧉theorem dft_inverse (κ : ℕ) (hκ : Fintype.card F = 4 * κ + 1) (g : F)

102        Matrix (Fin (Fintype.card F - 1)) (Fin (Fintype.card F - 1)) F) k j

104  have : NeZero (Fintype.card F - 1) := ⟨by have := Fintype.one_lt_card (α := F); omega⟩

126/-- 2:E3 (the fixed-shell gap of Prop. 5.5 at `p = 13`, `g = 2`, `H = 6`): no grid point `x / 2^n`

151/-- 2:E3 (the fixed-shell bound, every shell): for any depth cut `m`, every grid point `x / g^n` with
152`x ≤ 2κ` that lies below `1` is at most `max (1 − g^{−m}) (2κ · g^{−(m+1)})`; with `m = ⌊log_g(2κ+1)⌋`
153that bound is below `1` (`fixed_shell_bound_lt_one`), so the interval above it is free at every depth
154and the covering radius of the fixed-shell refinement in `[0, 1]` is bounded below independently of `N`. -/
155⧉theorem fixed_shell_bound (g κ m n x : ℕ) (hg : 2 ≤ g) (hx : x ≤ 2 * κ)
156    (hlt : (x : ℚ) / (g : ℚ) ^ n < 1) :
157    (x : ℚ) / (g : ℚ) ^ n ≤ max (1 - 1 / (g : ℚ) ^ m) (2 * (κ : ℚ) / (g : ℚ) ^ (m + 1)) := by
158  have hg0 : (0 : ℚ) < g := by exact_mod_cast (by omega : 0 < g)
159  have hg1 : (1 : ℚ) ≤ g := by exact_mod_cast (by omega : 1 ≤ g)
160  have hgn : (0 : ℚ) < (g : ℚ) ^ n := pow_pos hg0 n
161  rcases le_or_gt n m with hnm | hnm
162  · refine le_max_of_le_left ?_
163    have hxn : x < g ^ n := by
164      have h := (div_lt_one hgn).mp hlt
165      exact_mod_cast h
166    have hx1 : (x : ℚ) ≤ (g : ℚ) ^ n - 1 := by
167      have h : ((x + 1 : ℕ) : ℚ) ≤ ((g ^ n : ℕ) : ℚ) := by exact_mod_cast hxn
168      push_cast at h; linarith
169    have hgm : (0 : ℚ) < (g : ℚ) ^ m := pow_pos hg0 m
170    have hnm' : (g : ℚ) ^ n ≤ (g : ℚ) ^ m := pow_le_pow_right₀ hg1 hnm
171    have hq : (g : ℚ) ^ n / (g : ℚ) ^ m ≤ 1 := (div_le_one hgm).mpr hnm'
172    rw [div_le_iff₀ hgn]
173    calc (x : ℚ) ≤ (g : ℚ) ^ n - 1 := hx1
174      _ ≤ (g : ℚ) ^ n - (g : ℚ) ^ n / (g : ℚ) ^ m := by linarith
175      _ = (1 - 1 / (g : ℚ) ^ m) * (g : ℚ) ^ n := by ring
176  · refine le_max_of_le_right ?_
177    have hgm1 : (0 : ℚ) < (g : ℚ) ^ (m + 1) := pow_pos hg0 _
178    have hpow : (g : ℚ) ^ (m + 1) ≤ (g : ℚ) ^ n := pow_le_pow_right₀ hg1 hnm
179    have hx' : (x : ℚ) ≤ 2 * (κ : ℚ) := by exact_mod_cast hx
180    calc (x : ℚ) / (g : ℚ) ^ n ≤ (2 * (κ : ℚ)) / (g : ℚ) ^ n :=
181          div_le_div_of_nonneg_right hx' hgn.le
182      _ ≤ (2 * (κ : ℚ)) / (g : ℚ) ^ (m + 1) :=
183          div_le_div_of_nonneg_left (by positivity) hgm1 hpow
