27/-- 5:B5 (decidability): in a finite structure with decidable relations, the truth of every bounded formula is
28decided by evaluation — the quantifiers range over the finitely many elements. -/
29def decRealize {α : Type*} [Fintype M] [DecidableEq M]
30    (hR : ∀ (n : ℕ) (R : L.Relations n) (x : Fin n → M), Decidable (Structure.RelMap R x)) :
31    ∀ {n : ℕ} (φ : L.BoundedFormula α n) (v : α → M) (xs : Fin n → M), Decidable (φ.Realize v xs)
32  | _, .falsum, _, _ => isFalse id
33  | _, .equal t₁ t₂, v, xs =>
34    decidable_of_iff (t₁.realize (Sum.elim v xs) = t₂.realize (Sum.elim v xs)) Iff.rfl
35  | _, .rel R ts, v, xs =>
36    haveI := hR _ R fun i => (ts i).realize (Sum.elim v xs)
37    decidable_of_iff (Structure.RelMap R fun i => (ts i).realize (Sum.elim v xs)) Iff.rfl
38  | _, .imp f₁ f₂, v, xs =>
39    haveI := decRealize hR f₁ v xs
40    haveI := decRealize hR f₂ v xs
41    decidable_of_iff (f₁.Realize v xs → f₂.Realize v xs) Iff.rfl
42  | _, .all f, v, xs =>
43    haveI : ∀ x : M, Decidable (f.Realize v (Fin.snoc xs x)) := fun x => decRealize hR f v (Fin.snoc xs x)
44    decidable_of_iff (∀ x : M, f.Realize v (Fin.snoc xs x)) Iff.rfl

46/-- 5:B5 (Proposition frame-complete, decidability): membership in the theory of a finite structure is decided
47by evaluating the sentence. -/
48@[instance_reducible] def decTheory [Fintype M] [DecidableEq M]
49    (hR : ∀ (n : ℕ) (R : L.Relations n) (x : Fin n → M), Decidable (Structure.RelMap R x)) :
50    DecidablePred (· ∈ L.completeTheory M) := fun φ =>
51  haveI := decRealize hR (α := Empty) φ default default
52  decidable_of_iff (BoundedFormula.Realize φ default default) Iff.rfl

54/-- 5:B3 (Theorem witnessed, exit 2 — `Th(W_N)` does not interpret `Q`): a structure with an injective
55successor that never reaches zero is infinite; every model of `Q` is such a structure, and a finite frame's
56definable quotients are finite. -/
57theorem infinite_of_succ {A : Type*} (S : A → A) (z : A) (hinj : Function.Injective S) (hz : ∀ x, S x ≠ z) :
58    Infinite A := by
59  by_contra hfin
60  have : Finite A := not_infinite_iff_finite.mp hfin
61  obtain ⟨x, hx⟩ := Finite.injective_iff_surjective.mp hinj z
62  exact hz x hx

113/-- 5:E3 (Theorem equivariant-choice, one orbit): for a `G`-family `π : A → I` over a transitive `G`-set `I`, an
114equivariant section exists iff the stabiliser of a base point `i₀` fixes some point of its fibre. -/
115theorem equivariant_section_iff [MulAction.IsPretransitive G I] (π : A → I)
116    (hπ : ∀ (g : G) (a : A), π (g • a) = g • π a) (i₀ : I) :
117    (∃ s : I → A, (∀ i, π (s i) = i) ∧ ∀ (g : G) (i : I), s (g • i) = g • s i) ↔
118      ∃ a, π a = i₀ ∧ ∀ g ∈ MulAction.stabilizer G i₀, g • a = a := by
119  constructor
120  · rintro ⟨s, hs, heq⟩
121    refine ⟨s i₀, hs i₀, fun g hg => ?_⟩
122    rw [← heq, MulAction.mem_stabilizer_iff.mp hg]
123  · rintro ⟨a, ha, hfix⟩
124    classical
125    choose g hg using fun i => MulAction.exists_smul_eq G i₀ i
126    -- `s i := g i • a`, where `g i • i₀ = i`; independence of the choice is the stabiliser condition
127    have key : ∀ (h : G) (i : I), h • i₀ = i → h • a = g i • a := by
128      intro h i hi
129      have hst : (g i)⁻¹ * h ∈ MulAction.stabilizer G i₀ := by
130        rw [MulAction.mem_stabilizer_iff, mul_smul, hi]
131        have e := inv_smul_smul (g i) i₀
132        rw [hg i] at e
133        exact e
134      have := hfix _ hst
135      rw [mul_smul] at this
136      calc h • a = g i • (g i)⁻¹ • h • a := (smul_inv_smul _ _).symm
137        _ = g i • a := by rw [this]
138    refine ⟨fun i => g i • a, fun i => ?_, fun h i => ?_⟩
139    · rw [hπ, ha, hg]
140    · show g (h • i) • a = h • (g i • a)
141      rw [← mul_smul]
142      exact (key (h * g i) (h • i) (by rw [mul_smul, hg])).symm

149`X ⊕ X ↪ X`, so no decomposition reassembles two copies of `X` from pieces of `X`. -/
