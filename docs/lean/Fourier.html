40/-- `g^((-j).val) = (g^(j.val))⁻¹` for an `n`-th root of unity `g`. -/
41⧉lemma pow_neg_val {g : K} (hg : g ^ n = 1) (j : Fin n) :
42    g ^ ((-j : Fin n) : ℕ) = (g ^ (j : ℕ))⁻¹ := by
43  have hne : g ≠ 0 := by rintro rfl; simp [NeZero.ne n] at hg

65/-- `W² = n • J` for any primitive `n`-th root of unity `g` (the general identity). -/
66⧉theorem W_sq (g : K) (hg : IsPrimitiveRoot g n) :
67    (W g : Matrix (Fin n) (Fin n) K) ^ 2 = (n : K) • J := by
68  ext k l
69  rw [sq, Matrix.mul_apply, Matrix.smul_apply, J_apply]
70  simp only [W_apply]
71  have hrw : ∀ j : Fin n, g ^ ((j : ℕ) * (k : ℕ)) * g ^ ((l : ℕ) * (j : ℕ))
72      = (g ^ ((k : ℕ) + (l : ℕ))) ^ (j : ℕ) := by
73    intro j; rw [← pow_add, ← pow_mul]; congr 1; ring
74  simp_rw [hrw]
75  have hdvd_iff : n ∣ (k : ℕ) + (l : ℕ) ↔ l = -k := by
76    constructor
77    · intro hd
78      rw [eq_neg_iff_add_eq_zero, add_comm]
79      apply Fin.ext
80      rw [Fin.val_add, Fin.val_zero]
81      exact Nat.mod_eq_zero_of_dvd hd
82    · intro hl
83      have h : k + l = 0 := by rw [hl, add_neg_cancel]
84      have := congrArg Fin.val h
85      rw [Fin.val_add, Fin.val_zero] at this
86      exact Nat.dvd_of_mod_eq_zero this
87  by_cases hkl : l = -k
88  · obtain ⟨c, hc⟩ := hdvd_iff.2 hkl
89    rw [hc, pow_mul, hg.pow_eq_one, one_pow]
90    simp [hkl]
91  · have hne : g ^ ((k : ℕ) + (l : ℕ)) ≠ 1 := by
92      intro h1
93      exact hkl (hdvd_iff.1 (hg.dvd_of_pow_eq_one _ h1))
94    have hpow : (g ^ ((k : ℕ) + (l : ℕ))) ^ n = 1 := by
95      rw [← pow_mul, mul_comm, pow_mul, hg.pow_eq_one, one_pow]
96    rw [sum_pow_eq_zero_of_ne_one hpow hne]
97    simp [hkl]

99/-- `W J = J W` for any `n`-th root of unity `g`. -/
100⧉theorem W_mul_J_comm (g : K) (hg : g ^ n = 1) :
101    (W g : Matrix (Fin n) (Fin n) K) * J = J * W g := by
102  ext k l
103  rw [Matrix.mul_apply, Matrix.mul_apply]
104  simp only [W_apply, J_apply, mul_ite, ite_mul, mul_one, mul_zero, one_mul, zero_mul]
105  rw [Finset.sum_eq_single (-l), Finset.sum_eq_single (-k)]
106  · simp only [neg_neg, ite_true]
107    -- g^((-l)·k) = g^(l·(-k))
108    have hl : (g ^ (l : ℕ)) ^ n = 1 := by rw [← pow_mul, mul_comm, pow_mul, hg, one_pow]
109    rw [pow_mul, pow_mul, pow_neg_val hg l, inv_pow, pow_neg_val hl k]

113    have : l ≠ -x := fun h => hx (by rw [h, neg_neg])

124⧉lemma natCast_card_pred_eq_neg_one : ((Fintype.card F - 1 : ℕ) : F) = -1 := by

126  rw [Nat.cast_sub h2, FiniteField.cast_card_eq_zero, Nat.cast_one, zero_sub]

128/-- The quarter-turn `i = −g^κ` squares to `−1` (`g^{2κ}` is a primitive square root of unity). -/
129⧉lemma quarter_turn_sq (κ : ℕ) (hκ : Fintype.card F = 4 * κ + 1) (g : F)
130    (hg : IsPrimitiveRoot g (Fintype.card F - 1)) :
131    (-(g ^ κ)) ^ 2 = -1 := by
132  have hκ' : Fintype.card F - 1 = 4 * κ := by omega
133  have hκpos : κ ≠ 0 := by
134    intro h; subst h; have := Fintype.one_lt_card (α := F); omega
135  have hsq : (-(g ^ κ)) ^ 2 = g ^ (2 * κ) := by ring
136  rw [hsq]
137  have h2 : IsPrimitiveRoot (g ^ (2 * κ)) 2 := by
138    have hd : 2 * κ ∣ Fintype.card F - 1 := by rw [hκ']; exact ⟨2, by ring⟩
139    have := hg.pow_of_dvd (show 2 * κ ≠ 0 by omega) hd
140    have hq : (Fintype.card F - 1) / (2 * κ) = 2 := by
141      rw [hκ', show 4 * κ = 2 * κ * 2 by ring]; exact Nat.mul_div_cancel_left _ (by omega)
142    rwa [hq] at this
143  exact h2.eq_neg_one_of_two_right

145/-- 6:B5, 6:B7 — every shell: on a field with `p = 4κ + 1` elements, `n = p − 1`,
146`g` a primitive root and `F = i • W`: `W² = −J`, `F² = J`, `F⁴ = 1`, `W J = J W`. -/
147⧉theorem shell_relations (κ : ℕ) (hκ : Fintype.card F = 4 * κ + 1) (g : F)
148    (hg : IsPrimitiveRoot g (Fintype.card F - 1)) :
149    haveI : NeZero (Fintype.card F - 1) := ⟨by have := Fintype.one_lt_card (α := F); omega⟩
150    (W g : Matrix (Fin (Fintype.card F - 1)) (Fin (Fintype.card F - 1)) F) ^ 2 = -J ∧
151    ((-(g ^ κ)) • W g : Matrix (Fin (Fintype.card F - 1)) (Fin (Fintype.card F - 1)) F) ^ 2 = J ∧
152    ((-(g ^ κ)) • W g : Matrix (Fin (Fintype.card F - 1)) (Fin (Fintype.card F - 1)) F) ^ 4 = 1 ∧
153    (W g : Matrix (Fin (Fintype.card F - 1)) (Fin (Fintype.card F - 1)) F) * J = J * W g := by
154  have : NeZero (Fintype.card F - 1) := ⟨by have := Fintype.one_lt_card (α := F); omega⟩
155  have hW : (W g : Matrix (Fin (Fintype.card F - 1)) (Fin (Fintype.card F - 1)) F) ^ 2 = -J := by
156    rw [W_sq g hg, natCast_card_pred_eq_neg_one, neg_one_smul]
157  have hi := quarter_turn_sq κ hκ g hg
158  have hF : ((-(g ^ κ)) • W g : Matrix (Fin (Fintype.card F - 1)) (Fin (Fintype.card F - 1)) F) ^ 2
159      = J := by
160    rw [smul_pow, hW, hi, neg_one_smul, neg_neg]
161  refine ⟨hW, hF, ?_, W_mul_J_comm g hg.pow_eq_one⟩
162  rw [show (4 : ℕ) = 2 * 2 by norm_num, pow_mul, hF, J_sq]

164/-- 6:B5, 6:B7 on the prime shell `𝔽_p = ZMod p`, an instance: the relations hold there for every prime
165`p = 4κ + 1` and primitive root `g`. -/
166⧉theorem shell_relations_zmod (p κ : ℕ) [hp : Fact (Nat.Prime p)] (hκ : p = 4 * κ + 1)
167    (g : ZMod p) (hg : IsPrimitiveRoot g (p - 1)) :
168    haveI : NeZero (Fintype.card (ZMod p) - 1) :=
169      ⟨by have := Fintype.one_lt_card (α := ZMod p); omega⟩
170    ((-(g ^ κ)) • W g : Matrix (Fin (Fintype.card (ZMod p) - 1))
171      (Fin (Fintype.card (ZMod p) - 1)) (ZMod p)) ^ 2 = J := by
172  have hc : Fintype.card (ZMod p) = p := ZMod.card p
173  exact (shell_relations κ (by rw [hc, hκ]) g (by rw [hc]; exact hg)).2.1
