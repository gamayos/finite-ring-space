691  have hdi : dd i i = 1 := by unfold dd; rw [hinv]; field_simp; ring
692  have hcm : cc (-1 : K) = -1 := by unfold cc; rw [inv_neg, inv_one]; field_simp; ring

695  have hdmi : dd i (-i) = -1 := by unfold dd; rw [inv_neg, hinv]; field_simp; ring

698  · rw [rot, hci, hdi]
699  · rw [rot, hcm, hdm]; ext a b; fin_cases a <;> fin_cases b <;> simp
700  · rw [rot, hcmi, hdmi]; simp

719`SO(2, K) ≃ K^×` when `K` has a quarter-turn. -/

721  toFun u := ⟨(cc (u : K), dd i u), circle i hi h2 u u.ne_zero⟩

723    (by linear_combination x.2 - x.1.2 ^ 2 * hi) (by linear_combination x.2 - x.1.2 ^ 2 * hi)

725    have hi0 := quarter_turn_ne_zero i hi

735    have hinv : (c + i * d)⁻¹ = c - i * d := inv_eq_of_mul_eq_one_right hmul
736    apply Subtype.ext
737    show (cc (c + i * d), dd i (c + i * d)) = (c, d)
738    unfold cc dd

740    refine Prod.ext ?_ ?_
741    · show (c + i * d + (c - i * d)) / 2 = c
742      field_simp; ring
743    · show (c + i * d - (c - i * d)) / (2 * i) = d
744      field_simp; ring
