87/-- 20:B7 — the flat ground state, in any integral domain with a primitive `p`-th root of unity `ω` (the

89ω^{an}` equals `−1` for every `n ≢ 0 (mod p)`, and `p − 1` at the origin; the mode at the spectral origin carries
90no prime information. -/
91⧉theorem ramanujan_sum {p : ℕ} (hp : 0 < p) {ω : R} (hω : IsPrimitiveRoot ω p) (n : ℕ) :
92    (¬ p ∣ n → ∑ a ∈ Ico 1 p, ω ^ (a * n) = -1) ∧
93    (p ∣ n → ∑ a ∈ Ico 1 p, ω ^ (a * n) = ((p - 1 : ℕ) : R)) := by
94  have hsplit : ∑ a ∈ range p, ω ^ (a * n) = 1 + ∑ a ∈ Ico 1 p, ω ^ (a * n) := by
95    obtain ⟨m, rfl⟩ : ∃ m, p = m + 1 := ⟨p - 1, by omega⟩
96    rw [sum_range_succ', sum_Ico_eq_sum_range, Nat.add_sub_cancel, zero_mul, pow_zero, add_comm]
97    congr 1
98    refine sum_congr rfl fun a _ => ?_
99    rw [add_comm 1 a]

101  · intro hn
102    have hx1 : ω ^ n ≠ 1 := fun h => hn ((hω.pow_eq_one_iff_dvd n).1 h)
103    have hxp : (ω ^ n) ^ p = 1 := by rw [← pow_mul, mul_comm, pow_mul, hω.pow_eq_one, one_pow]
104    have hsum : ∑ a ∈ range p, ω ^ (a * n) = 0 := by
105      have h := geom_sum_mul (ω ^ n) p
106      rw [hxp, sub_self] at h
107      have h' : ∑ a ∈ range p, ω ^ (a * n) = ∑ a ∈ range p, (ω ^ n) ^ a :=

441/-! ## The scale-shift on the shell: the `𝔽_p` reading (rows E1, E12(iii), E13) -/

447/-- 20:E1, 20:E12 — the scale-shift `S : x ↦ g^r x` is a permutation of `F^×` with the power characters
448`x ↦ x^k` as eigenvectors in the `𝔽_p` reading, `S x^k = g^{rk} x^k`, and its fixed-point count — the trace of
449its permutation matrix — is `(p − 1)·[(p − 1) ∣ r]`, which carries no prime data. -/
450⧉theorem scale_shift (g : F) (hg : IsPrimitiveRoot g (Fintype.card F - 1)) (r : ℕ) :
451    Function.Bijective (fun x : Fˣ => (Units.mk0 g (hg.ne_zero (by have := Fintype.one_lt_card (α := F); omega))) ^ r * x) ∧

454      if (Fintype.card F - 1) ∣ r then Fintype.card F - 1 else 0 := by
455  refine ⟨(Group.mulLeft_bijective _), fun k x => mul_pow _ _ _, ?_⟩

458    have : univ.filter (fun x : Fˣ => g ^ r * (x : F) = x) = univ := by
459      ext x; simp [h1]

463      rw [Finset.filter_eq_empty_iff]

499/-- The `j`-th chart mode of the cycle `C_n`: `χ_j(m) = e^{2πi·jm/n}`. On the shell, `m` is the discrete
500logarithm of the residue `g^m` and `n = p − 1`. -/
501⧉noncomputable def chi (j m : ZMod n) : ℂ := ZMod.stdAddChar (j * m)

503/-- The coefficient of `v` on the mode `χ_j`: `⟨v, χ_j⟩ = Σ_m v(m) χ_j(m)̄`. -/
504⧉noncomputable def coef (v : ZMod n → ℂ) (j : ZMod n) : ℂ := ∑ m, v m * (starRingEnd ℂ) (chi j m)

509/-- 20:E13 — orthogonality of the chart modes: `Σ_m χ_j(m) χ_k(m)̄ = n·[j = k]`, each of squared norm
510`n = p − 1` (with `chi_inversion`, the `p − 1` characters are an orthogonal basis of `ℂ^{F^×}`; the orthonormal
511system is `χ_j/√n`). -/
512⧉theorem chi_orthogonal (j k : ZMod n) :

514  have : ∀ m, chi j m * (starRingEnd ℂ) (chi k m) = ZMod.stdAddChar (m * (j - k)) := fun m => by

517  simp_rw [this]

522/-- 20:E12, 20:E1 — clause (iii): the scale-shift `m ↦ m + 1` (the residue `g^m ↦ g^{m+1}`) has every chart mode
523as an eigenvector, with eigenphase `e^{2πi·j/n}` independent of the vector: `χ_j(m + 1) = e^{2πi j/n} χ_j(m)`. -/
524⧉theorem chi_shift (j m : ZMod n) :

527  constructor

531/-- 20:E12, 20:E13 — clause (i): Fourier inversion on the cycle, `v = (1/n) Σ_j ⟨v, χ_j⟩ χ_j`, with the constant
532mode carrying the mean, `⟨v, χ_0⟩ = Σ_m v(m)`, so the nontrivial modes carry `v − v̄·1`. -/
533⧉theorem chi_inversion (v : ZMod n → ℂ) :
534    (∀ m, v m = (n : ℂ)⁻¹ * ∑ j, coef v j * chi j m) ∧ coef v 0 = ∑ m, v m ∧
535    ∀ m, v m - (n : ℂ)⁻¹ * ∑ m', v m' = (n : ℂ)⁻¹ * ∑ j ∈ univ.erase 0, coef v j * chi j m := by
536  have hn : (n : ℂ) ≠ 0 := by exact_mod_cast NeZero.ne n
537  have hkey : ∀ m, ∑ j, coef v j * chi j m = (n : ℂ) * v m := by
538    intro m
539    simp only [coef, sum_mul]
540    rw [sum_comm]
541    have : ∀ m', ∑ j, v m' * (starRingEnd ℂ) (chi j m') * chi j m
542        = v m' * ∑ j, chi j m * (starRingEnd ℂ) (chi j m') := fun m' => by
543      rw [mul_sum]; exact sum_congr rfl fun j _ => by ring
544    simp_rw [this]
545    have horth : ∀ m', ∑ j, chi j m * (starRingEnd ℂ) (chi j m') = if m' = m then (n : ℂ) else 0 := by
546      intro m'
547      have : ∀ j, chi j m * (starRingEnd ℂ) (chi j m') = ZMod.stdAddChar (j * (m - m')) := fun j => by
548        rw [chi_conj, chi, ← AddChar.map_add_eq_mul]
549        congr 1; ring
550      simp_rw [this]
551      rw [AddChar.sum_mulShift _ (ZMod.isPrimitive_stdAddChar n), ZMod.card]
552      by_cases h : m' = m
553      · subst h; simp
554      · rw [ite_eq_right (fun e => h (sub_eq_zero.1 e).symm), ite_eq_right h]; simp
555    simp_rw [horth]
556    simp [mul_comm]
557  have hmean : coef v 0 = ∑ m, v m := by
558    simp only [coef, chi, zero_mul, AddChar.map_zero_eq_one, map_one, mul_one]
559  refine ⟨fun m => ?_, hmean, fun m => ?_⟩
560  · rw [hkey, ← mul_assoc, inv_mul_cancel₀ hn, one_mul]
561  · have hsplit := sum_erase_add univ (fun j => coef v j * chi j m) (mem_univ 0)
562    rw [hkey] at hsplit
563    have h0 : coef v 0 * chi 0 m = ∑ m', v m' := by
564      rw [hmean, chi, zero_mul, AddChar.map_zero_eq_one, mul_one]
565    rw [h0] at hsplit
566    have : ∑ j ∈ univ.erase 0, coef v j * chi j m = (n : ℂ) * v m - ∑ m', v m' := by
567      linear_combination hsplit
568    rw [this, mul_sub, ← mul_assoc, inv_mul_cancel₀ hn, one_mul]

570/-- 20:E13, 20:C5 — Parseval on the cycle: `Σ_j |⟨v, χ_j⟩|² = n Σ_m |v(m)|²`; the chart modes are complete. -/
571⧉theorem chi_parseval (v : ZMod n → ℂ) :
572    ∑ j, normSq (coef v j) = n * ∑ m, normSq (v m) := by
573  have h := (chi_inversion v).1
574  apply Complex.ofReal_injective
575  push_cast
576  simp_rw [← Complex.mul_conj]
577  have hc : ∀ j, (starRingEnd ℂ) (coef v j) = ∑ m, (starRingEnd ℂ) (v m) * chi j m := fun j => by
578    simp only [coef, map_sum, map_mul, Complex.conj_conj]
579  calc ∑ j, coef v j * (starRingEnd ℂ) (coef v j)
580      = ∑ j, ∑ m, (starRingEnd ℂ) (v m) * (coef v j * chi j m) := by
581        refine sum_congr rfl fun j _ => ?_
582        rw [hc, mul_sum]; exact sum_congr rfl fun m _ => by ring
583    _ = ∑ m, (starRingEnd ℂ) (v m) * ∑ j, coef v j * chi j m := by
584        rw [sum_comm]; exact sum_congr rfl fun m _ => by rw [mul_sum]
585    _ = ∑ m, (starRingEnd ℂ) (v m) * ((n : ℂ) * v m) := by
586        refine sum_congr rfl fun m _ => ?_
587        congr 1
588        have hn : (n : ℂ) ≠ 0 := by exact_mod_cast NeZero.ne n
589        have := h m
590        rw [this, ← mul_assoc, mul_inv_cancel₀ hn, one_mul]
591    _ = (n : ℂ) * ∑ m, v m * (starRingEnd ℂ) (v m) := by
592        rw [mul_sum]; exact sum_congr rfl fun m _ => by ring

594/-- 20:C5 — mean-square flatness: for a mean-removed vector `u ≠ 0` on the cycle (`Σ_m u(m) = 0`), the
595normalised correlations `r_j = |⟨u, χ_j⟩|/(‖u‖ ‖χ_j‖)` with the nonconstant modes satisfy `Σ_{j≠0} r_j² = 1`
596exactly, so their mean square over the `n − 1 = p − 2` nonconstant modes is `1/(p − 2)`: the root-mean-square
597correlation is `1/√(p − 2)`. -/
598⧉theorem flatness (u : ZMod n → ℂ) (hmean : ∑ m, u m = 0) (hu : u ≠ 0) :
599    ∑ j ∈ univ.erase 0, normSq (coef u j) / (n * ∑ m, normSq (u m)) = 1 ∧

603      by_contra h; push Not at h; exact hu (funext h)
604    exact sum_pos' (fun m _ => normSq_nonneg _) ⟨m, mem_univ _, normSq_pos.2 hm⟩
605  have hn : (0 : ℝ) < n := by exact_mod_cast Nat.pos_of_ne_zero (NeZero.ne n)
606  have hden : (n : ℝ) * ∑ m, normSq (u m) ≠ 0 := (mul_pos hn hpos).ne'
607  have h0 : coef u 0 = 0 := by rw [(chi_inversion u).2.1, hmean]
608  have hsplit := sum_erase_add univ (fun j => normSq (coef u j)) (mem_univ 0)
609  rw [chi_parseval, h0, normSq_zero, add_zero] at hsplit
610  have h1 : ∑ j ∈ univ.erase 0, normSq (coef u j) / (n * ∑ m, normSq (u m)) = 1 := by
611    rw [← sum_div, hsplit, div_self hden]
612  exact ⟨h1, by rw [h1]⟩

641/-- 20:B2 — below the horizon the reading is frame-exact: on any shell `ZMod q` with `H² < q`, residues

644⧉theorem window_readback {q : ℕ} [NeZero q] {H : ℕ} (hH : H * H < q) :
645    (∀ m, m ≤ H → ((m : ZMod q)).val = m) ∧

651  refine ⟨fun m hm => ZMod.val_natCast_of_lt (hle m hm), fun a b ha hb => ?_⟩
652  rw [← Nat.cast_mul, ZMod.val_natCast_of_lt]

655/-- 20:B2 — primality equals irreducibility in the window: for `2 ≤ m ≤ H` and any shell with `H² < q`, `m`
656is prime iff it admits no factorisation `a · b = m` with `2 ≤ a, b ≤ H` read on the shell. -/
657⧉theorem prime_iff_window_irreducible {q : ℕ} [NeZero q] {H : ℕ} (hH : H * H < q) (m : ℕ) (hm2 : 2 ≤ m)
658    (hmH : m ≤ H) :

660  have hrb := window_readback (q := q) hH

662  · rintro hp ⟨a, b, ha2, hb2, haH, hbH, hab⟩
663    have hval := congrArg ZMod.val hab
664    rw [hrb.2 a b haH hbH, hrb.1 m hmH] at hval
665    have hdvd : a ∣ m := ⟨b, hval.symm⟩
666    rcases (Nat.dvd_prime hp).1 hdvd with h | h
667    · omega
668    · rw [h] at hval
669      have hb1 : b = 1 := Nat.eq_of_mul_eq_mul_left (show 0 < m by omega)
670        (show m * b = m * 1 by rw [Nat.mul_one]; exact hval)
671      omega
672  · intro h
673    rw [Nat.prime_def]
674    refine ⟨hm2, fun a ham => ?_⟩
675    by_contra hane
676    push Not at hane
677    obtain ⟨b, hb⟩ := ham
678    have hb0 : b ≠ 0 := by rintro rfl; rw [Nat.mul_zero] at hb; omega
679    have hb1 : b ≠ 1 := by rintro rfl; rw [Nat.mul_one] at hb; exact hane.2 hb.symm
680    have ha0 : a ≠ 0 := by rintro rfl; rw [Nat.zero_mul] at hb; omega
681    apply h
682    refine ⟨a, b, by omega, by omega, le_trans (Nat.le_of_dvd (by omega) ⟨b, hb⟩) hmH,
683      le_trans (Nat.le_of_dvd (by omega) ⟨a, by rw [mul_comm]; exact hb⟩) hmH, ?_⟩
684    rw [← Nat.cast_mul, ← hb]

686/-- 20:B2 — `⌊√p⌋² < p` for a prime `p`: the horizon window `H = ⌊√p⌋` is wrap-free. -/
687⧉lemma sqrt_sq_lt_prime {p : ℕ} (hp : p.Prime) : Nat.sqrt p * Nat.sqrt p < p := by
688  rcases lt_or_eq_of_le (Nat.sqrt_le p) with h | h
689  · exact h
690  · exfalso
691    have hdvd : Nat.sqrt p ∣ p := ⟨Nat.sqrt p, h.symm⟩
