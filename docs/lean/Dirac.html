582/-! ## The Cayley step preserves any Hermitian pairing (8:C4, D10) -/
583section cayley
584variable {R : Type*} [Ring R]
585
586/-- 8:C4, 8:D10 — the exact Hermitian preservation, for any anti-involution `σ` of the operator ring (the
587adjoint of a nondegenerate Hermitian form, or the `X`-adjoint of the spinor form): if `σ h = h`, the scalar
588`a` is central with `σ a = −a`, and `1 − a h` is a unit, then `u = (1 − a h)⁻¹ (1 + a h)` satisfies
589`σ u · u = 1` — the Cayley step is unitary (Theorem `cayley-preservation`). For D10 the premise
590`σ h = h` is the `X`-self-adjointness of `𝒟^s − m` (D9's operator clause, decided by the package at
591`p = 5, 13, 17` and not proved here). -/
592theorem cayley_unitary (σ : R → R) (hmul : ∀ x y, σ (x * y) = σ y * σ x) (hadd : ∀ x y, σ (x + y) = σ x + σ y)
593    (hone : σ 1 = 1) (hneg : ∀ x, σ (-x) = -σ x)
594    {h a : R} (hh : σ h = h) (ha : σ a = -a) (hc : ∀ x, Commute a x)
595    (hu : IsUnit (1 - a * h)) :
596    let u := (hu.unit⁻¹ : Rˣ) * (1 + a * h)
597    σ (u : R) * u = 1 := by
598  intro u
599  -- σ(1 − a h) = 1 + a h and σ(1 + a h) = 1 − a h
600  have hsub : σ (1 - a * h) = 1 + a * h := by
601    rw [sub_eq_add_neg, hadd, hone, hneg, hmul, hh, ha, mul_neg, neg_neg, (hc h).eq]
602  have hadd' : σ (1 + a * h) = 1 - a * h := by
603    rw [hadd, hone, hmul, hh, ha, mul_neg, (hc h).eq, sub_eq_add_neg]
604  -- the inverse commutes with 1 + a h
605  have hcomm : Commute (1 - a * h) (1 + a * h) :=
606    (Commute.one_left _).sub_left ((Commute.one_right (a * h)).add_right (Commute.refl (a * h)))
607  have hinv_comm : Commute ((hu.unit⁻¹ : Rˣ) : R) (1 + a * h) := by
608    have hcu : Commute (hu.unit : R) (1 + a * h) := by rw [IsUnit.unit_spec]; exact hcomm
609    exact (Commute.units_inv_left_iff).2 hcu
610  -- σ of the inverse is the inverse of σ
611  have hσinv : σ ((hu.unit⁻¹ : Rˣ) : R) * (1 + a * h) = 1 := by
612    have e : (1 - a * h) * ((hu.unit⁻¹ : Rˣ) : R) = 1 := by
613      have := hu.unit.mul_inv
614      rw [IsUnit.unit_spec] at this
615      exact this
616    have := congrArg σ e
617    rw [hmul, hsub, hone] at this
618    exact this
619  show σ (((hu.unit⁻¹ : Rˣ) : R) * (1 + a * h)) * (((hu.unit⁻¹ : Rˣ) : R) * (1 + a * h)) = 1
620  rw [hmul, hadd']
621  -- (1 − a h) · σ(inv) · inv · (1 + a h): use σ(inv)(1 + a h) = 1 and commutation
622  calc (1 - a * h) * σ ((hu.unit⁻¹ : Rˣ) : R) * (((hu.unit⁻¹ : Rˣ) : R) * (1 + a * h))
623      = (1 - a * h) * σ ((hu.unit⁻¹ : Rˣ) : R) * ((1 + a * h) * ((hu.unit⁻¹ : Rˣ) : R)) := by
624        rw [hinv_comm.eq]
625    _ = (1 - a * h) * (σ ((hu.unit⁻¹ : Rˣ) : R) * (1 + a * h)) * ((hu.unit⁻¹ : Rˣ) : R) := by
626        simp only [mul_assoc]
627    _ = (1 - a * h) * ((hu.unit⁻¹ : Rˣ) : R) := by rw [hσinv, mul_one]
628    _ = 1 := by
629        have := hu.unit.mul_inv
630        rw [IsUnit.unit_spec] at this
631        exact this

639/-! ## Kinetic Hamiltonians: nilpotent, fully admissible, of period `p` (8:C6, E4) -/
640section kinetic

643/-- 8:E4 (1), 8:C6, 8:D10 — the translation `t` of order `p` (`t^p = 1`) has `(t − 1)^p = 0` in
644characteristic `p`: the kinetic Hamiltonians and the symmetric differences are nilpotent (Theorem
645`period-dichotomy` (1), Corollary `kinetic-admissible`). -/
646theorem translation_nilpotent {t : R} (ht : t ^ p = 1) : (t - 1) ^ p = 0 := by

651`𝒜(H) = K⁻` (Corollary `kinetic-admissible`; `H = −Δ` at `c = −1`). -/

653    (hac : Commute a c) :

655    (a * h) ^ p = 0 ∧ IsNilpotent (a * h) ∧ IsUnit (1 - a * h) := by
656  intro h

662      have : t ^ (p - 1) * t = 1 := by rw [← pow_succ, hp1, ht]
663      have h2 : t ^ (p - 1) * t ^ 2 = t := by rw [pow_two, ← mul_assoc, this, one_mul]
664      calc t ^ (p - 1) * (t - 1) ^ 2 = t ^ (p - 1) * t ^ 2 - 2 * (t ^ (p - 1) * t) + t ^ (p - 1) := by
665            noncomm_ring

667    rw [mul_assoc, e]

669    (hac.mul_right (hat.pow_right _)).mul_right ((hat.sub_right (Commute.one_right a)).pow_right 2)

673      ((hct.sub_right (Commute.one_right c)).pow_right 2).mul_left ((Commute.refl t).pow_left _ |>.sub_right (Commute.one_right _) |>.pow_right 2)
674    exact h1.mul_left h2
675  have key : (a * h) ^ p = 0 := by
676    rw [hfac, ← mul_assoc, hcomm2.mul_pow, ← pow_mul, show 2 * p = p + p by ring, pow_add,
677      translation_nilpotent ht, mul_zero, mul_zero]
678  refine ⟨key, ⟨p, key⟩, IsNilpotent.isUnit_one_sub ⟨p, key⟩⟩

680/-- 8:E4 (1), 8:C6 — the period: for a nilpotent `n` with `n^p = 0` and `2` a unit, the Cayley step
681`u = (1 − n)⁻¹(1 + n)` is unipotent — `u − 1 = 2(1 − n)⁻¹ n`, `(u − 1)^p = 0`, `u^p = 1` — and `u ≠ 1` when
682`n ≠ 0`, so in the prime characteristic `ord(U) = p` exactly (Theorem `period-dichotomy` (1); on `𝔽₁₃`,
683`ord(U_{kw, −Δ}) = 13` for every `k ≠ 0`, Example `cayley-13`). -/

704      _ = 2 * (((hu.unit⁻¹ : Rˣ) : R) * n) := by noncomm_ring
705  have hpow : ((u : R) - 1) ^ p = 0 := by

723/-! ## The Hermitian form on a finite configuration space (8:C2) -/
724section hermitian
725variable {K : Type*} [CommRing K] [StarRing K] {X : Type*} [Fintype X]
726
727/-- `⟨ψ, φ⟩_X = Σ_x ψ(x)* φ(x)` (Definition `hermitian-form`). -/
728def herm (ψ φ : X → K) : K := ∑ x, star (ψ x) * φ x
729
730/-- 8:C2 — Hermitian symmetry `⟨ψ, φ⟩* = ⟨φ, ψ⟩`; linearity in the second slot, conjugate-linearity in the first. -/
731theorem herm_star (ψ φ : X → K) : star (herm ψ φ) = herm φ ψ := by
732  simp [herm, star_sum, star_mul, mul_comm]
733
734theorem herm_add_right (ψ φ φ' : X → K) : herm ψ (φ + φ') = herm ψ φ + herm ψ φ' := by
735  simp [herm, mul_add, Finset.sum_add_distrib]
736
737theorem herm_sub_right (ψ φ φ' : X → K) : herm ψ (φ - φ') = herm ψ φ - herm ψ φ' := by
738  simp [herm, mul_sub, Finset.sum_sub_distrib]
739
740theorem herm_smul_right (ψ φ : X → K) (c : K) : herm ψ (c • φ) = c * herm ψ φ := by
741  simp [herm, Finset.mul_sum, mul_left_comm]
742
743theorem herm_add_left (ψ ψ' φ : X → K) : herm (ψ + ψ') φ = herm ψ φ + herm ψ' φ := by
744  simp [herm, add_mul, Finset.sum_add_distrib]
745
746theorem herm_sub_left (ψ ψ' φ : X → K) : herm (ψ - ψ') φ = herm ψ φ - herm ψ' φ := by
747  simp [herm, sub_mul, Finset.sum_sub_distrib]
748
749theorem herm_smul_left (ψ φ : X → K) (c : K) : herm (c • ψ) φ = star c * herm ψ φ := by
750  simp [herm, Finset.mul_sum, mul_assoc]
751
752/-- 8:C2 — nondegeneracy: `⟨ψ, φ⟩ = 0` for every `φ` forces `ψ = 0` (Lemma `hermitian-nondeg`; the test
753function is the point mass at `x₀`). -/
754theorem herm_nondegenerate [DecidableEq X] (ψ : X → K) (h : ∀ φ, herm ψ φ = 0) : ψ = 0 := by
755  funext x₀
756  have := h (Pi.single x₀ 1)
757  simp [herm, Pi.single_apply] at this
758  have h2 := congrArg star this
759  simpa using h2
760
761/-- The translation `(T_u ψ)(x) = ψ(x + u)` on an additive group. -/
762def translate {G : Type*} [AddGroup G] (u : G) (ψ : G → K) : G → K := fun x => ψ (x + u)
763
764/-- 8:C2 — every translation is unitary for the form, and `T_u† = T_{−u}` (Proposition
765`translation-unitary`). -/
766theorem herm_translate {G : Type*} [AddGroup G] [Fintype G] (u : G) (ψ φ : G → K) :
767    herm (translate u ψ) (translate u φ) = herm ψ φ := by
768  unfold herm translate
769  exact Fintype.sum_equiv (Equiv.addRight u) _ _ (fun x => rfl)
770
771/-- 8:C2 — the adjoint of a translation is the opposite translation, `T_u† = T_{−u}` (Proposition
772`translation-unitary`). -/
773theorem herm_translate_adjoint {G : Type*} [AddGroup G] [Fintype G] (u : G) (ψ φ : G → K) :
774    herm (translate u ψ) φ = herm ψ (translate (-u) φ) := by
775  unfold herm translate
776  refine (Fintype.sum_equiv (Equiv.addRight u) _ _ (fun x => ?_))
777  simp
778
779/-- The finite Laplacian along one direction, `Δ = T_e + T_{−e} − 2`, and the Hamiltonian
780`H = −ϰ Δ + M_V` (`(M_V ψ)(x) = V(x) ψ(x)`). -/
781def laplacian {G : Type*} [AddGroup G] (e : G) (ψ : G → K) : G → K :=
782  translate e ψ + translate (-e) ψ - (2 : K) • ψ
783
784def hamiltonian {G : Type*} [AddGroup G] (e : G) (ϰ : K) (V : G → K) (ψ : G → K) : G → K :=
785  (-ϰ) • laplacian e ψ + fun x => V x * ψ x
786
787/-- 8:C2 — the Laplacian along one direction is self-adjoint, `⟨Δψ, φ⟩ = ⟨ψ, Δφ⟩` (Proposition
788`laplacian-sa`; the `d`-dimensional Laplacian is the sum over the directions), and with it every Hamiltonian
789`−ϰΔ + M_V` with `ϰ` and `V` Frobenius-fixed (Corollary `hamiltonian-sa`). -/
790theorem herm_laplacian {G : Type*} [AddGroup G] [Fintype G] (e : G) (ψ φ : G → K) :
791    herm (laplacian e ψ) φ = herm ψ (laplacian e φ) := by
792  unfold laplacian
793  rw [herm_sub_left, herm_add_left, herm_smul_left, herm_sub_right, herm_add_right, herm_smul_right,
794    herm_translate_adjoint, herm_translate_adjoint (-e), neg_neg, star_ofNat, add_comm (herm ψ (translate (-e) φ))]
795
796/-- 8:C2 — every Hamiltonian `−ϰΔ + M_V` with `ϰ` and `V` Frobenius-fixed is self-adjoint (Corollary
797`hamiltonian-sa`). -/
798theorem herm_hamiltonian {G : Type*} [AddGroup G] [Fintype G] (e : G) (ϰ : K) (hϰ : star ϰ = ϰ)
799    (V : G → K) (hV : ∀ x, star (V x) = V x) (ψ φ : G → K) :
800    herm (hamiltonian e ϰ V ψ) φ = herm ψ (hamiltonian e ϰ V φ) := by
801  unfold hamiltonian
802  rw [herm_add_left, herm_smul_left, herm_laplacian, herm_add_right, herm_smul_right, star_neg, hϰ]
803  congr 1
804  unfold herm
805  refine Finset.sum_congr rfl (fun x _ => ?_)
806  rw [star_mul, hV]; ring

808end hermitian
