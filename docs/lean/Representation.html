20/-- 4:B3 (Definition 2): `E : X → W` is adequate for the observation map `g : Z → X` when `E ∘ g` is a bijection
21`Z → W`; minimality (`W = φ(Z)`) is contained in the bijection. -/
22⧉abbrev Adequate (g : Z → X) (E : X → W) : Prop := Function.Bijective (E ∘ g)

24/-- 4:B4: adequacy makes the observation map injective on the latent domain. -/
25⧉theorem adequate_inj {g : Z → X} {E : X → W} (h : Adequate g E) : Function.Injective g := h.1.of_comp

27/-- 4:B5 (Lemma 2): two adequate representations of one observation map are related by the bijection
28`ψ = φ₂ ∘ φ₁⁻¹ : W₁ ≃ W₂` on the resolved observations, `E₂ ∘ g = ψ ∘ E₁ ∘ g`. -/
29⧉theorem transition_exists {g : Z → X} {E₁ : X → W₁} {E₂ : X → W₂} (h₁ : Adequate g E₁) (h₂ : Adequate g E₂) :
30    ∃ ψ : W₁ ≃ W₂, ∀ z, E₂ (g z) = ψ (E₁ (g z)) := by
31  refine ⟨(Equiv.ofBijective _ h₁).symm.trans (Equiv.ofBijective _ h₂), fun z => ?_⟩
32  show E₂ (g z) = Equiv.ofBijective (E₂ ∘ g) h₂ ((Equiv.ofBijective (E₁ ∘ g) h₁).symm ((E₁ ∘ g) z))
33  rw [Equiv.ofBijective_symm_apply_apply (E₁ ∘ g) h₁ z]
34  rfl

36/-- 4:B5 (the uniqueness clause): a map with that property is determined by it, since `E₁ ∘ g` is onto `W₁`. -/
37⧉theorem transition_unique {g : Z → X} {E₁ : X → W₁} {E₂ : X → W₂} (h₁ : Adequate g E₁) (ψ ψ' : W₁ → W₂)
38    (h : ∀ z, E₂ (g z) = ψ (E₁ (g z))) (h' : ∀ z, E₂ (g z) = ψ' (E₁ (g z))) : ψ = ψ' := by

48/-- 4:B6, 4:C2 (Theorem 1 (iii), the canonical choice): with `incl : Z → U` the inclusion of the latent domain and
49`φ = E ∘ g`, the embedding `ι := incl ∘ φ⁻¹` is injective and `ι ∘ φ = incl` — the chart `ψ_m` is the inclusion. -/
50⧉theorem canonical_iota {incl : Z → U} (hincl : Function.Injective incl) {g : Z → X} {E : X → W}
51    (h : Adequate g E) :
52    Function.Injective (incl ∘ (Equiv.ofBijective _ h).symm) ∧
53      ∀ z, (incl ∘ (Equiv.ofBijective _ h).symm) (E (g z)) = incl z := by
54  refine ⟨hincl.comp (Equiv.ofBijective _ h).symm.injective, fun z => ?_⟩
55  show incl ((Equiv.ofBijective (E ∘ g) h).symm ((E ∘ g) z)) = incl z
56  rw [Equiv.ofBijective_symm_apply_apply (E ∘ g) h z]

58/-- 4:C1 (Theorem 1 (i)): the chart `ψ_m = ι_m ∘ φ_m` is injective for every injective `ι_m`. -/
59⧉theorem chart_injective {ι : W → U} (hι : Function.Injective ι) {g : Z → X} {E : X → W} (h : Adequate g E) :

62/-- 4:C1 (Theorem 1 (ii)): the transition `Ψ_{m→n} = ψ_n ∘ ψ_m⁻¹` is a bijection between the images, carrying
63`ψ_m z` to `ψ_n z`. -/
64⧉theorem transition_images {ψm ψn : Z → U} (hm : Function.Injective ψm) (hn : Function.Injective ψn) :
65    ∃ Ψ : Set.range ψm ≃ Set.range ψn, ∀ z, (Ψ ⟨ψm z, z, rfl⟩ : U) = ψn z := by
66  refine ⟨(Equiv.ofInjective ψm hm).symm.trans (Equiv.ofInjective ψn hn), fun z => ?_⟩
67  show ((Equiv.ofInjective ψn hn) ((Equiv.ofInjective ψm hm).symm ⟨ψm z, z, rfl⟩) : U) = ψn z
68  rw [show (⟨ψm z, z, rfl⟩ : Set.range ψm) = Equiv.ofInjective ψm hm z from rfl, Equiv.symm_apply_apply]
69  rfl

71/-- 4:C3 (Corollary 1): the lift `L = φ⁻¹ ∘ E : X → Z` recovers the latent state, `L ∘ g = id`, and `L x` is the
72unique `z` with `E x = φ z`. -/
73⧉theorem lift {g : Z → X} {E : X → W} (h : Adequate g E) :
74    (∀ z, (Equiv.ofBijective _ h).symm (E (g z)) = z) ∧
75      ∀ x z, E x = (E ∘ g) z ↔ (Equiv.ofBijective _ h).symm (E x) = z := by
76  refine ⟨fun z => Equiv.ofBijective_symm_apply_apply _ h z, fun x z => ⟨fun hx => ?_, fun hz => ?_⟩⟩
77  · rw [hx]; exact Equiv.ofBijective_symm_apply_apply _ h z
78  · rw [← hz]
79    exact (Equiv.apply_symm_apply (Equiv.ofBijective (E ∘ g) h) (E x)).symm

81/-- 4:C4 (Corollary 2): the lifts of two modalities' observations of one latent state agree, so the canonical
82charts `ψ ∘ L_m ∘ g_m` and `ψ ∘ L_n ∘ g_n` both equal `ψ`. -/
83⧉theorem consistency {Xm Xn Wm Wn : Type*} {gm : Z → Xm} {gn : Z → Xn} {Em : Xm → Wm} {En : Xn → Wn}
84    (hm : Adequate gm Em) (hn : Adequate gn En) (z : Z) :
85    (Equiv.ofBijective _ hm).symm (Em (gm z)) = (Equiv.ofBijective _ hn).symm (En (gn z)) := by
86  rw [(lift hm).1 z, (lift hn).1 z]

106/-- 4:E3 (Gödel encoding): over distinct primes `p₁, …, p_d` the code `x ↦ ∏ pᵢ^{xᵢ}` is injective on
107`ℕ^d` — unique factorisation. -/
108⧉theorem godel_injective {d : ℕ} (p : Fin d → ℕ) (hp : ∀ i, (p i).Prime) (hinj : Function.Injective p) :
109    Function.Injective (fun x : Fin d → ℕ => ∏ i, p i ^ x i) := by
110  intro x y hxy
111  have key : ∀ (x : Fin d → ℕ) (j : Fin d), (∏ i, p i ^ x i).factorization (p j) = x j := by
112    intro x j
113    rw [Nat.factorization_prod (fun i _ => pow_ne_zero _ (hp i).ne_zero), Finsupp.finsetSum_apply,
114      Finset.sum_eq_single j]
115    · rw [Nat.factorization_pow, Finsupp.smul_apply, (hp j).factorization, Finsupp.single_eq_same, smul_eq_mul,
116        mul_one]
117    · intro i _ hij
118      rw [Nat.factorization_pow, Finsupp.smul_apply, (hp i).factorization, Finsupp.single_apply,
119        ite_eq_right (fun h => hij (hinj h)), smul_eq_mul, mul_zero]
120    · intro h; exact absurd (Finset.mem_univ j) h
121  funext j
122  have := congrArg (fun n => n.factorization (p j)) hxy
123  simpa only [key] using this

125/-- 4:E3 (the reduction into one field): an injective integer code with values below `q` stays injective

127⧉theorem mod_injective {α : Type*} (f : α → ℕ) (hf : Function.Injective f) (q : ℕ) (hq : ∀ a, f a < q) :
