227/-- Powers of an element of finite order reduce modulo the order. -/
228⧉theorem pow_mod_of_pow_eq_one {ω : Shell q} {m : Nat} (hm : 0 < m) (hω : ω ^ m = 1) (l : Nat) :

231  | ⟨c, hc⟩ =>
232    calc ω ^ l = ω ^ (m * c + l % m) := by rw [← hc]
233      _ = (ω ^ m) ^ c * ω ^ (l % m) := by rw [pow_add, pow_mul]
234      _ = ω ^ (l % m) := by rw [hω, one_pow, one_mul]

236/-- 20:B7 — the flat ground state: in a shell `𝔽_q` with a frame (no zero divisors) and an element `ω` of
237order exactly `m`, the Ramanujan sum `Σ_{a=1}^{m−1} ω^{an}` equals `−1` for every `n ≢ 0 (mod m)`. With
238`m = p` this is `c_p(n) ≡ −1`: the mode at the spectral origin carries no prime information. -/
239⧉theorem ramanujan_sum (F : Frame q κ' h) {m : Nat} (hm : 0 < m) {ω : Shell q} (hω : IsPrimitive ω m)

250    pow_mul ω n (a + 1)

367/-- 20:B8, 20:B9 — the finite critical line: `Tr z = 1` exactly when `Re z = 2⁻¹ = 2κ + 1`; it is the fixed locus

377      rw [two_mul', h, hinv]

462/-- 20:B2 — below the horizon the reading is frame-exact: on any shell with `H² < p`, a residue `m ≤ H` is
463the integer `m`, and the product of two window residues reads back as the integer product. -/
464⧉theorem window_readback {H : Nat} (hH : H * H < p) :
465    (∀ m, m ≤ H → (ofNat m : Shell p).val = m) ∧
466    ∀ a b, a ≤ H → b ≤ H → (ofNat a * ofNat b : Shell p).val = a * b := by

472/-- 20:B2 — primality equals irreducibility in the window: for `m ≤ H` on a shell with `H² < p`, `m` factors
473as `a · b` with `2 ≤ a, b ≤ H` read on the shell exactly when it factors so as an integer. The statement is the
474same on every shell above the window — the Subject `𝔽_p` and the Carrier `𝔽_Ω` read the same primes. -/
475⧉theorem factorisation_iff {H : Nat} (hH : H * H < p) (m : Nat) (hm : m ≤ H) :
476    (∃ a b, 2 ≤ a ∧ 2 ≤ b ∧ a ≤ H ∧ b ≤ H ∧ (ofNat a * ofNat b : Shell p) = ofNat m) ↔
477    ∃ a b, 2 ≤ a ∧ 2 ≤ b ∧ a * b = m := by
478  have hrb := window_readback (p := p) hH
479  constructor
480  · intro h
481    match h with
482    | ⟨a, b, ha, hb, haH, hbH, e⟩ =>
483      refine ⟨a, b, ha, hb, ?_⟩
484      have := val_injective e
485      rw [hrb.2 a b haH hbH, hrb.1 m hm] at this
486      exact this
487  · intro h
488    match h with
489    | ⟨a, b, ha, hb, e⟩ =>
490      have hm' : a * b ≤ H := by rw [e]; exact hm
491      have haab : a ≤ a * b := by
492        have := Nat.mul_le_mul_left a (Nat.le_trans (Nat.le_succ 1) hb)
493        rw [Nat.mul_one] at this; exact this
494      have hbab : b ≤ a * b := by
495        have := Nat.mul_le_mul_right b (Nat.le_trans (Nat.le_succ 1) ha)
496        rw [Nat.one_mul] at this; exact this
497      exact ⟨a, b, ha, hb, Nat.le_trans haab hm', Nat.le_trans hbab hm', by rw [ofNat_mul, e]⟩
498

500`H² < Ω`, a window residue factors on one shell exactly when it factors on the other. -/
501⧉theorem frame_coincidence {Ω : Nat} [Pos Ω] {H : Nat} (hp : H * H < p) (hΩ : H * H < Ω) (m : Nat) (hm : m ≤ H) :
502    (∃ a b, 2 ≤ a ∧ 2 ≤ b ∧ a ≤ H ∧ b ≤ H ∧ (ofNat a * ofNat b : Shell p) = ofNat m) ↔
503    (∃ a b, 2 ≤ a ∧ 2 ≤ b ∧ a ≤ H ∧ b ≤ H ∧ (ofNat a * ofNat b : Shell Ω) = ofNat m) :=
504  (factorisation_iff hp m hm).trans (factorisation_iff hΩ m hm).symm
