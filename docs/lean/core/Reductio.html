25/-- `mx a b`, the larger of two naturals, with its two bounds proved from the definition. -/
26def mx (a b : Nat) : Nat := if a ≤ b then b else a
27
28theorem le_mx_left (a b : Nat) : a ≤ mx a b := by
29  unfold mx
30  match Nat.decLe a b with
31  | isTrue h => rw [ite_eq_left h]; exact h
32  | isFalse h => rw [ite_eq_right h]; exact Nat.le_refl a
33
34theorem le_mx_right (a b : Nat) : b ≤ mx a b := by
35  unfold mx
36  match Nat.decLe a b with
37  | isTrue h => rw [ite_eq_left h]; exact Nat.le_refl b
38  | isFalse h => rw [ite_eq_right h]; exact Nat.le_of_lt (Nat.lt_of_not_le h)
39
40theorem lt_of_mx_lt_left {a b N : Nat} (h : mx a b < N) : a < N := Nat.lt_of_le_of_lt (le_mx_left a b) h
41theorem lt_of_mx_lt_right {a b N : Nat} (h : mx a b < N) : b < N := Nat.lt_of_le_of_lt (le_mx_right a b) h

43/-! ### 5:B6 — the bounded stability schema for an explicit `Δ₀` language -/

45/-- Terms: variables (de Bruijn indices), `0`, `1`, sum, product. -/
46inductive Term where
47  | var : Nat → Term
48  | zero : Term
49  | one : Term
50  | add : Term → Term → Term
51  | mul : Term → Term → Term
52
53/-- `Δ₀` formulas: atoms `s = t`, `s < t`, negation, conjunction, and the bounded quantifier `∀ x ≤ t, φ`
54(`x` is variable `0` in `φ`, the other variables shift up; `t` is read in the outer environment). -/
55inductive Form where
56  | eq : Term → Term → Form
57  | lt : Term → Term → Form
58  | neg : Form → Form
59  | conj : Form → Form → Form
60  | ball : Term → Form → Form
61
62/-- The environment extended by a value for variable `0`. -/
63def cons (x : Nat) (env : Nat → Nat) : Nat → Nat
64  | 0 => x
65  | i + 1 => env i
66
67/-- Terms over `ℕ`. -/
68def evalT (env : Nat → Nat) : Term → Nat
69  | .var i => env i
70  | .zero => 0
71  | .one => 1
72  | .add s t => evalT env s + evalT env t
73  | .mul s t => evalT env s * evalT env t
74
75/-- `f 0 && … && f (n − 1)`. -/
76def allBelow (f : Nat → Bool) : Nat → Bool
77  | 0 => true
78  | n + 1 => f n && allBelow f n
79
80/-- Formulas over `ℕ` (the standard value). -/
81def evalF (env : Nat → Nat) : Form → Bool
82  | .eq s t => decide (evalT env s = evalT env t)
83  | .lt s t => decide (evalT env s < evalT env t)
84  | .neg φ => !(evalF env φ)
85  | .conj φ ψ => evalF env φ && evalF env ψ
86  | .ball t φ => allBelow (fun x => evalF (cons x env) φ) (evalT env t + 1)
87
88/-- Terms in the frame `W_N`: a value is defined only while every intermediate value stays below `N`
89(overflow: `A(x, y, z)` fails for `x + y ≥ N`). -/
90def evalT? (N : Nat) (env : Nat → Nat) : Term → Option Nat
91  | .var i => if env i < N then some (env i) else none
92  | .zero => if 0 < N then some 0 else none
93  | .one => if 1 < N then some 1 else none
94  | .add s t => match evalT? N env s, evalT? N env t with
95      | some a, some b => if a + b < N then some (a + b) else none
96      | some _, none => none
97      | none, some _ => none
98      | none, none => none
99  | .mul s t => match evalT? N env s, evalT? N env t with
100      | some a, some b => if a * b < N then some (a * b) else none
101      | some _, none => none
102      | none, some _ => none
103      | none, none => none
104
105/-- Formulas in the frame `W_N`: an atom with an undefined term is false, a bounded quantifier is guarded by
106the definedness of its bound (the transcription `φ*` of the paper). -/
107def evalF? (N : Nat) (env : Nat → Nat) : Form → Bool
108  | .eq s t => match evalT? N env s, evalT? N env t with
109      | some a, some b => decide (a = b)
110      | some _, none => false
111      | none, some _ => false
112      | none, none => false
113  | .lt s t => match evalT? N env s, evalT? N env t with
114      | some a, some b => decide (a < b)
115      | some _, none => false
116      | none, some _ => false
117      | none, none => false
118  | .neg φ => !(evalF? N env φ)
119  | .conj φ ψ => evalF? N env φ && evalF? N env ψ
120  | .ball t φ => match evalT? N env t with
121      | some b => allBelow (fun x => evalF? N (cons x env) φ) (b + 1)
122      | none => false
123
124/-- The largest value a term's evaluation touches. -/
125def boundT (env : Nat → Nat) : Term → Nat
126  | .var i => env i
127  | .zero => 0
128  | .one => 1
129  | .add s t => mx (mx (boundT env s) (boundT env t)) (evalT env s + evalT env t)
130  | .mul s t => mx (mx (boundT env s) (boundT env t)) (evalT env s * evalT env t)
131
132/-- `max (f 0, …, f (n − 1))`. -/
133def maxBelow (f : Nat → Nat) : Nat → Nat
134  | 0 => 0
135  | n + 1 => mx (f n) (maxBelow f n)
136
137/-- `t(φ)`, the largest integer referenced in the evaluation of `φ` over `ℕ`. -/
138def boundF (env : Nat → Nat) : Form → Nat
139  | .eq s t => mx (boundT env s) (boundT env t)
140  | .lt s t => mx (boundT env s) (boundT env t)
141  | .neg φ => boundF env φ
142  | .conj φ ψ => mx (boundF env φ) (boundF env ψ)
143  | .ball t φ => mx (boundT env t) (maxBelow (fun x => boundF (cons x env) φ) (evalT env t + 1))
144
145theorem le_maxBelow (f : Nat → Nat) : ∀ {n x : Nat}, x < n → f x ≤ maxBelow f n
146  | 0, x, hx => absurd hx (Nat.not_lt_zero x)
147  | n + 1, x, hx =>
148    match Nat.decEq x n with
149    | isTrue e => by rw [e]; exact le_mx_left _ _
150    | isFalse e =>
151      have hlt : x < n := Nat.lt_of_le_of_ne (Nat.le_of_lt_succ hx) e
152      Nat.le_trans (le_maxBelow f hlt) (le_mx_right _ _)
153
154theorem allBelow_congr {f g : Nat → Bool} : ∀ {n : Nat}, (∀ x, x < n → f x = g x) → allBelow f n = allBelow g n
155  | 0, _ => rfl
156  | n + 1, h => by
157    show (f n && allBelow f n) = (g n && allBelow g n)
158    rw [h n (Nat.lt_succ_self n), allBelow_congr (fun x hx => h x (Nat.lt_succ_of_lt hx))]

160/-- 5:B6 (terms) — below its bound a term is defined in the frame, with its standard value. -/
161theorem evalT_frame (N : Nat) (env : Nat → Nat) : ∀ (t : Term), boundT env t < N → evalT? N env t = some (evalT env t)
162  | .var i, h => by
163    have h' : env i < N := h
164    show (if env i < N then some (env i) else none) = some (env i); rw [ite_eq_left h']
165  | .zero, h => by
166    have h' : 0 < N := h
167    show (if 0 < N then some 0 else none) = some 0; rw [ite_eq_left h']
168  | .one, h => by
169    have h' : 1 < N := h
170    show (if 1 < N then some 1 else none) = some 1; rw [ite_eq_left h']
171  | .add s t, h => by
172    have hs := evalT_frame N env s (lt_of_mx_lt_left (lt_of_mx_lt_left h))
173    have ht := evalT_frame N env t (lt_of_mx_lt_right (lt_of_mx_lt_left h))
174    show (match evalT? N env s, evalT? N env t with
175      | some a, some b => if a + b < N then some (a + b) else none
176      | some _, none => none
177      | none, some _ => none
178      | none, none => none) = some (evalT env s + evalT env t)
179    rw [hs, ht]
180    show (if evalT env s + evalT env t < N then some (evalT env s + evalT env t) else none) = some (evalT env s + evalT env t)
181    rw [ite_eq_left (lt_of_mx_lt_right h)]
182  | .mul s t, h => by
183    have hs := evalT_frame N env s (lt_of_mx_lt_left (lt_of_mx_lt_left h))
184    have ht := evalT_frame N env t (lt_of_mx_lt_right (lt_of_mx_lt_left h))
185    show (match evalT? N env s, evalT? N env t with
186      | some a, some b => if a * b < N then some (a * b) else none
187      | some _, none => none
188      | none, some _ => none
189      | none, none => none) = some (evalT env s * evalT env t)
190    rw [hs, ht]
191    show (if evalT env s * evalT env t < N then some (evalT env s * evalT env t) else none) = some (evalT env s * evalT env t)
192    rw [ite_eq_left (lt_of_mx_lt_right h)]

194/-- 5:B6 (Theorem stability, schema form) — every `Δ₀` formula takes its standard value in every frame `W_N`
195with `N > t(φ)`: `W_N ⊨ φ* ⟺ ℕ ⊨ φ`, for every environment, by induction on the formula. -/
196theorem stable (N : Nat) : ∀ (φ : Form) (env : Nat → Nat), boundF env φ < N → evalF? N env φ = evalF env φ
197  | .eq s t, env, h => by
198    show (match evalT? N env s, evalT? N env t with
199      | some a, some b => decide (a = b)
200      | some _, none => false
201      | none, some _ => false
202      | none, none => false) = decide (evalT env s = evalT env t)
203    rw [evalT_frame N env s (lt_of_mx_lt_left h), evalT_frame N env t (lt_of_mx_lt_right h)]
204  | .lt s t, env, h => by
205    show (match evalT? N env s, evalT? N env t with
206      | some a, some b => decide (a < b)
207      | some _, none => false
208      | none, some _ => false
209      | none, none => false) = decide (evalT env s < evalT env t)
210    rw [evalT_frame N env s (lt_of_mx_lt_left h), evalT_frame N env t (lt_of_mx_lt_right h)]
211  | .neg φ, env, h => by
212    show (!(evalF? N env φ)) = !(evalF env φ)
213    rw [stable N φ env h]
214  | .conj φ ψ, env, h => by
215    show (evalF? N env φ && evalF? N env ψ) = (evalF env φ && evalF env ψ)
216    rw [stable N φ env (lt_of_mx_lt_left h), stable N ψ env (lt_of_mx_lt_right h)]
217  | .ball t φ, env, h => by
218    show (match evalT? N env t with
219      | some b => allBelow (fun x => evalF? N (cons x env) φ) (b + 1)
220      | none => false) = allBelow (fun x => evalF (cons x env) φ) (evalT env t + 1)
221    rw [evalT_frame N env t (lt_of_mx_lt_left h)]
222    show allBelow (fun x => evalF? N (cons x env) φ) (evalT env t + 1) = allBelow (fun x => evalF (cons x env) φ) (evalT env t + 1)
223    exact allBelow_congr (fun x hx =>
224      stable N φ (cons x env) (Nat.lt_of_le_of_lt (le_maxBelow (fun x => boundF (cons x env) φ) hx) (lt_of_mx_lt_right h)))

229/-- The numeral `n` as a term, `((0 + 1) + 1) + …`. -/
230def num : Nat → Term
231  | 0 => Term.zero
232  | n + 1 => Term.add (num n) Term.one
233
234/-- `∃ x ≤ t, φ` as `¬∀ x ≤ t, ¬φ`. -/
235def bex (t : Term) (φ : Form) : Form := Form.neg (Form.ball t (Form.neg φ))
236
237/-- "variable `i` is prime": `1 < x ∧ ∀ d ≤ x, ∀ e ≤ x, ¬(d·e = x ∧ d ≠ 1 ∧ d ≠ x)`. -/
238def prime (i : Nat) : Form :=
239  Form.conj (Form.lt Term.one (Term.var i))
240    (Form.ball (Term.var i) (Form.ball (Term.var (i + 1))
241      (Form.neg (Form.conj (Form.eq (Term.mul (Term.var 1) (Term.var 0)) (Term.var (i + 2)))
242        (Form.conj (Form.neg (Form.eq (Term.var 1) Term.one)) (Form.neg (Form.eq (Term.var 1) (Term.var (i + 2)))))))))
243
244/-- "variable `0` is even": `∃ k ≤ m, k + k = m`. -/
245def even0 : Form := bex (Term.var 0) (Form.eq (Term.add (Term.var 0) (Term.var 0)) (Term.var 1))
246
247/-- Goldbach below `M`: every even `m ≤ M` with `4 ≤ m` is a sum of two primes `x, y ≤ m` — a `Δ₀` sentence. -/
248def goldbach (M : Nat) : Form :=
249  Form.ball (num M) (Form.neg (Form.conj (Form.conj even0 (Form.neg (Form.lt (Term.var 0) (num 4))))
250    (Form.neg (bex (Term.var 0) (bex (Term.var 1)
251      (Form.conj (prime 1) (Form.conj (prime 0) (Form.eq (Term.add (Term.var 1) (Term.var 0)) (Term.var 2)))))))))
252
253/-- 5:B6 (Example schema-instance) — Goldbach below `10` is true over `ℕ`, decided by the kernel. -/
254theorem goldbach_ten : evalF (fun _ => 0) (goldbach 10) = true := by decide +kernel
255
256/-- 5:B6 (Example schema-instance) — its bound: the evaluation touches `100 = 10·10` and nothing larger. -/
257theorem goldbach_ten_bound : boundF (fun _ => 0) (goldbach 10) = 100 := by decide +kernel
258
259/-- 5:B6 (Example schema-instance) — hence, by `stable`, every frame `W_N` with `N > 100` gives the sentence its
260standard value; `W_101` does so by direct evaluation as well. -/
261theorem goldbach_ten_frame : evalF? 101 (fun _ => 0) (goldbach 10) = true := by decide +kernel
262
263/-- 5:B6 — the schema instance `St(goldbach 10, M)` for every `M`: all frames `100 < N` agree with `ℕ`. -/
264theorem goldbach_ten_stable (N : Nat) (hN : 100 < N) : evalF? N (fun _ => 0) (goldbach 10) = true := by
265  rw [stable N (goldbach 10) (fun _ => 0) (goldbach_ten_bound ▸ hN)]; exact goldbach_ten

269/-- The number of strings of length at most `K` over `s` letters, `Σ_{i ≤ K} s^i`. -/
270def records (s : Nat) : Nat → Nat
271  | 0 => 1
272  | K + 1 => records s K + s ^ (K + 1)
273
274/-- 5:C2 (the count) — fewer than `s^(K+1)` records exist for an alphabet of `s ≥ 2` letters. -/
275theorem records_lt (s : Nat) (hs : 2 ≤ s) : ∀ K, records s K < s ^ (K + 1)
276  | 0 => by
277    show 1 < s ^ 1
278    rw [Nat.pow_succ, Nat.pow_zero, Nat.one_mul]
279    exact Nat.lt_of_lt_of_le (Nat.lt_succ_self 1) hs
280  | K + 1 => by
281    show records s K + s ^ (K + 1) < s ^ (K + 1 + 1)
282    have h1 : records s K + s ^ (K + 1) < s ^ (K + 1) + s ^ (K + 1) := Nat.add_lt_add_right (records_lt s hs K) _
283    have h2 : s ^ (K + 1) + s ^ (K + 1) ≤ s ^ (K + 1) * s := by
284      rw [← Nat.mul_two]; exact Nat.mul_le_mul_left _ hs
285    exact Nat.lt_of_lt_of_le h1 h2

287/-- 5:C2 (Proposition mirror) — a part with fewer records than the frame has elements holds no injective
288representation of the frame's domain: no `f : [0, N) → [0, R)` is injective when `R < N`. -/
289theorem no_mirror {N R : Nat} (hR : R < N) (f : Nat → Nat) (hf : ∀ i, i < N → f i < R)
290    (hinj : ∀ i j, i < N → j < N → f i = f j → i = j) : False :=
291  have hnd := FRC.Shell.imageList_nodup hinj (Nat.le_refl N)
292  have hlen := FRC.Shell.imageList_length f N
293  have hb : ∀ e, Pigeonhole.mem e (FRC.Shell.imageList f N) → e < R := fun _ he =>
294    match FRC.Shell.mem_imageList he with
295    | ⟨j, hj, ej⟩ => ej ▸ hf j hj
296  have := Pigeonhole.length_le_of_nodup_lt R _ hnd hb
297  Nat.lt_irrefl N (Nat.lt_of_le_of_lt (hlen ▸ this) hR)

299/-! ### 5:C5, 5:E5 — an iteration on finitely many states repeats: periodic König and bounded halting -/
300
301/-- `iter f x k = f^k x`. -/
302def iter (f : Nat → Nat) (x : Nat) : Nat → Nat
303  | 0 => x
304  | k + 1 => f (iter f x k)
305
306theorem iter_lt (f : Nat → Nat) {n : Nat} (hf : ∀ y, y < n → f y < n) {x : Nat} (hx : x < n) :
307    ∀ k, iter f x k < n
308  | 0 => hx
309  | k + 1 => hf _ (iter_lt f hf hx k)
310
311theorem iter_add (f : Nat → Nat) (x : Nat) (i : Nat) : ∀ k, iter f x (i + k) = iter f (iter f x i) k
312  | 0 => rfl
313  | k + 1 => by show f (iter f x (i + k)) = f (iter f (iter f x i) k); rw [iter_add f x i k]
314
315/-- 5:E5, 5:C5 (Lemma periodic König, the pigeonhole) — on `n` states the iteration from `x` revisits a state within
316`n` steps: some `i < j ≤ n` have `f^i x = f^j x`. -/
317theorem repeat_below (f : Nat → Nat) (n : Nat) (hf : ∀ y, y < n → f y < n) (x : Nat) (hx : x < n) :
318    ∃ i j, i < j ∧ j ≤ n ∧ iter f x i = iter f x j :=
319  have dec : Decidable (∃ j, j < n + 1 ∧ ∃ i, i < j ∧ iter f x i = iter f x j) :=
320    @FRC.Shell.decExistsLT (fun j => ∃ i, i < j ∧ iter f x i = iter f x j)
321      (fun j => FRC.Shell.decExistsLT (fun i => iter f x i = iter f x j) j) (n + 1)
322  match dec with
323  | isTrue ⟨j, hj, i, hij, e⟩ => ⟨i, j, hij, Nat.le_of_lt_succ hj, e⟩
324  | isFalse hno =>
325    have hinj : ∀ i j, i < n + 1 → j < n + 1 → iter f x i = iter f x j → i = j := fun i j hi hj e =>
326      match Nat.lt_or_ge i j with
327      | Or.inl hlt => absurd ⟨j, hj, i, hlt, e⟩ hno
328      | Or.inr hge => match Nat.lt_or_ge j i with
329        | Or.inl hlt => absurd ⟨i, hi, j, hlt, e.symm⟩ hno
330        | Or.inr hge' => Nat.le_antisymm hge' hge
331    have hnd := FRC.Shell.imageList_nodup hinj (Nat.le_refl (n + 1))
332    have hb : ∀ e, Pigeonhole.mem e (FRC.Shell.imageList (iter f x) (n + 1)) → e < n := fun _ he =>
333      match FRC.Shell.mem_imageList he with
334      | ⟨k, _, ek⟩ => ek ▸ iter_lt f hf hx k
335    have := Pigeonhole.length_le_of_nodup_lt n _ hnd hb
336    by rw [FRC.Shell.imageList_length] at this; exact absurd this (Nat.not_succ_le_self n)
337
338theorem sub_pos {i j : Nat} (h : i < j) : 0 < j - i :=
339  match Nat.decEq (j - i) 0 with
340  | isTrue e =>
341    have hj : j = i := by rw [← FRC.Nat.add_sub_of_le (Nat.le_of_lt h), e, Nat.add_zero]
342    absurd (hj ▸ h) (Nat.lt_irrefl i)
343  | isFalse e => Nat.pos_of_ne_zero e

345/-- 5:E5 — the walk is eventually periodic: from step `i` on, period `p = j − i` with `0 < p ≤ n`,
346`f^(i+k+p) x = f^(i+k) x` for every `k`. -/
347theorem eventually_periodic (f : Nat → Nat) (n : Nat) (hf : ∀ y, y < n → f y < n) (x : Nat) (hx : x < n) :
348    ∃ i p, 0 < p ∧ p ≤ n ∧ ∀ k, iter f x (i + k + p) = iter f x (i + k) :=
349  match repeat_below f n hf x hx with
350  | ⟨i, j, hij, hjn, e⟩ =>
351    ⟨i, j - i, sub_pos hij, Nat.le_trans (Nat.sub_le j i) hjn, fun k => by
352      rw [Nat.add_right_comm, FRC.Nat.add_sub_of_le (Nat.le_of_lt hij), iter_add f x j k, iter_add f x i k, e]⟩
353
354/-- 5:C5 (horizon separation) — a run on `n` states reaches a state `h` iff it reaches it within `n` steps: halting for bounded space is decided from above by `n` steps of simulation. -/
355theorem halts_iff_halts_below (f : Nat → Nat) (n : Nat) (hf : ∀ y, y < n → f y < n) (h : Nat)
356    (x : Nat) (hx : x < n) : (∃ k, iter f x k = h) ↔ ∃ k, k ≤ n ∧ iter f x k = h :=
357  ⟨fun ⟨k, hk⟩ =>
358    match repeat_below f n hf x hx with
359    | ⟨i, j, hij, hjn, e⟩ =>
360      match Nat.lt_or_ge k j with
361      | Or.inl hkj => ⟨k, Nat.le_of_lt (Nat.lt_of_lt_of_le hkj hjn), hk⟩
362      | Or.inr hjk =>
363        -- the run is periodic from `i` with period `p = j − i`; reduce `k − i` modulo `p`
364        have hp : 0 < j - i := sub_pos hij
365        have per : ∀ q r, iter f x (i + (r + (j - i) * q)) = iter f x (i + r) := fun q r => by
366          induction q with
367          | zero => rw [Nat.mul_zero, Nat.add_zero]
368          | succ q ih =>
369            rw [Nat.mul_succ, ← Nat.add_assoc r ((j - i) * q) (j - i), Nat.add_comm (r + (j - i) * q) (j - i),
370              ← Nat.add_assoc i (j - i) (r + (j - i) * q), FRC.Nat.add_sub_of_le (Nat.le_of_lt hij), iter_add f x j,
371              ← e, ← iter_add f x i, ih]
372        match FRC.Nat.mod_spec (j - i) hp (k - i) with
373        | ⟨q, hq⟩ =>
374          have hr : (k - i) % (j - i) < j - i := Nat.mod_lt _ hp
375          have hk' : k = i + ((k - i) % (j - i) + (j - i) * q) := by
376            rw [Nat.add_comm ((k - i) % (j - i)) _, ← hq, FRC.Nat.add_sub_of_le (Nat.le_trans (Nat.le_of_lt hij) hjk)]
377          ⟨i + (k - i) % (j - i),
378            Nat.le_of_lt (Nat.lt_of_lt_of_le (Nat.add_lt_add_left hr i)
379              (by rw [FRC.Nat.add_sub_of_le (Nat.le_of_lt hij)]; exact hjn)),
380            by rw [← per q, ← hk']; exact hk⟩,
381   fun ⟨k, _, hk⟩ => ⟨k, hk⟩⟩

388/-! ### 5:E1, 5:E2 — definable choice by the least element -/
389
390/-- The least `x < n` with `P x`, if any: the choice function `ch(A) = min A` of Definition global-choice. -/
391def leastBelow (P : Nat → Bool) : Nat → Option Nat
392  | 0 => none
393  | n + 1 => match leastBelow P n with
394    | some x => some x
395    | none => if P n then some n else none
396
397/-- 5:E1 — a nonempty family has a choice: if some `x < n` has `P x`, `leastBelow P n` is defined. -/
398theorem leastBelow_some (P : Nat → Bool) : ∀ (n x : Nat), x < n → P x = true → ∃ y, leastBelow P n = some y
399  | 0, x, hx, _ => absurd hx (Nat.not_lt_zero x)
400  | n + 1, x, hx, hP =>
401    match hl : leastBelow P n with
402    | some y => ⟨y, by
403        show (match leastBelow P n with
404          | some x => some x
405          | none => if P n then some n else none) = some y
406        rw [hl]⟩
407    | none =>
408      match Nat.decEq x n with
409      | isTrue e => ⟨n, by
410          show (match leastBelow P n with
411            | some x => some x
412            | none => if P n then some n else none) = some n
413          rw [hl]
414          show (if P n then some n else none) = some n
415          have hPn : P n = true := e ▸ hP
416          rw [hPn, ite_eq_left rfl]⟩
417      | isFalse e =>
418        absurd (leastBelow_some P n x (Nat.lt_of_le_of_ne (Nat.le_of_lt_succ hx) e) hP)
419          (fun ⟨z, hz⟩ => nomatch (hl ▸ hz : (none : Option Nat) = some z))
420
421/-- 5:E1 (Proposition finite-choice) — the choice is a member, below the bound, and the least one. -/
422theorem leastBelow_spec (P : Nat → Bool) : ∀ (n x : Nat), leastBelow P n = some x →
423    x < n ∧ P x = true ∧ ∀ y, y < x → P y = false
424  | 0, x, h => nomatch (h : (none : Option Nat) = some x)
425  | n + 1, x, h => by
426    have h' : (match leastBelow P n with
427      | some x => some x
428      | none => if P n then some n else none) = some x := h
429    clear h
430    revert h'
431    match hl : leastBelow P n with
432    | some y =>
433      intro h'
434      have e : y = x := Option.some.inj h'
435      have := leastBelow_spec P n y hl
436      exact ⟨Nat.lt_succ_of_lt (e ▸ this.1), e ▸ this.2.1, e ▸ this.2.2⟩
437    | none =>
438      intro h'
439      have h'' : (if P n then some n else none) = some x := h'
440      match hP : P n with
441      | true =>
442        rw [hP, ite_eq_left rfl] at h''
443        have e : n = x := Option.some.inj h''
444        refine ⟨Nat.lt_succ_of_le (Nat.le_of_eq e.symm), e ▸ hP, fun y hy => ?_⟩
445        match hPy : P y with
446        | false => rfl
447        | true =>
448          exact absurd (leastBelow_some P n y (e.symm ▸ hy) hPy)
449            (fun ⟨z, hz⟩ => nomatch (hl ▸ hz : (none : Option Nat) = some z))
450      | false =>
451        rw [hP, ite_eq_right Bool.false_ne_true] at h''
452        exact nomatch (h'' : (none : Option Nat) = some x)
453
454/-- 5:E2 (Lemma periodic choice) — the choice depends on the set alone: pointwise equal families choose alike,
455so an equality-periodic family `A_{i+N} = A_i` has a choice function of the same period. -/
456theorem leastBelow_congr {P Q : Nat → Bool} : ∀ {n : Nat}, (∀ x, x < n → P x = Q x) → leastBelow P n = leastBelow Q n
457  | 0, _ => rfl
458  | n + 1, h => by
459    show (match leastBelow P n with
460      | some x => some x
461      | none => if P n then some n else none) = (match leastBelow Q n with
462      | some x => some x
463      | none => if Q n then some n else none)
464    rw [leastBelow_congr (fun x hx => h x (Nat.lt_succ_of_lt hx)), h n (Nat.lt_succ_self n)]
