3/-!
4# FrcCore.Poly — polynomials over the shell and the root criterion (1:G1)
5
6A polynomial is its coefficient sequence `Nat → Shell p` with a degree bound (`Bound f n`: the coefficients
7beyond `n` vanish); equality is coefficientwise, so no function extensionality is needed.  The Cauchy
8product, evaluation as a finite sum, the evaluation homomorphism (`eval_mul`, by a triangular reindexing of
9sums), synthetic division by `X − a` at a root (`quot_linear_spec`), the root bound (a polynomial of degree
10`n` vanishing at `n + 1` distinct points is zero, `root_bound`), and 1:G1's criterion: `f` has a root iff `f`
11and `X^p − X` share a factor of positive degree (`root_iff_common_factor`).  The reverse direction is
12constructive — the root is found by deciding `∃ i < p, d(i) = 0`.  No axioms.
13-/

20⧉theorem sum_split (f : Nat → Shell p) (n k : Nat) :
21    sumRange f (n + k) = sumRange f n + sumRange (fun t => f (n + t)) k := by
22  induction k with
23  | zero => rw [Nat.add_zero, sumRange_zero, add_zero]
24  | succ k ih =>
25    show sumRange f (n + k + 1) = _
26    rw [sumRange_succ, sumRange_succ, ih, add_assoc]

28/-- The triangular reindexing `Σ_{l<N} Σ_{j≤l} F j (l − j) = Σ_{j<N} Σ_{k<N−j} F j k`. -/
29⧉theorem sum_triangle (F : Nat → Nat → Shell p) (N : Nat) :
30    sumRange (fun l => sumRange (fun j => F j (l - j)) (l + 1)) N =
31    sumRange (fun j => sumRange (fun k => F j k) (N - j)) N := by
32  induction N with
33  | zero => rfl
34  | succ N ih =>
35    rw [sumRange_succ, ih, sumRange_succ (fun j => sumRange (fun k => F j k) (N + 1 - j)) N]
36    have h1 : sumRange (fun j => sumRange (fun k => F j k) (N + 1 - j)) N =
37        sumRange (fun j => sumRange (fun k => F j k) (N - j) + F j (N - j)) N := by
38      apply sum_congr
39      intro j hj
40      show sumRange (fun k => F j k) (N + 1 - j) = sumRange (fun k => F j k) (N - j) + F j (N - j)
41      have e : N + 1 - j = (N - j) + 1 := by
42        have := FRC.Nat.add_sub_of_le (Nat.le_of_lt hj)
43        -- this : j + (N - j) = N
44        calc N + 1 - j = j + (N - j) + 1 - j := by rw [this]
45          _ = (N - j) + 1 + j - j := by rw [Nat.add_comm j (N - j), Nat.add_right_comm (N - j) j 1]
46          _ = (N - j) + 1 := FRC.Nat.add_sub_cancel _ _
47      rw [e, sumRange_succ]
48    rw [h1, sum_add]
49    have e2 : N + 1 - N = 1 := FRC.Nat.add_sub_cancel_left N 1
50    rw [e2, sumRange_succ (fun k => F N k) 0, sumRange_zero, zero_add,
51      sumRange_succ (fun j => F j (N - j)) N, Nat.sub_self, add_assoc]

88⧉theorem mul_bound {d q : Poly p} {m k : Nat} (hd : Bound d m) (hq : Bound q k) : Bound (mul d q) (m + k) := by
89  intro i hi
90  apply sum_zero
91  intro j hj
92  show d j * q (i - j) = 0
93  match Nat.decLe j m with
94  | .isTrue hjm =>
95    have : k < i - j := by
96      apply FRC.Nat.le_sub_of_add_le
97      -- k + 1 + j ≤ i  from  m + k < i and j ≤ m
98      show k + 1 + j ≤ i

100        _ = m + k + 1 := by rw [Nat.add_comm (k + 1) m, Nat.add_assoc]
101        _ ≤ i := hi
102    rw [hq _ this, mul_zero]

105/-- The evaluation is multiplicative: `(f·g)(a) = f(a)·g(a)`, with the bounds added. -/
106⧉theorem eval_mul {f g : Poly p} {n m : Nat} (hf : Bound f n) (hg : Bound g m) (a : Shell p) :
107    eval (mul f g) (n + m) a = eval f n a * eval g m a := by
108  unfold eval mul
109  have h1 : sumRange (fun l => sumRange (fun j => f j * g (l - j)) (l + 1) * a ^ l) (n + m + 1) =
110      sumRange (fun l => sumRange (fun j => f j * a ^ j * (g (l - j) * a ^ (l - j))) (l + 1)) (n + m + 1) := by
111    apply sum_congr
112    intro l _
113    show sumRange (fun j => f j * g (l - j)) (l + 1) * a ^ l =
114      sumRange (fun j => f j * a ^ j * (g (l - j) * a ^ (l - j))) (l + 1)
115    rw [← sum_mul_right]
116    apply sum_congr
117    intro j hj
118    show f j * g (l - j) * a ^ l = f j * a ^ j * (g (l - j) * a ^ (l - j))
119    have : a ^ l = a ^ j * a ^ (l - j) := by rw [← pow_add, FRC.Nat.add_sub_of_le (Nat.le_of_lt_succ hj)]
120    rw [this, mul_assoc, mul_assoc, mul_left_comm (g (l - j))]
121  rw [h1, sum_triangle (fun j k => f j * a ^ j * (g k * a ^ k)) (n + m + 1)]
122  have h2 : sumRange (fun j => sumRange (fun k => f j * a ^ j * (g k * a ^ k)) (n + m + 1 - j)) (n + m + 1) =
123      sumRange (fun j => f j * a ^ j * sumRange (fun i => g i * a ^ i) (m + 1)) (n + m + 1) := by
124    apply sum_congr
125    intro j hj
126    show sumRange (fun k => f j * a ^ j * (g k * a ^ k)) (n + m + 1 - j) =
127      f j * a ^ j * sumRange (fun i => g i * a ^ i) (m + 1)
128    rw [sum_mul_left]
129    match Nat.decLe j n with
130    | .isTrue hjn =>
131      have e : n + m + 1 - j = (n + m - j) + 1 := by
132        have := FRC.Nat.add_sub_of_le (Nat.le_trans hjn (Nat.le_add_right n m))
133        calc n + m + 1 - j = j + (n + m - j) + 1 - j := by rw [this]
134          _ = (n + m - j) + 1 + j - j := by rw [Nat.add_comm j (n + m - j), Nat.add_right_comm (n + m - j) j 1]
135          _ = (n + m - j) + 1 := FRC.Nat.add_sub_cancel _ _
136      rw [e]
137      have hm : m ≤ n + m - j := by
138        apply FRC.Nat.le_sub_of_add_le
139        rw [Nat.add_comm n m]; exact Nat.add_le_add_left hjn m
140      exact congrArg (f j * a ^ j * ·) (eval_bound hg hm a)
141    | .isFalse hjn =>
142      rw [hf j (Nat.lt_of_not_le hjn), zero_mul, zero_mul, zero_mul]
143  rw [h2, sum_mul_right]
144  exact congrArg (· * sumRange (fun i => g i * a ^ i) (m + 1)) (eval_bound hf (Nat.le_add_right n m) a)

173/-- Synthetic division: at a root `a` of `f`, `f = (X − a)·q` with `q` of degree one less. -/
174⧉theorem quot_linear_spec {f : Poly p} {n : Nat} (hf : Bound f n) {a : Shell p} (ha : eval f n a = 0) :
175    Bound (quotLinear f n a) (n - 1) ∧ ∀ i, f i = mul (linear a) (quotLinear f n a) i := by
176  constructor
177  · intro i hi
178    have hni : n ≤ i := by
179      match n, hi with
180      | 0, _ => exact Nat.zero_le i
181      | n' + 1, hi => exact Nat.succ_le_of_lt hi
182    show sumRange (fun t => f (i + 1 + t) * a ^ t) (n - i) = 0
183    rw [FRC.Nat.sub_eq_zero_of_le hni]
184    rfl
185  · intro i
186    match i with
187    | 0 =>
188      rw [mul_linear_zero]
189      -- f 0 + a · q 0 = f(a) = 0
190      have e1 : n + 1 = 1 + n := Nat.add_comm n 1
191      unfold eval at ha
192      rw [e1, sum_split, sumRange_succ, sumRange_zero, zero_add] at ha
193      have ha' : f 0 * a ^ 0 + sumRange (fun t => f (1 + t) * a ^ (1 + t)) n = 0 := ha
194      rw [pow_zero, mul_one] at ha'
195      have h2 : sumRange (fun t => f (1 + t) * a ^ (1 + t)) n = a * quotLinear f n a 0 := by
196        show _ = a * sumRange (fun t => f (0 + 1 + t) * a ^ t) (n - 0)
197        rw [← sum_mul_left]
198        apply sum_congr
199        intro t _
200        show f (1 + t) * a ^ (1 + t) = a * (f (0 + 1 + t) * a ^ t)
201        rw [pow_add, pow_one, mul_left_comm]
202      rw [h2] at ha'
203      exact eq_neg_of_add_eq_zero ha'
204    | i + 1 =>
205      rw [mul_linear_succ]
206      match Nat.decLt i n with
207      | .isTrue hin =>
208        -- n − i = (n − (i + 1)) + 1
209        have hk := FRC.Nat.add_sub_of_le (Nat.succ_le_of_lt hin)
210        -- hk : i + 1 + (n - (i + 1)) = n
211        have e : n - i = (n - (i + 1)) + 1 := by
212          calc n - i = i + 1 + (n - (i + 1)) - i := by rw [hk]
213            _ = i + ((n - (i + 1)) + 1) - i := by rw [Nat.add_assoc, Nat.add_comm 1]
214            _ = (n - (i + 1)) + 1 := FRC.Nat.add_sub_cancel_left _ _
215        show f (i + 1) = sumRange (fun t => f (i + 1 + t) * a ^ t) (n - i) +
216          -(a * sumRange (fun t => f (i + 1 + 1 + t) * a ^ t) (n - (i + 1)))
217        rw [e, Nat.add_comm (n - (i + 1)) 1, sum_split, sumRange_succ, sumRange_zero, zero_add]
218        show f (i + 1) = f (i + 1 + 0) * a ^ 0 + sumRange (fun t => f (i + 1 + (1 + t)) * a ^ (1 + t)) (n - (i + 1)) +
219          -(a * sumRange (fun t => f (i + 1 + 1 + t) * a ^ t) (n - (i + 1)))
220        have h3 : sumRange (fun t => f (i + 1 + (1 + t)) * a ^ (1 + t)) (n - (i + 1)) =
221            a * sumRange (fun t => f (i + 1 + 1 + t) * a ^ t) (n - (i + 1)) := by
222          rw [← sum_mul_left]
223          apply sum_congr
224          intro t _
225          show f (i + 1 + (1 + t)) * a ^ (1 + t) = a * (f (i + 1 + 1 + t) * a ^ t)
226          rw [← Nat.add_assoc, pow_add, pow_one, mul_left_comm]
227        rw [h3, Nat.add_zero, pow_zero, mul_one, add_assoc, add_neg, add_zero]
228      | .isFalse hin =>
229        have hni : n ≤ i := Nat.le_of_not_lt hin
230        have z1 : n - i = 0 := FRC.Nat.sub_eq_zero_of_le hni
231        have z2 : n - (i + 1) = 0 := FRC.Nat.sub_eq_zero_of_le (Nat.le_succ_of_le hni)
232        show f (i + 1) = sumRange (fun t => f (i + 1 + t) * a ^ t) (n - i) +
233          -(a * sumRange (fun t => f (i + 1 + 1 + t) * a ^ t) (n - (i + 1)))
234        rw [z1, z2, sumRange_zero, sumRange_zero, mul_zero, neg_zero, add_zero]
235        exact hf _ (Nat.lt_succ_of_le hni)

250/-- 1:G1, the root bound — a polynomial of degree at most `n` that vanishes at `n + 1` distinct points is
251zero: the distinct roots are counted by the degree. -/
252⧉theorem root_bound (F : Frame p κ g) : ∀ (n : Nat) (f : Poly p), Bound f n → ∀ (r : Nat → Shell p),
253    (∀ i j, i ≤ n → j ≤ n → r i = r j → i = j) → (∀ i, i ≤ n → eval f n (r i) = 0) → ∀ i, f i = 0 := by
254  intro n
255  induction n with
256  | zero =>
257    intro f hf r _ hroot i
258    have h0 := hroot 0 (Nat.le_refl 0)
259    unfold eval at h0
260    rw [sumRange_succ, sumRange_zero, zero_add] at h0
261    have h0' : f 0 * r 0 ^ 0 = 0 := h0
262    rw [pow_zero, mul_one] at h0'
263    match i with
264    | 0 => exact h0'
265    | i + 1 => exact hf _ (Nat.zero_lt_succ i)
266  | succ n ih =>
267    intro f hf r hinj hroot
268    have ha := hroot 0 (Nat.zero_le _)
269    have spec := quot_linear_spec hf ha
270    have hq : Bound (quotLinear f (n + 1) (r 0)) n := spec.1
271    have hf' : ∀ b, eval f (n + 1) b = (b + -(r 0)) * eval (quotLinear f (n + 1) (r 0)) n b := fun b => by
272      rw [eval_congr spec.2, eval_mul_linear hq]
273    have hz : ∀ i, quotLinear f (n + 1) (r 0) i = 0 := by
274      apply ih _ hq (fun k => r (k + 1))
275      · intro i j hi hj e
276        exact Nat.succ.inj (hinj (i + 1) (j + 1) (Nat.succ_le_succ hi) (Nat.succ_le_succ hj) e)
277      · intro k hk
278        have h := hroot (k + 1) (Nat.succ_le_succ hk)
279        rw [hf'] at h
280        match F.mul_eq_zero h with
281        | .inl e =>
282          have : r (k + 1) = r 0 := by
283            calc r (k + 1) = r (k + 1) + 0 := (add_zero _).symm
284              _ = r (k + 1) + (-(r 0) + r 0) := by rw [neg_add]
285              _ = (r (k + 1) + -(r 0)) + r 0 := (add_assoc _ _ _).symm
286              _ = r 0 := by rw [e, zero_add]
287          exact absurd (hinj (k + 1) 0 (Nat.succ_le_succ hk) (Nat.zero_le _) this) (FRC.Nat.succ_ne_zero k)
288        | .inr e => exact e
289    intro i
290    rw [spec.2 i]
291    match i with
292    | 0 => rw [mul_linear_zero, hz, mul_zero, neg_zero]
293    | i + 1 => rw [mul_linear_succ, hz, hz, mul_zero, neg_zero, add_zero]

345/-- 1:G1, the root criterion — `f` has a root in the shell iff `f` and `X^p − X` share a factor of positive
346degree.  Forward: the factor is `X − a` (synthetic division, Fermat).  Backward: a common factor `d` of
347degree `m ≥ 1` with no root would force its cofactor in `X^p − X`, of degree `p − m < p`, to vanish at
348all `p` residues, hence to be zero (`root_bound`), against `X^p − X ≠ 0`; the root of `d` is found by
349deciding `∃ i < p, d(i) = 0`. -/
350⧉theorem root_iff_common_factor (F : Frame p κ g) {f : Poly p} {n : Nat} (hf : Bound f n) :
351    (∃ a, eval f n a = 0) ↔ CommonFactor f n (xpx : Poly p) p := by
352  constructor
353  · intro ⟨a, ha⟩
354    have s1 := quot_linear_spec hf ha
355    have s2 := quot_linear_spec (xpx_bound (p := p)) (eval_xpx F a)
356    exact ⟨linear a, 1, quotLinear f n a, quotLinear xpx p a, Nat.le_refl 1, linear_bound a, F.one_ne_zero,
357      s1.1, s2.1, s1.2, s2.2⟩
358  · intro ⟨d, m, q1, q2, hm, hd, hdm, hq1, hq2, e1, e2⟩
359    -- every residue is a root of `d · q2`
360    have hprod : ∀ a, eval d m a * eval q2 (p - m) a = 0 := fun a => by
361      rw [← eval_mul hd hq2, ← eval_bound (mul_bound hd hq2) (Nat.add_le_add_left (Nat.sub_le p m) m) a,
362        ← eval_congr e2, eval_bound xpx_bound (Nat.le_add_left p m), eval_xpx F]
363    have : ∀ v, Decidable (∃ i, i < p ∧ eval d m (ofNat i) = v) :=
364      fun v => decExistsLT (fun i => eval d m (ofNat i) = v) p
365    match this 0 with
366    | .isTrue ⟨i, _, hi⟩ =>
367      refine ⟨ofNat i, ?_⟩
368      rw [← eval_bound hf (Nat.le_add_right n m), eval_congr e1,
369        eval_bound (mul_bound hd hq1) (by rw [Nat.add_comm n m]; exact Nat.add_le_add_left (Nat.sub_le n m) m),
370        eval_mul hd hq1, hi, zero_mul]
371    | .isFalse hno =>
372      -- `q2` vanishes at the `p − m + 1 ≤ p` residues `0, …, p − m`, so it is zero
373      have hk : p - m < p := Nat.sub_lt Pos.pos hm
374      have hz : ∀ i, q2 i = 0 := by
375        apply root_bound F (p - m) q2 hq2 (fun i => ofNat i)
376        · intro i j hi hj e
377          exact ofNat_inj_lt (Nat.lt_of_le_of_lt hi hk) (Nat.lt_of_le_of_lt hj hk) e
378        · intro i hi
379          match F.mul_eq_zero (hprod (ofNat i)) with
380          | .inl e => exact absurd ⟨i, Nat.lt_of_le_of_lt hi hk, e⟩ hno
381          | .inr e => exact e
382      have : (xpx : Poly p) p = 0 := by
383        rw [e2 p]
384        apply sum_zero
385        intro j _
386        show d j * q2 (p - j) = 0
387        rw [hz, mul_zero]
388      rw [xpx_p] at this
389      exact absurd this F.one_ne_zero
390
391end Frame
392end Poly
393end Shell
394end FRC
