43/-- 1:B2 (Theorem 1 of 1-algebra) — off the fourth roots of unity, `x`, `−x`, `x⁻¹`, `−x⁻¹` are four
44distinct residues (`y` stands for the inverse: `x·y = 1`). -/
45⧉theorem klein_orbit_four (F : Frame p κ g) {x y : Shell p} (hxy : x * y = 1) (h4 : x ^ 4 ≠ 1) :
46    x ≠ -x ∧ x ≠ y ∧ x ≠ -y ∧ -x ≠ y ∧ -x ≠ -y ∧ y ≠ -y := by
47  have hx0 : x ≠ 0 := fun h => F.one_ne_zero (by rw [← hxy, h, zero_mul])
48  have hy0 : y ≠ 0 := fun h => F.one_ne_zero (by rw [← hxy, h, mul_zero])
49  have h4' : x ^ 4 = (x * x) * (x * x) := by
50    rw [show (4 : Nat) = 2 * 2 from rfl, pow_mul, pow_two, pow_two]
51  have hsq : x * x ≠ 1 := fun h => h4 (by rw [h4', h, one_mul])
52  have hsqn : x * x ≠ -1 := fun h => h4 (by rw [h4', h, neg_mul_neg, one_mul])
53  have hxy' : x = y → False := fun e => hsq (by rw [← hxy, e])
54  have hxny : x = -y → False := fun e => hsqn (by
55    have : x * x = x * -y := by rw [← e]
56    rw [this, ← mul_neg, hxy])
57  refine ⟨fun h => hx0 (F.eq_zero_of_eq_neg h), hxy', hxny, ?_, ?_, ?_⟩
58  · intro h; exact hxny (by rw [← neg_neg x, h])
59  · intro h; exact hxy' (by rw [← neg_neg x, h, neg_neg])
60  · intro h; exact hy0 (F.eq_zero_of_eq_neg h)

80/-- 1:D2 (the window law, injectivity) — two window integers `x, y ≤ H` with `2H < p` that read as the same
81residue are equal: `ofNat x = ofNat y → x = y`. (Both are below `p`, so the residues are the integers.) -/
82⧉theorem window_injective {H x y : Nat} (hH : 2 * H < p) (hx : x ≤ H) (hy : y ≤ H)
83    (h : (ofNat x : Shell p) = ofNat y) : x = y := by
84  have hxp : x < p := Nat.lt_of_le_of_lt hx (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
85  have hyp : y < p := Nat.lt_of_le_of_lt hy (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
86  have := val_injective h
87  rw [val_ofNat, val_ofNat, FRC.Nat.mod_eq_of_lt hxp, FRC.Nat.mod_eq_of_lt hyp] at this
88  exact this

90/-- 1:D2, the signed window: `x` and `−y` (`x, y ≤ H`, `2H < p`) read as the same residue only when both
91are zero — the window's positive and negative halves do not overlap. -/
92⧉theorem window_signed {H x y : Nat} (hH : 2 * H < p) (hx : x ≤ H) (hy : y ≤ H)
93    (h : (ofNat x : Shell p) = -(ofNat y)) : x = 0 ∧ y = 0 := by
94  have hxp : x < p := Nat.lt_of_le_of_lt hx (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
95  have hyp : y < p := Nat.lt_of_le_of_lt hy (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
96  have hv := val_injective h
97  rw [val_ofNat, val_neg, val_ofNat, FRC.Nat.mod_eq_of_lt hxp, FRC.Nat.mod_eq_of_lt hyp] at hv
98  -- hv : x = (p - y) % p
99  match Nat.decEq y 0 with
100  | .isTrue hy0 =>
101    rw [hy0, Nat.sub_zero, FRC.Nat.mod_self p hp] at hv
102    exact ⟨hv, hy0⟩
103  | .isFalse hy0 =>
104    have hpy : p - y < p := Nat.sub_lt hp (Nat.pos_of_ne_zero hy0)
105    rw [FRC.Nat.mod_eq_of_lt hpy] at hv
106    -- x = p − y with x ≤ H, y ≤ H gives p = x + y ≤ 2H < p
107    have : p = x + y := by rw [hv, FRC.Nat.sub_add_cancel (Nat.le_of_lt hyp)]
108    have hle : x + y ≤ 2 * H := by rw [Nat.two_mul]; exact Nat.add_le_add hx hy
109    exact absurd (Nat.lt_of_le_of_lt (this ▸ hle) hH) (Nat.lt_irrefl p)

115/-- 1:B2 (Theorem 1, the structural set) — the fourth roots of unity are exactly `1, −1, i, −i`. -/
116⧉theorem fourth_roots (F : Frame p κ g) (x : Shell p) :
117    x ^ 4 = 1 ↔ x = 1 ∨ x = -1 ∨ x = quarterTurn g κ ∨ x = -(quarterTurn g κ) := by
118  have h4 : x ^ 4 = (x * x) * (x * x) := by
119    rw [show (4 : Nat) = 2 * 2 from rfl, pow_mul, pow_two, pow_two]
120  have hi := F.quarter_turn_sq
121  constructor
122  · intro h
123    rw [h4] at h
124    match F.sq_eq_one h with
125    | .inl e => match F.sq_eq_one e with
126      | .inl e1 => exact .inl e1
127      | .inr e1 => exact .inr (.inl e1)
128    | .inr e =>
129      -- x² = −1 = i²: (x + −i)(x + i) = 0
130      have e2 : (x + -(quarterTurn g κ)) * (x + quarterTurn g κ) = 0 := by
131        rw [right_distrib, left_distrib, left_distrib, e, ← neg_mul, ← neg_mul, hi, neg_neg, mul_comm (quarterTurn g κ) x]
132        rw [add_assoc, ← add_assoc (x * quarterTurn g κ), add_neg, zero_add, neg_add]
133      match F.mul_eq_zero e2 with
134      | .inl e3 => exact .inr (.inr (.inl (by
135          calc x = x + 0 := (add_zero x).symm
136            _ = x + (-(quarterTurn g κ) + quarterTurn g κ) := by rw [neg_add]
137            _ = (x + -(quarterTurn g κ)) + quarterTurn g κ := (add_assoc _ _ _).symm
138            _ = quarterTurn g κ := by rw [e3, zero_add])))
139      | .inr e3 => exact .inr (.inr (.inr (eq_neg_of_add_eq_zero e3)))
140  · intro h
141    match h with
142    | .inl e => rw [e, one_pow]
143    | .inr (.inl e) => rw [h4, e, neg_mul_neg, one_mul, one_mul]
144    | .inr (.inr (.inl e)) => rw [h4, e, hi, neg_mul_neg, one_mul]
145    | .inr (.inr (.inr e)) => rw [h4, e, neg_mul_neg, hi, neg_mul_neg, one_mul]

164/-- 1:E2 (Proposition 5 reversed) — on a shell that already has a square root of `−1` the complex chart is
165not a field: `(i, 1)·(−i, 1) = (0, 0)` with both factors nonzero (`X + i` and `X − i` are zero divisors). -/
166⧉theorem complex_chart_zero_divisor (F : Frame p κ g) :
167    cmul (quarterTurn g κ, (1 : Shell p)) (-(quarterTurn g κ), 1) = (0, 0) ∧
168    (quarterTurn g κ, (1 : Shell p)) ≠ (0, 0) ∧ (-(quarterTurn g κ), (1 : Shell p)) ≠ (0, 0) := by
169  refine ⟨?_, fun h => F.one_ne_zero (congrArg Prod.snd h), fun h => F.one_ne_zero (congrArg Prod.snd h)⟩
170  unfold cmul
171  show (quarterTurn g κ * -(quarterTurn g κ) + -(1 * 1), quarterTurn g κ * 1 + 1 * -(quarterTurn g κ)) = (0, 0)
172  rw [← mul_neg, F.quarter_turn_sq, neg_neg, one_mul, add_neg, mul_one, one_mul, add_neg]
