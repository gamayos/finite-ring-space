41/-- 1:B2 (Theorem 1 of 1-algebra) — off the fourth roots of unity, `x`, `−x`, `x⁻¹`, `−x⁻¹` are four
42distinct residues (`y` stands for the inverse: `x·y = 1`). -/
43⧉theorem klein_orbit_four (F : Frame p κ g) {x y : Shell p} (hxy : x * y = 1) (h4 : x ^ 4 ≠ 1) :
44    x ≠ -x ∧ x ≠ y ∧ x ≠ -y ∧ -x ≠ y ∧ -x ≠ -y ∧ y ≠ -y := by
45  have hx0 : x ≠ 0 := fun h => F.one_ne_zero (by rw [← hxy, h, zero_mul])
46  have hy0 : y ≠ 0 := fun h => F.one_ne_zero (by rw [← hxy, h, mul_zero])
47  have h4' : x ^ 4 = (x * x) * (x * x) := by
48    rw [show (4 : Nat) = 2 * 2 from rfl, pow_mul, pow_two, pow_two]
49  have hsq : x * x ≠ 1 := fun h => h4 (by rw [h4', h, one_mul])
50  have hsqn : x * x ≠ -1 := fun h => h4 (by rw [h4', h, neg_mul_neg, one_mul])
51  have hxy' : x = y → False := fun e => hsq (by rw [← hxy, e])
52  have hxny : x = -y → False := fun e => hsqn (by
53    have : x * x = x * -y := by rw [← e]
54    rw [this, ← mul_neg, hxy])
55  refine ⟨fun h => hx0 (F.eq_zero_of_eq_neg h), hxy', hxny, ?_, ?_, ?_⟩
56  · intro h; exact hxny (by rw [← neg_neg x, h])
57  · intro h; exact hxy' (by rw [← neg_neg x, h, neg_neg])
58  · intro h; exact hy0 (F.eq_zero_of_eq_neg h)

66/-- The affine frame `(a, b)` of 1-algebra (Definition 2): the transported product

70/-- 1:B4 (Definition 2 of 1-algebra, corrected) — in the affine frame `(a, b)` the multiplicative unit
71is `a + b`, not `b`: `(a + b) ⊗ z = z` for every `z`. -/
72⧉theorem affine_frame_unit {a b y : Shell p} (hby : b * y = 1) (z : Shell p) :
73    affineMul a b y (a + b) z = z := by
74  unfold affineMul

76  rw [e1, hby, mul_one, mul_left_comm b, hby, mul_one, add_comm z (-a), ← add_assoc, add_neg, zero_add]

78/-- 1:D2 (the window law, injectivity) — two window integers `x, y ≤ H` with `2H < p` that read as the same
79residue are equal: `ofNat x = ofNat y → x = y`. (Both are below `p`, so the residues are the integers.) -/
80⧉theorem window_injective {H x y : Nat} (hH : 2 * H < p) (hx : x ≤ H) (hy : y ≤ H)
81    (h : (ofNat x : Shell p) = ofNat y) : x = y := by
82  have hxp : x < p := Nat.lt_of_le_of_lt hx (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
83  have hyp : y < p := Nat.lt_of_le_of_lt hy (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
84  have := val_injective h
85  rw [val_ofNat, val_ofNat, FRC.Nat.mod_eq_of_lt hxp, FRC.Nat.mod_eq_of_lt hyp] at this
86  exact this

88/-- 1:D2, the signed window: `x` and `−y` (`x, y ≤ H`, `2H < p`) read as the same residue only when both
89are zero — the window's positive and negative halves do not overlap. -/
90⧉theorem window_signed {H x y : Nat} (hH : 2 * H < p) (hx : x ≤ H) (hy : y ≤ H)
91    (h : (ofNat x : Shell p) = -(ofNat y)) : x = 0 ∧ y = 0 := by
92  have hxp : x < p := Nat.lt_of_le_of_lt hx (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
93  have hyp : y < p := Nat.lt_of_le_of_lt hy (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
94  have hv := val_injective h
95  rw [val_ofNat, val_neg, val_ofNat, FRC.Nat.mod_eq_of_lt hxp, FRC.Nat.mod_eq_of_lt hyp] at hv
96  -- hv : x = (p - y) % p
97  match Nat.decEq y 0 with
98  | .isTrue hy0 =>
99    rw [hy0, Nat.sub_zero, FRC.Nat.mod_self p hp] at hv
100    exact ⟨hv, hy0⟩
101  | .isFalse hy0 =>
102    have hpy : p - y < p := Nat.sub_lt hp (Nat.pos_of_ne_zero hy0)
103    rw [FRC.Nat.mod_eq_of_lt hpy] at hv
104    -- x = p − y with x ≤ H, y ≤ H gives p = x + y ≤ 2H < p
105    have : p = x + y := by rw [hv, FRC.Nat.sub_add_cancel (Nat.le_of_lt hyp)]
106    have hle : x + y ≤ 2 * H := by rw [Nat.two_mul]; exact Nat.add_le_add hx hy
107    exact absurd (Nat.lt_of_le_of_lt (this ▸ hle) hH) (Nat.lt_irrefl p)
