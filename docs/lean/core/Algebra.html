18/-- 1:B2 (Theorem 1 of 1-algebra), 2:D7 — off the fourth roots of unity, `x`, `−x`, `x⁻¹`, `−x⁻¹` are four
19distinct residues (`y` stands for the inverse: `x·y = 1`). -/
20theorem klein_orbit_four (F : Frame p κ g) {x y : Shell p} (hxy : x * y = 1) (h4 : x ^ 4 ≠ 1) :
21    x ≠ -x ∧ x ≠ y ∧ x ≠ -y ∧ -x ≠ y ∧ -x ≠ -y ∧ y ≠ -y := by
22  have hx0 : x ≠ 0 := fun h => F.one_ne_zero (by rw [← hxy, h, zero_mul])
23  have hy0 : y ≠ 0 := fun h => F.one_ne_zero (by rw [← hxy, h, mul_zero])
24  have h4' : x ^ 4 = (x * x) * (x * x) := by
25    rw [show (4 : Nat) = 2 * 2 from rfl, pow_mul, pow_two, pow_two]
26  have hsq : x * x ≠ 1 := fun h => h4 (by rw [h4', h, one_mul])
27  have hsqn : x * x ≠ -1 := fun h => h4 (by rw [h4', h, neg_mul_neg, one_mul])
28  have hxy' : x = y → False := fun e => hsq (by rw [← hxy, e])
29  have hxny : x = -y → False := fun e => hsqn (by
30    have : x * x = x * -y := by rw [← e]
31    rw [this, ← mul_neg, hxy])
32  refine ⟨fun h => hx0 (F.eq_zero_of_eq_neg h), hxy', hxny, ?_, ?_, ?_⟩
33  · intro h; exact hxny (by rw [← neg_neg x, h])
34  · intro h; exact hxy' (by rw [← neg_neg x, h, neg_neg])
35  · intro h; exact hy0 (F.eq_zero_of_eq_neg h)

55/-- 1:D2 (the window law, injectivity) — two window integers `x, y ≤ H` with `2H < p` that read as the same
56residue are equal: `ofNat x = ofNat y → x = y`. (Both are below `p`, so the residues are the integers.) -/
57theorem window_injective {H x y : Nat} (hH : 2 * H < p) (hx : x ≤ H) (hy : y ≤ H)
58    (h : (ofNat x : Shell p) = ofNat y) : x = y := by
59  have hxp : x < p := Nat.lt_of_le_of_lt hx (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
60  have hyp : y < p := Nat.lt_of_le_of_lt hy (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
61  have := val_injective h
62  rw [val_ofNat, val_ofNat, FRC.Nat.mod_eq_of_lt hxp, FRC.Nat.mod_eq_of_lt hyp] at this
63  exact this

85/-- 1:D2, the signed window: `x` and `−y` (`x, y ≤ H`, `2H < p`) read as the same residue only when both
86are zero — the window's positive and negative halves do not overlap. -/
87theorem window_signed {H x y : Nat} (hH : 2 * H < p) (hx : x ≤ H) (hy : y ≤ H)
88    (h : (ofNat x : Shell p) = -(ofNat y)) : x = 0 ∧ y = 0 := by
89  have hxp : x < p := Nat.lt_of_le_of_lt hx (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
90  have hyp : y < p := Nat.lt_of_le_of_lt hy (Nat.lt_of_le_of_lt (Nat.le_add_left H H) (Nat.two_mul H ▸ hH))
91  have hv := val_injective h
92  rw [val_ofNat, val_neg, val_ofNat, FRC.Nat.mod_eq_of_lt hxp, FRC.Nat.mod_eq_of_lt hyp] at hv
93  -- hv : x = (p - y) % p
94  match Nat.decEq y 0 with
95  | .isTrue hy0 =>
96    rw [hy0, Nat.sub_zero, FRC.Nat.mod_self p hp] at hv
97    exact ⟨hv, hy0⟩
98  | .isFalse hy0 =>
99    have hpy : p - y < p := Nat.sub_lt hp (Nat.pos_of_ne_zero hy0)
100    rw [FRC.Nat.mod_eq_of_lt hpy] at hv
101    -- x = p − y with x ≤ H, y ≤ H gives p = x + y ≤ 2H < p
102    have : p = x + y := by rw [hv, FRC.Nat.sub_add_cancel (Nat.le_of_lt hyp)]
103    have hle : x + y ≤ 2 * H := by rw [Nat.two_mul]; exact Nat.add_le_add hx hy
104    exact absurd (Nat.lt_of_le_of_lt (this ▸ hle) hH) (Nat.lt_irrefl p)

110/-- 1:B2 (Theorem 1, the structural set) — the fourth roots of unity are exactly `1, −1, i, −i`. -/
111theorem fourth_roots (F : Frame p κ g) (x : Shell p) :
112    x ^ 4 = 1 ↔ x = 1 ∨ x = -1 ∨ x = quarterTurn g κ ∨ x = -(quarterTurn g κ) := by
113  have h4 : x ^ 4 = (x * x) * (x * x) := by
114    rw [show (4 : Nat) = 2 * 2 from rfl, pow_mul, pow_two, pow_two]
115  have hi := F.quarter_turn_sq
116  constructor
117  · intro h
118    rw [h4] at h
119    match F.sq_eq_one h with
120    | .inl e => match F.sq_eq_one e with
121      | .inl e1 => exact .inl e1
122      | .inr e1 => exact .inr (.inl e1)
123    | .inr e =>
124      -- x² = −1 = i²: (x + −i)(x + i) = 0
125      have e2 : (x + -(quarterTurn g κ)) * (x + quarterTurn g κ) = 0 := by
126        rw [right_distrib, left_distrib, left_distrib, e, ← neg_mul, ← neg_mul, hi, neg_neg, mul_comm (quarterTurn g κ) x]
127        rw [add_assoc, ← add_assoc (x * quarterTurn g κ), add_neg, zero_add, neg_add]
128      match F.mul_eq_zero e2 with
129      | .inl e3 => exact .inr (.inr (.inl (by
130          calc x = x + 0 := (add_zero x).symm
131            _ = x + (-(quarterTurn g κ) + quarterTurn g κ) := by rw [neg_add]
132            _ = (x + -(quarterTurn g κ)) + quarterTurn g κ := (add_assoc _ _ _).symm
133            _ = quarterTurn g κ := by rw [e3, zero_add])))
134      | .inr e3 => exact .inr (.inr (.inr (eq_neg_of_add_eq_zero e3)))
135  · intro h
136    match h with
137    | .inl e => rw [e, one_pow]
138    | .inr (.inl e) => rw [h4, e, neg_mul_neg, one_mul, one_mul]
139    | .inr (.inr (.inl e)) => rw [h4, e, hi, neg_mul_neg, one_mul]
140    | .inr (.inr (.inr e)) => rw [h4, e, neg_mul_neg, hi, neg_mul_neg, one_mul]

153not a field: `(i, 1)·(−i, 1) = (0, 0)` with both factors nonzero (`X + i` and `X − i` are zero divisors). -/
154theorem complex_chart_zero_divisor (F : Frame p κ g) :

179/-- The exponent `m` of `x = g^m` reduced to its orbit representative in `[1, κ)`. -/

194      refine ⟨κ + κ - m, ?_, ?_, Or.inr (Or.inr (Or.inr ?_))⟩
195      · refine Nat.lt_of_add_lt_add_right (n := m) ?_
196        rw [Nat.zero_add, FRC.Nat.sub_add_cancel (Nat.le_of_lt hlt)]; exact hlt
197      · refine Nat.lt_of_add_lt_add_right (n := m) ?_
198        rw [FRC.Nat.sub_add_cancel (Nat.le_of_lt hlt)]; exact Nat.add_lt_add_left hgt κ
199      · rw [← mul_neg, ← pow_add, FRC.Nat.sub_add_cancel (Nat.le_of_lt hlt), hπ, neg_neg]
200    | Or.inr hge2 => match Nat.lt_or_ge m (κ + κ + κ) with

208          rw [FRC.Nat.sub_add_cancel hge2, Nat.add_comm κ (κ + κ)]; exact hlt

253/-- The orbit relation is transitive. -/
254theorem inOrbit_trans {a b c : Shell p} (h1 : InOrbit a b) (h2 : InOrbit b c) : InOrbit a c := by
255  have hab : b = a ∨ b = -a ∨ a * b = 1 ∨ a * -b = 1 := h1
256  have hbc : c = b ∨ c = -b ∨ b * c = 1 ∨ b * -c = 1 := h2
257  unfold InOrbit
258  match hab, hbc with
259  | Or.inl e, h => exact e ▸ h
260  | Or.inr (Or.inl e), Or.inl e' => exact Or.inr (Or.inl (e' ▸ e))
261  | Or.inr (Or.inl e), Or.inr (Or.inl e') => exact Or.inl (by rw [e', e, neg_neg])
262  | Or.inr (Or.inl e), Or.inr (Or.inr (Or.inl e')) => exact Or.inr (Or.inr (Or.inr (by rw [← mul_neg, neg_mul, ← e]; exact e')))
263  | Or.inr (Or.inl e), Or.inr (Or.inr (Or.inr e')) => exact Or.inr (Or.inr (Or.inl (by rw [← neg_mul_neg, ← e]; exact e')))
264  | Or.inr (Or.inr (Or.inl e)), Or.inl e' => exact Or.inr (Or.inr (Or.inl (e' ▸ e)))
265  | Or.inr (Or.inr (Or.inl e)), Or.inr (Or.inl e') => exact Or.inr (Or.inr (Or.inr (by rw [e', neg_neg]; exact e)))
266  | Or.inr (Or.inr (Or.inl e)), Or.inr (Or.inr (Or.inl e')) =>
267    exact Or.inl (inv_unique (x := c) (x' := a) (y := b) (by rw [mul_comm]; exact e') e)
268  | Or.inr (Or.inr (Or.inl e)), Or.inr (Or.inr (Or.inr e')) =>
269    exact Or.inr (Or.inl (by
270      have := inv_unique (x := -c) (x' := a) (y := b) (by rw [mul_comm]; exact e') e
271      rw [← this, neg_neg]))
272  | Or.inr (Or.inr (Or.inr e)), Or.inl e' => exact Or.inr (Or.inr (Or.inr (e' ▸ e)))
273  | Or.inr (Or.inr (Or.inr e)), Or.inr (Or.inl e') => exact Or.inr (Or.inr (Or.inl (by rw [e']; exact e)))
274  | Or.inr (Or.inr (Or.inr e)), Or.inr (Or.inr (Or.inl e')) =>
275    exact Or.inr (Or.inl (inv_unique (x := -a) (x' := c) (y := b) (neg_mul_of_mul_neg e) (by rw [mul_comm]; exact e')).symm)
276  | Or.inr (Or.inr (Or.inr e)), Or.inr (Or.inr (Or.inr e')) =>
277    exact Or.inl (neg_eq_neg (inv_unique (x := -a) (x' := -c) (y := b) (neg_mul_of_mul_neg e) (by rw [mul_comm]; exact e'))).symm

279/-- The exponents of the orbit of `g^r`: `g^s ∈ orbit(g^r)`, `s < n`, forces `s ∈ {r, r + 2κ, 4κ − r, 2κ − r}`
280(for `1 ≤ r < κ`). -/
281theorem orbit_exponent (F : Frame p κ g) {r s : Nat} (hr1 : 1 ≤ r) (hrκ : r < κ) (hs : s < p - 1)
282    (h : InOrbit (g ^ r) (g ^ s)) : s = r ∨ s = r + (κ + κ) ∨ s = κ + κ + κ + κ - r ∨ s = κ + κ - r := by
283  have hn := F.n_eq
284  have hπ := F.half_period
285  rw [two_mul_eq] at hπ
286  rw [hn, four_mul_eq] at hs
287  have hr2 : r < κ + κ := Nat.lt_of_lt_of_le hrκ (Nat.le_add_right κ κ)
288  have hr4 : r < κ + κ + κ + κ := Nat.lt_of_lt_of_le hr2 (Nat.le_trans (Nat.le_add_right _ κ) (Nat.le_add_right _ κ))
289  have hpow_inj : ∀ {i j : Nat}, i < κ + κ + κ + κ → j < κ + κ + κ + κ → g ^ i = g ^ j → i = j :=
290    fun hi hj e => F.pow_inj (by rw [hn, four_mul_eq]; exact hi) (by rw [hn, four_mul_eq]; exact hj) e
291  have hpn : g ^ (κ + κ + κ + κ) = 1 := by rw [← four_mul_eq, ← hn]; exact F.pow_n
292  match h with
293  | Or.inl e => exact Or.inl (hpow_inj hs hr4 e)
294  | Or.inr (Or.inl e) =>
295    have e' : g ^ s = g ^ (r + (κ + κ)) := by rw [e, pow_add, hπ, mul_comm, neg_one_mul]
296    have hlt : r + (κ + κ) < κ + κ + κ + κ := by
297      rw [show κ + κ + κ + κ = (κ + κ) + (κ + κ) by rw [Nat.add_assoc]]
298      exact Nat.add_lt_add_right hr2 _
299    exact Or.inr (Or.inl (hpow_inj hs hlt e'))
300  | Or.inr (Or.inr (Or.inl e)) =>
301    have hlt : κ + κ + κ + κ - r < κ + κ + κ + κ := Nat.sub_lt (Nat.lt_of_lt_of_le (Nat.zero_lt_succ 0) (Nat.le_trans hr1 (Nat.le_of_lt hr4))) hr1
302    have e2 : g ^ r * g ^ (κ + κ + κ + κ - r) = 1 := by rw [← pow_add, FRC.Nat.add_sub_of_le (Nat.le_of_lt hr4), hpn]
303    have e' : g ^ s = g ^ (κ + κ + κ + κ - r) := F.mul_left_cancel (F.pow_ne_zero r) (e.trans e2.symm)
304    exact Or.inr (Or.inr (Or.inl (hpow_inj hs hlt e')))
305  | Or.inr (Or.inr (Or.inr e)) =>
306    have hlt : κ + κ - r < κ + κ + κ + κ := Nat.lt_of_le_of_lt (Nat.sub_le _ _)
307      (Nat.lt_of_lt_of_le (Nat.lt_add_of_pos_right F.cap_pos) (Nat.le_add_right _ κ))
308    have e2 : g ^ r * -(g ^ (κ + κ - r)) = 1 := by
309      rw [← mul_neg, ← pow_add, FRC.Nat.add_sub_of_le (Nat.le_of_lt hr2), hπ, neg_neg]
310    have e' : -(g ^ s) = -(g ^ (κ + κ - r)) := F.mul_left_cancel (F.pow_ne_zero r) (e.trans e2.symm)
311    exact Or.inr (Or.inr (Or.inr (hpow_inj hs hlt (neg_eq_neg e'))))

313/-- 1:B2 (Theorem 1, the count, uniqueness), 2:D7 — two representatives `g^r`, `g^{r'}` with `1 ≤ r, r' < κ` whose
314orbits meet are the same: the Klein orbits off `Q₄` are exactly `κ − 1`, one for each `r ∈ [1, κ)`. -/
315theorem orbit_rep_unique (F : Frame p κ g) {r r' : Nat} (hr1 : 1 ≤ r) (hrκ : r < κ) (_hr1' : 1 ≤ r')
316    (hrκ' : r' < κ) {x : Shell p} (hx : InOrbit (g ^ r) x) (hx' : InOrbit (g ^ r') x) : r = r' := by
317  have hn := F.n_eq
318  have hr' : r' < p - 1 := by
319    rw [hn, four_mul_eq]
320    exact Nat.lt_of_lt_of_le hrκ' (Nat.le_trans (Nat.le_add_right κ κ) (Nat.le_trans (Nat.le_add_right _ κ) (Nat.le_add_right _ κ)))
321  have h := inOrbit_trans hx (inOrbit_symm hx')
322  match F.orbit_exponent hr1 hrκ hr' h with
323  | Or.inl e => exact e.symm
324  | Or.inr (Or.inl e) =>
325    exact absurd hrκ' (Nat.not_lt_of_le (by rw [e]; exact Nat.le_trans (Nat.le_add_right κ κ) (Nat.le_add_left _ r)))
326  | Or.inr (Or.inr (Or.inl e)) =>
327    -- 4κ − r > κ since r < κ
328    have : κ ≤ κ + κ + κ + κ - r := by
329      apply FRC.Nat.le_sub_of_add_le
330      rw [Nat.add_assoc, Nat.add_assoc]
331      exact Nat.add_le_add_left (Nat.le_trans (Nat.le_of_lt hrκ) (Nat.le_add_right κ _)) κ
332    exact absurd hrκ' (Nat.not_lt_of_le (e ▸ this))
333  | Or.inr (Or.inr (Or.inr e)) =>
334    -- 2κ − r > κ since r < κ
335    have : κ ≤ κ + κ - r := by
336      apply FRC.Nat.le_sub_of_add_le
337      exact Nat.add_le_add_left (Nat.le_of_lt hrκ) κ
338    exact absurd hrκ' (Nat.not_lt_of_le (e ▸ this))

340/-! ### 1:C2 — the frame group: the affine maps `x ↦ a + b·x`, `b ≠ 0`, act simply transitively on the frames -/

342/-- An affine map of the shell, `x ↦ a + b·x` with `b ≠ 0`. -/
343structure Affine (p : Nat) [Pos p] where
344  a : Shell p
345  b : Shell p
346  hb : b ≠ 0
347
348namespace Affine
349
350def apply (φ : Affine p) (x : Shell p) : Shell p := φ.a + φ.b * x
351
352theorem ext' {φ ψ : Affine p} (ha : φ.a = ψ.a) (hb : φ.b = ψ.b) : φ = ψ := by
353  cases φ; cases ψ; cases ha; cases hb; rfl
354
355/-- The identity `x ↦ 0 + 1·x`. -/
356def one (F : Frame p κ g) : Affine p := ⟨0, 1, F.one_ne_zero⟩
357
358/-- Composition: `(a, b) ∘ (c, d) = (a + b·c, b·d)`. -/
359def comp (F : Frame p κ g) (φ ψ : Affine p) : Affine p := ⟨φ.a + φ.b * ψ.a, φ.b * ψ.b, F.mul_ne_zero φ.hb ψ.hb⟩
360
361theorem comp_apply (F : Frame p κ g) (φ ψ : Affine p) (x : Shell p) :
362    (comp F φ ψ).apply x = φ.apply (ψ.apply x) := by
363  unfold comp apply
364  show φ.a + φ.b * ψ.a + φ.b * ψ.b * x = φ.a + φ.b * (ψ.a + ψ.b * x)
365  rw [left_distrib, add_assoc, mul_assoc]
366
367theorem one_apply (F : Frame p κ g) (x : Shell p) : (one F).apply x = x := by
368  unfold one apply; show 0 + 1 * x = x; rw [one_mul, zero_add]
369
370theorem comp_assoc (F : Frame p κ g) (φ ψ χ : Affine p) : comp F (comp F φ ψ) χ = comp F φ (comp F ψ χ) :=
371  ext' (by show φ.a + φ.b * ψ.a + φ.b * ψ.b * χ.a = φ.a + φ.b * (ψ.a + ψ.b * χ.a); rw [left_distrib, add_assoc, mul_assoc])
372    (by show φ.b * ψ.b * χ.b = φ.b * (ψ.b * χ.b); exact mul_assoc _ _ _)
373
374/-- The inverse of `(a, b)`: `(−a·b⁻¹, b⁻¹)`, with `y` the inverse of `b`. -/
375def inv (F : Frame p κ g) (φ : Affine p) (y : Shell p) (hy : φ.b * y = 1) : Affine p :=
376  ⟨-(φ.a * y), y, fun h => F.one_ne_zero (by rw [← hy, h, mul_zero])⟩
377
378theorem comp_inv (F : Frame p κ g) (φ : Affine p) (y : Shell p) (hy : φ.b * y = 1) :
379    comp F φ (inv F φ y hy) = one F :=
380  ext' (by
381      show φ.a + φ.b * -(φ.a * y) = 0
382      calc φ.a + φ.b * -(φ.a * y) = φ.a + -(φ.b * (φ.a * y)) := by rw [← mul_neg]
383        _ = φ.a + -(φ.a * (φ.b * y)) := by rw [mul_left_comm]
384        _ = φ.a + -φ.a := by rw [hy, mul_one]
385        _ = 0 := add_neg _)
386    (by show φ.b * y = 1; exact hy)
387
388/-- 1:C2, 4:C5 (the frame group), simple transitivity — for frames `(a, b)` and `(c, d)` (`b, d ≠ 0`) there
389is exactly one affine map carrying the first to the second: `φ (a, b) := (φ.apply a, φ.b · b)`. -/
390theorem simply_transitive (F : Frame p κ g) (a b c d : Shell p) (hb : b ≠ 0) (hd : d ≠ 0) :
391    (∃ φ : Affine p, φ.apply a = c ∧ φ.b * b = d) ∧
392    (∀ φ ψ : Affine p, φ.apply a = c → φ.b * b = d → ψ.apply a = c → ψ.b * b = d → φ = ψ) := by
393  match F.exists_inv hb with
394  | ⟨y, hy⟩ =>
395    constructor
396    · refine ⟨⟨c + -(d * y * a), d * y, ?_⟩, ?_, ?_⟩
397      · exact F.mul_ne_zero hd (fun h => F.one_ne_zero (by rw [← hy, h, mul_zero]))
398      · show c + -(d * y * a) + d * y * a = c
399        rw [add_assoc, neg_add, add_zero]
400      · show d * y * b = d
401        rw [mul_assoc, mul_comm y b, hy, mul_one]
402    · intro φ ψ h1 h2 h3 h4
403      have eb : φ.b = ψ.b := by
404        calc φ.b = φ.b * (b * y) := by rw [hy, mul_one]
405          _ = (φ.b * b) * y := (mul_assoc _ _ _).symm
406          _ = (ψ.b * b) * y := by rw [h2, h4]
407          _ = ψ.b := by rw [mul_assoc, hy, mul_one]
408      have ea : φ.a = ψ.a := by
409        have e1 : φ.a + φ.b * a = c := h1
410        have e2 : ψ.a + ψ.b * a = c := h3
411        rw [← eb] at e2
412        exact add_right_cancel (e1.trans e2.symm)
413      exact ext' ea eb
414
415end Affine

417/-- The number of `x < n` with `P x`, as a sum of `0`s and `1`s. -/
418def natCount (P : Nat → Prop) [DecidablePred P] : Nat → Nat
419  | 0 => 0
420  | n + 1 => natCount P n + if P n then 1 else 0
421
422theorem natCount_ne_zero (n : Nat) : natCount (fun x => x ≠ 0) (n + 1) = n := by
423  induction n with
424  | zero => rfl
425  | succ n ih =>
426    show natCount (fun x => x ≠ 0) (n + 1) + (if n + 1 ≠ 0 then 1 else 0) = n + 1
427    rw [ih, ite_eq_left (Nat.succ_ne_zero n)]

429/-- 1:C2, 4:C5, the order — the frames `(a, b)`, `b ≠ 0`, number `p·(p − 1)`: `p` choices of the origin, `p − 1`
430of the unit. -/
431theorem frame_count (_F : Frame p κ g) :
432    natCount (fun _ => True) p * natCount (fun b => b ≠ 0) p = p * (p - 1) := by
433  have h1 : ∀ n, natCount (fun _ => True) n = n := fun n => by
434    induction n with
435    | zero => rfl
436    | succ n ih => show natCount (fun _ => True) n + (if True then 1 else 0) = n + 1; rw [ih, ite_eq_left trivial]
437  have hp : p = (p - 1) + 1 := (FRC.Nat.sub_add_cancel Pos.pos).symm
438  rw [h1]
439  have h2 : natCount (fun b => b ≠ 0) p = p - 1 := by
440    have := natCount_ne_zero (p - 1)
441    rw [← hp] at this; exact this
442  rw [h2]
