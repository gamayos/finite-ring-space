3/-!
4# FrcCore.Complex — the orbital shell and its spherical completion, coded and decided
5
6The completion of the orbital shell `S_p` (Def. 3.1 of 2-geometry) as a finite cell complex on vertex codes:
7`N = 0`, the interior vertices `(a, m)` for `1 ≤ a ≤ π − 1`, `m < n`, and `S`; the faces as cyclic vertex lists
8(north triangles, quadrilaterals, south triangles); the edges as the faces' sides. `closed p` checks that every
9edge lies in exactly two faces and that every vertex link is a single cycle (2:C3, the exhaustive incidence of
10the ledger); `cellular p u` checks whether the reindexing `ρ_u` carries faces to faces (2:B4); `poleExchange`
11checks the meridian reversal on the completion. Everything is a computation the kernel evaluates by `decide`.
12No axioms.
13-/

18/-- The vertex code of `(a, m)` on the shell with `n` phases: `1 + (a − 1)·n + m`. -/
19⧉def vtx (n a m : Nat) : Nat := 1 + (a - 1) * n + m % n
20
21/-- The south pole: the vertex after the last interior latitude `a = π − 1`. -/
22⧉def south (n pi : Nat) : Nat := 1 + (pi - 1) * n
23
24/-- The faces of the completion, as vertex lists. -/
25⧉def faces (n pi : Nat) : List (List Nat) :=
26  let north := (List.range n).map fun m => [0, vtx n 1 m, vtx n 1 (m + 1)]
27  let quads := ((List.range (pi - 2)).map fun a' => (List.range n).map fun m =>
28      [vtx n (a' + 1) m, vtx n (a' + 2) m, vtx n (a' + 2) (m + 1), vtx n (a' + 1) (m + 1)]).foldr (· ++ ·) []
29  let souths := (List.range n).map fun m => [vtx n (pi - 1) m, south n pi, vtx n (pi - 1) (m + 1)]
30  north ++ quads ++ souths
31
32/-- The sides of a face, as unordered pairs `(min, max)`. -/
33⧉def sides : List Nat → List (Nat × Nat)
34  | [] => []
35  | f@(v :: _) =>
36    let rec go : List Nat → List (Nat × Nat)
37      | [] => []
38      | [w] => [(min w v, max w v)]
39      | w :: (w' :: rest) => (min w w', max w w') :: go (w' :: rest)
40    go f
41
42⧉def vertexCount (n pi : Nat) : Nat := south n pi + 1
43
44/-- A face with its sides computed once: `(vertices, sides)`. -/
45⧉abbrev Face := List Nat × List (Nat × Nat)
46
47⧉def withSides (fs : List (List Nat)) : List Face := fs.map fun f => (f, sides f)
48
49/-- Two faces are adjacent around `v` when they share a side through `v`. -/
50⧉def adjacentAt (v : Nat) (f f' : Face) : Bool :=
51  f.2.any fun s => (s.1 == v || s.2 == v) && f'.2.contains s
52
53/-- The neighbours of `cur` around `v` other than `prev` (and `cur` itself). -/
54⧉def nextFaces (v : Nat) (L : List Face) (prev cur : Face) : List Face :=
55  L.filter fun f' => !(f'.1 == cur.1) && !(f'.1 == prev.1) && adjacentAt v cur f'
56
57/-- Walk the link from `f0` along `cur`, never turning back, for `fuel` steps; the number of steps taken until
58`f0` is reached again (`fuel` if never). -/
59⧉def walk (v : Nat) (L : List Face) (f0 : Face) : Nat → Face → Face → Nat → Nat
60  | 0, _, _, k => k
61  | fuel + 1, prev, cur, k =>
62    match nextFaces v L prev cur with
63    | f' :: [] => if f'.1 == f0.1 then k + 1 else walk v L f0 fuel cur f' (k + 1)
64    | [] => fuel + k + 1
65    | _ :: _ :: _ => fuel + k + 1
66
67/-- The link of `v` is a single cycle: every face around `v` has exactly two neighbours there, and the walk from
68the first face returns to it after exactly `|L|` steps. -/
69⧉def linkIsCycle (fs : List Face) (v : Nat) : Bool :=
70  let L := fs.filter fun f => f.1.contains v
71  match L with
72  | [] => false
73  | f0 :: _ =>
74    L.all (fun f => (L.filter (fun f' => !(f'.1 == f.1) && adjacentAt v f f')).length == 2) &&
75    (match nextFaces v L f0 f0 with
76     | f1 :: _ => walk v L f0 L.length f0 f1 1 == L.length
77     | [] => false)
78
79/-- The number of occurrences of `s` in `E`. -/
80⧉def count (s : Nat × Nat) : List (Nat × Nat) → Nat
81  | [] => 0
82  | e :: E => (if e == s then 1 else 0) + count s E
83
84/-- Every side of every face lies in exactly two faces: the sides are bucketed by their smaller vertex and each
85side occurs exactly twice in its bucket. -/
86⧉def edgesInTwoFaces (fs : List (List Nat)) : Bool :=
87  let E := (withSides fs).foldr (fun f acc => f.2 ++ acc) []
88  let V := fs.foldr (fun f acc => f.foldr (fun w acc' => if acc'.contains w then acc' else w :: acc') acc) []
89  V.all fun v =>
90    let B := E.filter fun s => s.1 == v
91    B.all fun s => count s B == 2
92
93/-- 2:C3, the exhaustive incidence: the completion is a closed surface. -/
94⧉def closed (p : Nat) : Bool :=
95  let n := p - 1
96  let pi := n / 2
97  let fs := faces n pi
98  edgesInTwoFaces fs && (List.range (vertexCount n pi)).all fun v => linkIsCycle (withSides fs) v
99
100/-- The reindexing `ρ_u` on vertex codes: `(a, m) ↦ (a, u·m mod n)`, the poles fixed. -/
101⧉def rhoV (n pi u : Nat) (v : Nat) : Nat :=
102  if v == 0 then 0 else if v == south n pi then v else
103    let a := (v - 1) / n + 1
104    let m := (v - 1) % n
105    vtx n a (u * m)
106
107/-- The meridian reversal on the completion: `N ↔ S`, `(a, m) ↦ (π − a, m)`. -/
108⧉def sigmaV (n pi : Nat) (v : Nat) : Nat :=
109  if v == 0 then south n pi else if v == south n pi then 0 else
110    let a := (v - 1) / n + 1
111    let m := (v - 1) % n
112    vtx n (pi - a) m
113
114/-- A vertex map is cellular when it carries every face onto a face (as vertex sets). -/
115⧉def cellularMap (fs : List (List Nat)) (φ : Nat → Nat) : Bool :=
116  fs.all fun f => fs.any fun f' => (f.map φ).all (fun w => f'.contains w) && f'.all (fun w => (f.map φ).contains w)
117
118/-- 2:B4 — whether `ρ_u` is a cellular automorphism of the completion. -/
119⧉def cellular (p u : Nat) : Bool :=
120  let n := p - 1
121  let pi := n / 2
122  cellularMap (faces n pi) (rhoV n pi u)
123
124/-- 2:B4 — whether the pole exchange is cellular on the completion. -/
125⧉def poleExchange (p : Nat) : Bool :=
126  let n := p - 1
127  let pi := n / 2
128  cellularMap (faces n pi) (sigmaV n pi)

142/-- The orbital shell before the collapse: the terminal latitude `a = π` kept, so the last ring of quadrilaterals
143ends on it and nothing closes over it. -/
144⧉def facesOpen (n pi : Nat) : List (List Nat) :=
145  let north := (List.range n).map fun m => [0, vtx n 1 m, vtx n 1 (m + 1)]
146  let quads := ((List.range (pi - 1)).map fun a' => (List.range n).map fun m =>
147      [vtx n (a' + 1) m, vtx n (a' + 2) m, vtx n (a' + 2) (m + 1), vtx n (a' + 1) (m + 1)]).foldr (· ++ ·) []
148  north ++ quads

150/-- 2:C3, 2:B4 [value] — the shell itself is not closed (its terminal latitude is a boundary: those sides lie in
151one face only), so the meridian reversal has nothing to act on before the collapse. -/
152⧉theorem open13 : edgesInTwoFaces (facesOpen 12 6) = false ∧ (facesOpen 12 6).length = 72 := by decide +kernel
