17/-- 2:C2 (Remark 3.4) — the counts: with `π = m + 1` phases-deep and `n = p − 1` phases,
18`|V| − |E| + |F| = 1` for the shell and `2` for its completion, written without subtraction:
19`(πn + 1) + πn = 2πn + 1` and `((π−1)n + 2) + πn = (2π−1)n + 2`. -/
20⧉theorem euler_characteristic (m n : Nat) :
21    ((m + 1) * n + 1) + (m + 1) * n = 2 * ((m + 1) * n) + 1 ∧
22    (m * n + 2) + (m + 1) * n = (2 * m + 1) * n + 2 := by

28/-- 2:E3 — the fixed-shell gap at `p = 13`, `g = 2` (window `x ≤ 6`): no grid point `x / 2^n`
29lies strictly between `3/4` and `1` at any depth `n` — the covering radius of the fixed-shell
30refinement in `[0, 1]` is at least `1/8` for every depth. -/
31⧉theorem fixed_shell_gap (n x : Nat) (hx : x ≤ 6) : ¬ (3 * 2 ^ n < 4 * x ∧ x < 2 ^ n) := by
32  intro h
33  match Nat.lt_or_ge n 3 with
34  | .inl hn =>
35    exact (by decide : ∀ n, n < 3 → ∀ x, x ≤ 6 → ¬ (3 * 2 ^ n < 4 * x ∧ x < 2 ^ n)) n hn x hx h
36  | .inr hn =>
37    have h8 : 2 ^ 3 ≤ 2 ^ n := Nat.pow_le_pow_right (Nat.zero_lt_succ 1) hn
38    have h24 : 4 * x ≤ 4 * 6 := Nat.mul_le_mul_left 4 hx
39    have h24' : 3 * 2 ^ 3 ≤ 3 * 2 ^ n := Nat.mul_le_mul_left 3 h8
40    exact Nat.lt_irrefl _ (Nat.lt_of_lt_of_le (Nat.lt_of_le_of_lt h24' h.1) h24)

42/-- 2:E3 (the fixed-shell bound, every shell) — for `g ≥ 2`, any cut `m`, depth `n` and `x ≤ 2κ` with
43`x < g^n` (the grid point `x/g^n` below `1`): either `x/g^n ≤ 1 − g^{−m}`, i.e. `x·g^m + g^n ≤ g^m·g^n`,
44or `x/g^n ≤ 2κ·g^{−(m+1)}`, i.e. `x·g^{m+1} ≤ 2κ·g^n`. With `2κ + 1 < g^{m+1}` both bounds are below `1`, so the
45covering radius of the fixed-shell refinement in `[0, 1]` is bounded below at every depth. -/
46⧉theorem fixed_shell_bound (g κ m n x : Nat) (hg : 2 ≤ g) (hx : x ≤ 2 * κ) (hlt : x < g ^ n) :
47    x * g ^ m + g ^ n ≤ g ^ m * g ^ n ∨ x * g ^ (m + 1) ≤ 2 * κ * g ^ n := by
48  have hg1 : 1 ≤ g := Nat.le_trans (Nat.le_succ 1) hg
49  match Nat.lt_or_ge n (m + 1) with
50  | .inl hnm =>
51    -- n ≤ m: x + 1 ≤ g^n, so x·g^m + g^m ≤ g^n·g^m, and g^n ≤ g^m
52    refine .inl ?_
53    have hn : n ≤ m := Nat.le_of_lt_succ hnm
54    have h1 : (x + 1) * g ^ m ≤ g ^ n * g ^ m := Nat.mul_le_mul_right _ hlt
55    have h2 : g ^ n ≤ g ^ m := Nat.pow_le_pow_right (Nat.lt_of_lt_of_le (Nat.zero_lt_succ 0) hg1) hn
56    rw [FRC.Nat.add_mul, Nat.one_mul] at h1
57    calc x * g ^ m + g ^ n ≤ x * g ^ m + g ^ m := Nat.add_le_add_left h2 _
58      _ ≤ g ^ n * g ^ m := h1
59      _ = g ^ m * g ^ n := Nat.mul_comm _ _
60  | .inr hnm =>
61    refine .inr ?_
62    have h1 : g ^ (m + 1) ≤ g ^ n := Nat.pow_le_pow_right (Nat.lt_of_lt_of_le (Nat.zero_lt_succ 0) hg1) hnm
63    exact Nat.mul_le_mul hx h1
64
65/-- The cut `m = ⌊log_g(2κ+1)⌋`, i.e. `2κ + 1 < g^{m+1}`, puts the second bound below `1`
66(`2κ·g^n < g^{m+1}·g^n`); the first is below `1` for every `m` (`g^m·g^n − g^n < g^m·g^n`). -/
67⧉theorem fixed_shell_bound_lt_one (g κ m n : Nat) (hg : 2 ≤ g) (hm : 2 * κ + 1 < g ^ (m + 1)) :
68    2 * κ * g ^ n < g ^ (m + 1) * g ^ n ∧ 0 < g ^ n := by
69  have hg1 : 1 ≤ g := Nat.le_trans (Nat.le_succ 1) hg
70  have hpos : 0 < g ^ n := Nat.pow_pos (Nat.lt_of_lt_of_le (Nat.zero_lt_succ 0) hg1)
71  exact ⟨FRC.Nat.mul_lt_mul_of_lt_of_pos (Nat.lt_of_le_of_lt (Nat.le_succ _) hm) hpos, hpos⟩

73/-! ### 2:B4 — which reindexings of the phase cycle are cellular -/

75/-- `m` and `m'` are adjacent on the phase cycle of length `n`. -/
76⧉def Adj (n m m' : Nat) : Prop := (m + 1) % n = m' ∨ (m' + 1) % n = m
77
78/-- The reindexing `ρ_u : m ↦ u·m mod n`. -/
79⧉def rho (n u m : Nat) : Nat := (u * m) % n
80
81/-- 2:B4 (Prop. 2.8) — `ρ_u` preserves the adjacency of the phase cycle (`n ≥ 3`) exactly when
82`u ≡ 1` or `u ≡ −1 (mod n)`. -/
83⧉theorem rho_adj_iff (n u : Nat) (hn : 3 ≤ n) :
84    (∀ m, m < n → Adj n (rho n u m) (rho n u ((m + 1) % n))) ↔ (u % n = 1 ∨ u % n = n - 1) := by
85  have hn0 : 0 < n := Nat.lt_of_lt_of_le (Nat.zero_lt_succ 2) hn
86  have h1n : 1 < n := Nat.lt_of_lt_of_le (Nat.lt_succ_self 1) (Nat.le_trans (Nat.le_succ 2) hn)
87  constructor
88  · intro h
89    have h0 := h 0 hn0
90    unfold rho at h0
91    rw [Nat.mul_zero, FRC.Nat.zero_mod, Nat.zero_add, FRC.Nat.mod_eq_of_lt h1n, Nat.mul_one] at h0
92    -- Adj n 0 (u % n): (0 + 1) % n = u % n, or (u % n + 1) % n = 0
93    match h0 with
94    | Or.inl e => exact Or.inl (by rw [Nat.zero_add, FRC.Nat.mod_eq_of_lt h1n] at e; exact e.symm)
95    | Or.inr e =>
96      refine Or.inr ?_
97      have hu := Nat.mod_lt u hn0
98      exact match Nat.lt_or_ge (u % n + 1) n with
99        | Or.inl hlt => by
100            rw [FRC.Nat.mod_eq_of_lt hlt] at e
101            exact absurd e (Nat.succ_ne_zero _)
102        | Or.inr hge =>
103            have : u % n + 1 = n := Nat.le_antisymm hu hge
104            calc u % n = (u % n + 1) - 1 := (FRC.Nat.add_sub_cancel _ _).symm
105              _ = n - 1 := by rw [this]
106  · intro h m hm
107    unfold rho Adj
108    match h with
109    | Or.inl e =>
110      -- u ≡ 1: ρ is the identity on residues
111      have e1 : ∀ x, x < n → (u * x) % n = x := fun x hx => by
112        rw [← FRC.Nat.mod_mul_mod _ _ _ hn0, e, Nat.one_mul, FRC.Nat.mod_eq_of_lt hx]
113      rw [e1 m hm, e1 _ (Nat.mod_lt _ hn0)]
114      exact Or.inl rfl
115    | Or.inr e =>
116      -- u ≡ −1: ρ is the reflection m ↦ (n − m) % n, and rev (m+1) + 1 ≡ rev m
117      have e1 : ∀ x, x < n → (u * x) % n = Shell.Frame.rev n x := fun x hx => by
118        unfold Shell.Frame.rev
119        rw [← FRC.Nat.mod_mul_mod _ _ _ hn0, e]
120        exact match Nat.decEq x 0 with
121          | isTrue hx0 => by rw [hx0, Nat.mul_zero, Nat.sub_zero, FRC.Nat.zero_mod, FRC.Nat.mod_self n hn0]
122          | isFalse hx0 => by
123              have hx0' := Nat.pos_of_ne_zero hx0
124              -- (n − 1)·x = n·(x − 1) + (n − x)
125              have : (n - 1) * x = n * (x - 1) + (n - x) := by
126                have h1 : (n - 1) * x + x = n * x := by
127                  rw [← Nat.succ_mul, Nat.succ_eq_add_one, FRC.Nat.sub_add_cancel (Nat.le_of_lt h1n)]
128                have h2 : n * (x - 1) + n = n * x := by
129                  rw [← Nat.mul_succ, Nat.succ_eq_add_one, FRC.Nat.sub_add_cancel hx0']
130                have h3 : n * (x - 1) + (n - x) + x = n * x := by
131                  rw [Nat.add_assoc, FRC.Nat.sub_add_cancel (Nat.le_of_lt hx), h2]
132                exact FRC.Nat.add_right_cancel (h1.trans h3.symm)
133              rw [this, FRC.Nat.add_mul_mod_self_left _ _ _ hn0]
134      rw [e1 m hm, e1 _ (Nat.mod_lt _ hn0)]
135      -- rev (m+1 mod n) + 1 ≡ rev m: both ≡ −m
136      refine Or.inr ?_
137      exact match Nat.lt_or_ge (m + 1) n with
138        | Or.inl hlt => by
139            rw [FRC.Nat.mod_eq_of_lt hlt]
140            exact match Nat.decEq m 0 with
141              | isTrue e0 => by
142                  rw [e0] at hlt ⊢
143                  rw [Shell.Frame.rev_zero _ hn0, Shell.Frame.rev_of_pos hlt (Nat.zero_lt_succ 0), Nat.zero_add,
144                    FRC.Nat.sub_add_cancel (Nat.le_of_lt h1n)]
145                  exact FRC.Nat.mod_self n hn0
146              | isFalse e0 => by
147                  have hm0 := Nat.pos_of_ne_zero e0
148                  rw [Shell.Frame.rev_of_pos hlt (Nat.zero_lt_succ m), Shell.Frame.rev_of_pos hm hm0]
149                  -- (n − (m+1)) + 1 = n − m, below n
150                  have : n - (m + 1) + 1 = n - m := by
151                    have h1 : n - (m + 1) + (m + 1) = n := FRC.Nat.sub_add_cancel (Nat.le_of_lt hlt)
152                    have h2 : n - m + m = n := FRC.Nat.sub_add_cancel (Nat.le_of_lt hm)
153                    apply FRC.Nat.add_right_cancel (c := m)
154                    rw [h2, Nat.add_assoc, Nat.add_comm 1 m, h1]
155                  rw [this]
156                  exact FRC.Nat.mod_eq_of_lt (Nat.sub_lt hn0 hm0)
157        | Or.inr hge => by
158            -- m + 1 = n: m = n − 1, rev 0 + 1 = 1, rev (n−1) = 1
159            have hmn : m + 1 = n := Nat.le_antisymm hm hge
160            rw [hmn, FRC.Nat.mod_self _ hn0, Shell.Frame.rev_zero _ hn0, Nat.zero_add, FRC.Nat.mod_eq_of_lt h1n]
161            have hm0 : 0 < m := by
162              refine Nat.lt_of_add_lt_add_right (n := 1) ?_
163              rw [Nat.zero_add, hmn]; exact h1n
164            rw [Shell.Frame.rev_of_pos hm hm0, ← hmn, FRC.Nat.add_sub_cancel_left]

179/-- 2:C4 (Prop. 3.5) — the complex is one for every generator; the frame `g' = g^u` moves the

181⧉theorem label_covariance {κ : Nat} {g : Shell p} (F : Shell.Frame p κ g) (u : Nat) (a : Shell p) (m : Nat) :
