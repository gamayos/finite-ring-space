39/-- 2:E3 (the fixed-shell bound, every shell) — for `g ≥ 2`, any cut `m`, depth `n` and `x ≤ 2κ` with
40`x < g^n` (the grid point `x/g^n` below `1`): either `x/g^n ≤ 1 − g^{−m}`, i.e. `x·g^m + g^n ≤ g^m·g^n`,
41or `x/g^n ≤ 2κ·g^{−(m+1)}`, i.e. `x·g^{m+1} ≤ 2κ·g^n`. With `2κ + 1 < g^{m+1}` both bounds are below `1`, so the
42covering radius of the fixed-shell refinement in `[0, 1]` is bounded below at every depth. -/
43⧉theorem fixed_shell_bound (g κ m n x : Nat) (hg : 2 ≤ g) (hx : x ≤ 2 * κ) (hlt : x < g ^ n) :
44    x * g ^ m + g ^ n ≤ g ^ m * g ^ n ∨ x * g ^ (m + 1) ≤ 2 * κ * g ^ n := by
45  have hg1 : 1 ≤ g := Nat.le_trans (Nat.le_succ 1) hg
46  match Nat.lt_or_ge n (m + 1) with
47  | .inl hnm =>
48    -- n ≤ m: x + 1 ≤ g^n, so x·g^m + g^m ≤ g^n·g^m, and g^n ≤ g^m
49    refine .inl ?_
50    have hn : n ≤ m := Nat.le_of_lt_succ hnm
51    have h1 : (x + 1) * g ^ m ≤ g ^ n * g ^ m := Nat.mul_le_mul_right _ hlt
52    have h2 : g ^ n ≤ g ^ m := Nat.pow_le_pow_right (Nat.lt_of_lt_of_le (Nat.zero_lt_succ 0) hg1) hn
53    rw [FRC.Nat.add_mul, Nat.one_mul] at h1
54    calc x * g ^ m + g ^ n ≤ x * g ^ m + g ^ m := Nat.add_le_add_left h2 _
55      _ ≤ g ^ n * g ^ m := h1
56      _ = g ^ m * g ^ n := Nat.mul_comm _ _
57  | .inr hnm =>
58    refine .inr ?_
59    have h1 : g ^ (m + 1) ≤ g ^ n := Nat.pow_le_pow_right (Nat.lt_of_lt_of_le (Nat.zero_lt_succ 0) hg1) hnm
60    exact Nat.mul_le_mul hx h1
