14/-- Membership, by recursion (decidable). -/
15⧉def mem (v : Nat) : List Nat → Prop
16  | [] => False
17  | a :: l => v = a ∨ mem v l
18
19⧉def decMem (v : Nat) : (l : List Nat) → Decidable (mem v l)
20  | [] => isFalse id
21  | a :: l => match Nat.decEq v a, decMem v l with
22    | isTrue h, _ => isTrue (Or.inl h)
23    | isFalse _, isTrue h' => isTrue (Or.inr h')
24    | isFalse h, isFalse h' => isFalse (fun e => match e with | Or.inl e => h e | Or.inr e => h' e)
25
26instance (v : Nat) (l : List Nat) : Decidable (mem v l) := decMem v l

28/-- No duplicates, by recursion. -/
29⧉def NoDup : List Nat → Prop
30  | [] => True
31  | a :: l => ¬ mem a l ∧ NoDup l

33/-- Remove the first occurrence of `v`. -/
34⧉def erase (v : Nat) : List Nat → List Nat
35  | [] => []
36  | a :: l => if a = v then l else a :: erase v l

38⧉theorem length_erase_of_mem {v : Nat} : ∀ {l : List Nat}, mem v l → (erase v l).length + 1 = l.length
39  | [], h => absurd h id
40  | a :: l, h => by
41    show (if a = v then l else a :: erase v l).length + 1 = l.length + 1
42    exact match Nat.decEq a v with
43      | isTrue e => by rw [if_pos e]
44      | isFalse e => by
45          rw [if_neg e]
46          show (erase v l).length + 1 + 1 = l.length + 1
47          have hm : mem v l := match h with
48            | Or.inl h' => absurd h'.symm e
49            | Or.inr h' => h'
50          rw [length_erase_of_mem hm]

52⧉theorem mem_of_mem_erase {w v : Nat} : ∀ {l : List Nat}, mem w (erase v l) → mem w l
53  | [], h => absurd h id
54  | a :: l, h => by
55    show w = a ∨ mem w l
56    exact match Nat.decEq a v with
57      | isTrue e => by
58          rw [show erase v (a :: l) = l from if_pos e] at h
59          exact Or.inr h
60      | isFalse e => by
61          rw [show erase v (a :: l) = a :: erase v l from if_neg e] at h
62          exact match h with
63            | Or.inl h' => Or.inl h'
64            | Or.inr h' => Or.inr (mem_of_mem_erase h')

66⧉theorem mem_erase_of_ne {w v : Nat} (hwv : w ≠ v) : ∀ {l : List Nat}, mem w l → mem w (erase v l)
67  | [], h => absurd h id

70      | isTrue e => by
71          rw [show erase v (a :: l) = l from if_pos e]

73            | Or.inl h' => absurd (h'.trans e) hwv
74            | Or.inr h' => h'
75      | isFalse e => by

78            | Or.inl h' => Or.inl h'

81⧉theorem nodup_erase (v : Nat) : ∀ {l : List Nat}, NoDup l → NoDup (erase v l)
82  | [], _ => trivial
83  | a :: l, ⟨ha, hl⟩ => by
84    exact match Nat.decEq a v with
85      | isTrue e => by rw [show erase v (a :: l) = l from if_pos e]; exact hl
86      | isFalse e => by
87          rw [show erase v (a :: l) = a :: erase v l from if_neg e]
88          exact ⟨fun h => ha (mem_of_mem_erase h), nodup_erase v hl⟩

90⧉theorem not_mem_erase_self (v : Nat) : ∀ {l : List Nat}, NoDup l → ¬ mem v (erase v l)
91  | [], _, h => h
92  | a :: l, ⟨ha, hl⟩, h => by
93    exact match Nat.decEq a v with
94      | isTrue e => by
95          rw [show erase v (a :: l) = l from if_pos e] at h
96          exact ha (e ▸ h)
97      | isFalse e => by
98          rw [show erase v (a :: l) = a :: erase v l from if_neg e] at h
99          exact match h with
100            | Or.inl h' => e h'.symm
101            | Or.inr h' => not_mem_erase_self v hl h'

103/-- The bound: a list of distinct numbers in `[1, n]` has at most `n` entries. -/
104⧉theorem length_le_of_nodup : ∀ (n : Nat) (l : List Nat), NoDup l → (∀ e, mem e l → 1 ≤ e ∧ e ≤ n) →
105    l.length ≤ n
106  | 0, [], _, _ => Nat.le_refl 0
107  | 0, a :: l, _, hb => absurd (hb a (Or.inl rfl)) (fun ⟨h1, h0⟩ => Nat.lt_irrefl 0 (Nat.lt_of_lt_of_le h1 h0))
108  | n + 1, l, hnd, hb =>
109    match (inferInstance : Decidable (mem (n + 1) l)) with
110    | isTrue hm =>
111      have h1 : (erase (n + 1) l).length ≤ n :=
112        length_le_of_nodup n (erase (n + 1) l) (nodup_erase _ hnd) (fun e he =>
113          have hb' := hb e (mem_of_mem_erase he)
114          ⟨hb'.1, match Nat.lt_or_ge e (n + 1) with
115            | Or.inl hlt => Nat.le_of_lt_succ hlt
116            | Or.inr hge =>
117              have : e = n + 1 := Nat.le_antisymm hb'.2 hge
118              absurd (this ▸ he) (not_mem_erase_self (n + 1) hnd)⟩)
119      by rw [← length_erase_of_mem hm]; exact Nat.succ_le_succ h1
120    | isFalse hm =>
121      Nat.le_succ_of_le (length_le_of_nodup n l hnd (fun e he =>
122        have hb' := hb e he
123        ⟨hb'.1, match Nat.lt_or_ge e (n + 1) with
124          | Or.inl hlt => Nat.le_of_lt_succ hlt
125          | Or.inr hge => absurd he ((Nat.le_antisymm hb'.2 hge) ▸ hm)⟩))

127/-- The pigeonhole: `n` distinct numbers in `[1, n]` are all of them. -/
128⧉theorem mem_of_nodup_of_length (n : Nat) (l : List Nat) (hnd : NoDup l) (hb : ∀ e, mem e l → 1 ≤ e ∧ e ≤ n)
129    (hlen : l.length = n) (v : Nat) (hv1 : 1 ≤ v) (hvn : v ≤ n) : mem v l :=
130  match decMem v l with
131  | isTrue h => h
132  | isFalse h =>
133    have hnd' : NoDup (v :: l) := ⟨h, hnd⟩
134    have hb' : ∀ e, mem e (v :: l) → 1 ≤ e ∧ e ≤ n := fun e he => match he with
135      | Or.inl e' => e' ▸ ⟨hv1, hvn⟩
136      | Or.inr e' => hb e e'
137    have := length_le_of_nodup n (v :: l) hnd' hb'
138    absurd (hlen ▸ this : n + 1 ≤ n) (Nat.not_succ_le_self n)
