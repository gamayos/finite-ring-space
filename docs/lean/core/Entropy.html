35/-- 14:A1, 14:B9 — the quarter identity: `Ω = 4S + 1` has `Ω ≡ 1 (mod 4)` and `Ω − 1 = 4S`, so the import fixes the

37theorem quarter_identity (S : Nat) : (4 * S + 1) % 4 = 1 ∧ 4 * S + 1 - 1 = 4 * S :=
38  ⟨FRC.Nat.add_mul_mod_self_left 1 S 4 (Nat.zero_lt_succ 3), FRC.Nat.add_sub_cancel (4 * S) 1⟩

40/-- 14:A2 — the triality centre: `S ≡ 1 (mod 3)` gives `(Ω + 1) % 3 = 0` and `(Ω − 1) % 3 = 1`, so the centre
41`3 ∣ Ω + 1` is carried and `3 ∤ 4S`. -/
42theorem triality_residues (S : Nat) (h : S % 3 = 1) : (4 * S + 2) % 3 = 0 ∧ (4 * S) % 3 = 1 := by
43  have h3 : 0 < 3 := Nat.zero_lt_succ 2
44  have h4S : (4 * S) % 3 = 1 := by
45    rw [FRC.Nat.mul_mod 4 S 3 h3, h]
46  refine ⟨?_, h4S⟩
47  rw [FRC.Nat.add_mod (4 * S) 2 3 h3, h4S]

49/-- 14:C6, 14:X4 — the octant sector `C₈ ⊂ C_{4S}` exists (`8 ∣ 4S`) exactly when `S` is even. -/
50theorem octant_divides (S : Nat) : (∃ k, 4 * S = 8 * k) ↔ (∃ m, S = 2 * m) := by

54    rw [hk, ← FRC.Nat.mul_assoc]
55  · rintro ⟨m, hm⟩
56    exact ⟨m, by rw [hm, ← FRC.Nat.mul_assoc]⟩

58/-- 14:B2, 14:C5 — the dictionary on the count face: the full cycle is `4S` (`2π`); the half-cycle `2S` (`π`)
59twice over is the cycle, and with `S = 2m` the octant `m = S/2` (`π/4`) eight times over is the cycle — an
60integer count on every admissible Carrier. -/
61theorem octant_count (S m : Nat) (h : S = 2 * m) : 2 * (2 * S) = 4 * S ∧ 8 * m = 4 * S :=
62  ⟨by rw [← FRC.Nat.mul_assoc], by rw [h, ← FRC.Nat.mul_assoc]⟩

64/-- 14:C5, 14:B6 — the closure count: given B6's accounting (a two-way registration of depth `x` costs `2x` of

66octant `x = m` closes the budget. -/
67theorem closure_count (m x : Nat) : 2 * x ≤ 2 * m ↔ x ≤ m := by
68  constructor

70    match Nat.lt_or_ge m x with
71    | .inr hle => exact hle

73      have : m * 2 < x * 2 := FRC.Nat.mul_lt_mul_of_lt_of_pos hlt (Nat.zero_lt_succ 1)

75      exact absurd (Nat.lt_of_lt_of_le this h) (Nat.lt_irrefl _)

94    ← Shell.add_assoc a (-a) (1 + 1), Shell.add_neg a, Shell.zero_add]

98theorem even_capacity_powers (F : Frame p κ g) (m : Nat) (hm : κ = 2 * m) :

104/-- 14:C6 — the octant character: on every frame `(τ; 0, 1, g)` of even capacity `κ = 2m`, `ζ = g^m` has
105`ζ⁴ = −1` and `ζ⁸ = 1` — the sector `C₈ ⊂ C_{4κ}` is realised and `ζ` is its residue `ζ₈`. -/
106theorem octant_residue (F : Frame p κ g) (m : Nat) (hm : κ = 2 * m) :

111  · rw [← Shell.pow_mul, Nat.mul_comm m 8]; exact h8

113/-- 14:C6, 14:X4 — the Tsirelson square: with `ζ = g^m` and `ζ' = g^{7m}` its inverse (`ζζ' = 1`),
114`(ζ + ζ')² = 2` and `(2(ζ + ζ'))² = 8` on every frame of capacity `κ = 2m`; hence `2` is a square on every
115such shell — the direction "octant ⇒ c-square" of X4's second face (the converse is
116`even_capacity_of_two_square` below; the equivalence `two_is_square_iff`). -/
117theorem tsirelson_square (F : Frame p κ g) (m : Nat) (hm : κ = 2 * m) :
118    g ^ m * g ^ (7 * m) = 1 ∧ (g ^ m + g ^ (7 * m)) ^ 2 = 2 ∧
119    (2 * (g ^ m + g ^ (7 * m))) ^ 2 = 8 := by
120  obtain ⟨h4, h8⟩ := even_capacity_powers F m hm
121  have hz : g ^ m * g ^ (7 * m) = 1 := by
122    rw [← Shell.pow_add, show m + 7 * m = 8 * m by
123      rw [Nat.mul_comm 8 m, Nat.mul_succ, Nat.mul_comm m 7, Nat.add_comm]]
124    exact h8
125  -- `ζ'² = −ζ²`: `g^{14m} = g^{8m} g^{4m} g^{2m} = (−1) g^{2m}`
126  have h14 : 7 * m + 7 * m = 8 * m + (4 * m + (m + m)) := by
127    calc 7 * m + 7 * m = (7 + 7) * m := (FRC.Nat.add_mul 7 7 m).symm
128      _ = (8 + (4 + (1 + 1))) * m := rfl
129      _ = 8 * m + (4 * m + (1 * m + 1 * m)) := by
130        rw [FRC.Nat.add_mul, FRC.Nat.add_mul, FRC.Nat.add_mul]
131      _ = 8 * m + (4 * m + (m + m)) := by rw [Nat.one_mul]
132  have hz' : g ^ (7 * m) * g ^ (7 * m) = -(g ^ m * g ^ m) := by
133    rw [← Shell.pow_add, h14, Shell.pow_add, Shell.pow_add, Shell.pow_add, h8, h4, Shell.one_mul,
134      Shell.neg_one_mul]
135  -- `(ζ + ζ')² = ζ² + ζζ' + ζ'ζ + ζ'² = ζ² + 1 + (1 − ζ²) = 2`
136  have hsq : (g ^ m + g ^ (7 * m)) ^ 2 = 2 := by
137    rw [Shell.pow_two, Shell.left_distrib, Shell.right_distrib, Shell.right_distrib, hz,
138      Shell.mul_comm (g ^ (7 * m)) (g ^ m), hz, hz']
139    exact add_one_add_one_neg _
140  refine ⟨hz, hsq, ?_⟩
141  rw [Shell.mul_pow, hsq, Shell.pow_two, lit_mul 2 2, lit_mul 4 2]
142
143/-- 14:X4 — the direction "even capacity ⇒ `2` a square" at tier 0: on every frame of even capacity `2` is a

146  ⟨g ^ m + g ^ (7 * m), by rw [← Shell.pow_two]; exact (tsirelson_square F m hm).2.1⟩

155/-- `a b (c d) = a c (b d)` on every shell. -/
156theorem mul_mul_mul_comm (a b c d : Shell p) : a * b * (c * d) = a * c * (b * d) := by

159/-- Literals add as their values: `a + b = (a + b)` on every shell. -/
160theorem lit_add (a b : Nat) : (OfNat.ofNat a : Shell p) + OfNat.ofNat b = (OfNat.ofNat (a + b) : Shell p) :=
161  Shell.ext (by
162    show (a % p + b % p) % p = (a + b) % p
163    exact (FRC.Nat.add_mod a b p Pos.pos).symm)

165/-- `−1 ≠ 1` on every frame (`2 ≠ 0`). -/
166theorem neg_one_ne_one (F : Frame p κ g) : (-1 : Shell p) ≠ 1 := fun h => by
167  have h2 : (1 : Shell p) + 1 = 0 := by
168    have := Shell.neg_add (1 : Shell p); rw [h] at this; exact this
169  exact F.two_ne_zero (by rw [Frame.two_eq_one_add_one]; exact h2)

171/-- `−1 ≠ 0` on every frame. -/
172theorem neg_one_ne_zero (F : Frame p κ g) : (-1 : Shell p) ≠ 0 := fun h => by
173  have : (1 : Shell p) = 0 := by
174    have h' : -(-1 : Shell p) = -0 := congrArg Neg.neg h
175    rw [Shell.neg_neg, Shell.neg_zero] at h'
176    exact h'
177  exact F.one_ne_zero this

179/-- A square root of `2` gives a square root of the quarter-turn: with `i² = −1`, `2h = 1` and `r² = 2`,
180`ζ = (r + i r) h` has `ζ² = i`, hence `ζ⁴ = −1` and `ζ⁸ = 1` — an element of order eight wherever `−1 ≠ 1`,
181that is on every frame. -/
182theorem octant_of_two_square {r i h : Shell p} (hr : r * r = 2) (hi : i * i = -1)
183    (hh : 2 * h = 1) :
184    ∃ ζ : Shell p, ζ * ζ = i ∧ ζ ^ 4 = -1 ∧ ζ ^ 8 = 1 := by
185  -- `(r + i r)² = 4 i`
186  have hA : (r + i * r) * (r + i * r) = i * 4 := by
187    rw [Shell.left_distrib, Shell.right_distrib, Shell.right_distrib, hr,
188      Shell.mul_assoc i r r, hr, Shell.mul_left_comm r i r, hr,
189      Shell.mul_assoc i r (i * r), Shell.mul_left_comm r i r, ← Shell.mul_assoc i i (r * r), hi, hr,
190      Shell.neg_one_mul, Shell.add_comm 2 (i * 2), Shell.add_assoc, Shell.add_comm 2 (i * 2 + -2),
191      Shell.add_assoc, Shell.neg_add, Shell.add_zero, ← Shell.left_distrib, lit_add 2 2]
192  -- `4 h² = (2h)² = 1`
193  have h4 : (4 : Shell p) * (h * h) = 1 := by
194    rw [← lit_mul 2 2, ← mul_mul_mul_comm, hh, Shell.one_mul]
195  have hz : ((r + i * r) * h) * ((r + i * r) * h) = i := by
196    rw [mul_mul_mul_comm, hA, Shell.mul_assoc, h4, Shell.mul_one]
197  refine ⟨(r + i * r) * h, hz, ?_, ?_⟩
198  · rw [show (4 : Nat) = 2 * 2 from rfl, Shell.pow_mul, Shell.pow_two, Shell.pow_two, hz, hi]
199  · rw [show (8 : Nat) = 2 * 2 * 2 from rfl, Shell.pow_mul, Shell.pow_mul, Shell.pow_two, Shell.pow_two,
200      Shell.pow_two, hz, hi, Shell.neg_mul_neg, Shell.one_mul]
201
202/-- The arithmetic of the order: `(k · 8) % 4κ = 0` and `(k · 4) % 4κ ≠ 0` force `κ` even (a divisor `4κ` of
203`8k` that misses `4k` cannot be `4 · odd`; `0 < κ` is only what `mod_spec` needs). -/
204theorem even_of_order_eight {κ k : Nat} (hκ : 0 < κ) (h8 : (k * 8) % (4 * κ) = 0)
205    (h4 : (k * 4) % (4 * κ) ≠ 0) : ∃ m, κ = 2 * m := by
206  have h4κ : 0 < 4 * κ := Nat.mul_pos (Nat.zero_lt_succ 3) hκ
207  obtain ⟨q, hq⟩ := FRC.Nat.mod_spec (4 * κ) h4κ (k * 8)
208  rw [h8, Nat.add_zero] at hq
209  -- `2k = κ q`
210  have h2k : 2 * k = κ * q := by
211    apply Nat.eq_of_mul_eq_mul_left (Nat.zero_lt_succ 3)
212    rw [← FRC.Nat.mul_assoc, ← FRC.Nat.mul_assoc, Nat.mul_comm 4 2, Nat.mul_comm (2 * 4) k, ← hq]
213  have h2 : 0 < 2 := Nat.zero_lt_succ 1
214  obtain ⟨m, hm⟩ := FRC.Nat.mod_spec 2 h2 κ
215  have hlt : κ % 2 < 2 := FRC.Nat.mod_lt' κ h2
216  match hκ2 : κ % 2 with
217  | 0 => rw [hκ2, Nat.add_zero] at hm; exact ⟨m, hm⟩
218  | 1 =>
219    -- `κ` odd: `q` is even, `k = κ q'`, so `4κ ∣ 4k`, contradicting `h4`
220    have hq2 : q % 2 = 0 := by
221      have e1 : (κ * q) % 2 = q % 2 := by
222        rw [FRC.Nat.mul_mod κ q 2 h2, hκ2, Nat.one_mul, FRC.Nat.mod_mod q 2 h2]
223      have e2 : (2 * k) % 2 = 0 := by
224        rw [Nat.mul_comm 2 k]
225        exact (FRC.Nat.mod_unique (Nat.zero_lt_succ 1) (by rw [Nat.add_zero, Nat.mul_comm k 2]) : (k * 2) % 2 = 0)
226      rw [← e1, ← h2k]; exact e2
227    obtain ⟨q', hq'⟩ := FRC.Nat.mod_spec 2 h2 q
228    rw [hq2, Nat.add_zero] at hq'
229    have hk : k = κ * q' := by
230      apply Nat.eq_of_mul_eq_mul_left h2
231      rw [h2k, hq', Nat.mul_left_comm κ 2 q']
232    exact absurd (FRC.Nat.mod_unique h4κ
233      (by rw [hk, Nat.add_zero, Nat.mul_comm (κ * q') 4, ← FRC.Nat.mul_assoc])) h4
234  | n + 2 => exact absurd hlt (by rw [hκ2]; exact Nat.not_lt_of_ge (Nat.le_add_left 2 n))

236/-- 14:C6 — the octant forces even capacity: on every frame, an element `ζ` with `ζ⁴ = −1` and `ζ⁸ = 1` (an
237element of order eight) gives `8 ∣ 4κ`, so the capacity is even. -/
238theorem even_capacity_of_octant (F : Frame p κ g) {ζ : Shell p} (hz4 : ζ ^ 4 = -1) (hz8 : ζ ^ 8 = 1) :
239    ∃ m, κ = 2 * m := by
240  have hζ0 : ζ ≠ 0 := fun h0 => by
241    rw [h0, show (4 : Nat) = 3 + 1 from rfl, Shell.pow_succ, Shell.mul_zero] at hz4
242    exact neg_one_ne_zero F hz4.symm
243  obtain ⟨k, hk, hgk⟩ := F.eq_pow_of_ne_zero hζ0
244  have e8 : (k * 8) % (p - 1) = 0 :=
245    F.mod_eq_zero_of_pow_eq_one (by rw [Shell.pow_mul, hgk]; exact hz8)
246  have e4 : (k * 4) % (p - 1) ≠ 0 := fun e => by
247    have := F.pow_eq_one_of_mod e
248    rw [Shell.pow_mul, hgk, hz4] at this
249    exact neg_one_ne_one F this
250  rw [F.n_eq] at e8 e4
251  exact even_of_order_eight F.cap_pos e8 e4

253/-- 14:C6 — the octant sector at tier 0, as an equivalence: on every frame, an element with `ζ⁴ = −1` and
254`ζ⁸ = 1` exists exactly when the capacity is even. -/
255theorem octant_iff (F : Frame p κ g) : (∃ ζ : Shell p, ζ ^ 4 = -1 ∧ ζ ^ 8 = 1) ↔ ∃ m, κ = 2 * m :=
256  ⟨fun ⟨_, h4, h8⟩ => even_capacity_of_octant F h4 h8,
257   fun ⟨m, hm⟩ => ⟨g ^ m, octant_residue F m hm⟩⟩

259/-- 14:X4 — the converse of the second face at tier 0: on every frame, a square root of `2` forces the
260capacity even — `ζ = (r + i r)/2` has `ζ² = i`, so an element of order eight exists and `8 ∣ 4κ`. -/
261theorem even_capacity_of_two_square (F : Frame p κ g) {r : Shell p} (hr : r * r = 2) :

264  obtain ⟨ζ, _, hz4, hz8⟩ := octant_of_two_square hr F.quarter_turn_sq hh

277/-- 14:B10 — the nesting bound, the necessity direction: a shell `q` registered whole by a Subject `p` of the
278Carrier `Ω` (`q² < p`, `p² < Ω`) has `q⁴ < Ω`; for `q = 12N + 1` this bounds the hydrogen count `N`; a Subject
279between the two is exhibited on the laboratory Carrier (`lab_nesting`, `p = 1373`). -/
280theorem nesting_bound (q p Ω : Nat) (hq : q * q < p) (hp : p * p < Ω) : q * q * (q * q) < Ω := by
281  have hp0 : 0 < p := FRC.Nat.pos_of_lt hq
282  have h1 : q * q * (q * q) < p * p := by
283    match Nat.lt_or_ge 0 (q * q) with
284    | .inl hpos =>
285      have ha : q * q * (q * q) < p * (q * q) := FRC.Nat.mul_lt_mul_of_lt_of_pos hq hpos
286      have hb : q * q * p < p * p := FRC.Nat.mul_lt_mul_of_lt_of_pos hq hp0
287      rw [Nat.mul_comm (q * q) p] at hb
288      exact Nat.lt_trans ha hb
289    | .inr hz =>
290      have h0 : q * q = 0 := Nat.le_antisymm hz (Nat.zero_le _)
291      rw [h0]; exact Nat.mul_pos hp0 hp0
292  exact Nat.lt_trans h1 hp

294/-- Trial division up to `B` decides primality when `n < (B + 1)²`: a divisor `d ≥ 2` of `n` with
295`d > B` has a cofactor `q = n/d` with `2 ≤ q ≤ B`, itself a divisor. -/
296theorem isPrime_of_bounded (n B : Nat) (h2 : 2 ≤ n) (hB : n < (B + 1) * (B + 1))
297    (hd : ∀ d, d < B + 1 → 2 ≤ d → n % d ≠ 0) : Dimensions.isPrime n := by
298  refine ⟨h2, fun d hdn hd2 hmod => ?_⟩
299  have hd0 : 0 < d := Nat.lt_of_lt_of_le (Nat.zero_lt_succ 1) hd2
300  obtain ⟨q, hq⟩ := FRC.Nat.mod_spec d hd0 n
301  rw [hmod, Nat.add_zero] at hq
302  match Nat.lt_or_ge d (B + 1) with

316        exact absurd (Nat.lt_of_lt_of_le hB this) (hq ▸ Nat.lt_irrefl n)

319      FRC.Nat.mod_unique hq0 (by rw [hq, Nat.mul_comm d q, Nat.add_zero])
