33/-- 14:A1, 14:B9 — the quarter identity: `Ω = 4S + 1` has `Ω ≡ 1 (mod 4)` and `Ω − 1 = 4S`, so the import fixes the

35⧉theorem quarter_identity (S : Nat) : (4 * S + 1) % 4 = 1 ∧ 4 * S + 1 - 1 = 4 * S :=
36  ⟨FRC.Nat.add_mul_mod_self_left 1 S 4 (Nat.zero_lt_succ 3), FRC.Nat.add_sub_cancel (4 * S) 1⟩

38/-- 14:A2 — the triality centre: `S ≡ 1 (mod 3)` gives `(Ω + 1) % 3 = 0` and `(Ω − 1) % 3 = 1`, so the centre
39`3 ∣ Ω + 1` is carried and `3 ∤ 4S`. -/
40⧉theorem triality_residues (S : Nat) (h : S % 3 = 1) : (4 * S + 2) % 3 = 0 ∧ (4 * S) % 3 = 1 := by
41  have h3 : 0 < 3 := Nat.zero_lt_succ 2
42  have h4S : (4 * S) % 3 = 1 := by
43    rw [FRC.Nat.mul_mod 4 S 3 h3, h]
44  refine ⟨?_, h4S⟩
45  rw [FRC.Nat.add_mod (4 * S) 2 3 h3, h4S]

47/-- 14:C6, 14:X4 — the octant sector `C₈ ⊂ C_{4S}` exists (`8 ∣ 4S`) exactly when `S` is even. -/
48⧉theorem octant_divides (S : Nat) : (∃ k, 4 * S = 8 * k) ↔ (∃ m, S = 2 * m) := by

52    rw [hk, ← FRC.Nat.mul_assoc]
53  · rintro ⟨m, hm⟩
54    exact ⟨m, by rw [hm, ← FRC.Nat.mul_assoc]⟩

56/-- 14:B2, 14:C5 — the dictionary on the count face: the full cycle is `4S` (`2π`); the half-cycle `2S` (`π`)
57twice over is the cycle, and with `S = 2m` the octant `m = S/2` (`π/4`) eight times over is the cycle — an
58integer count on every admissible Carrier. -/
59⧉theorem octant_count (S m : Nat) (h : S = 2 * m) : 2 * (2 * S) = 4 * S ∧ 8 * m = 4 * S :=
60  ⟨by rw [← FRC.Nat.mul_assoc], by rw [h, ← FRC.Nat.mul_assoc]⟩

62/-- 14:C5, 14:B6 — the closure count: given B6's accounting (a two-way registration of depth `x` costs `2x` of
63the one-way budget `S = 2m`; the realisation, not formalised), the closable depths are exactly `x ≤ m`, and the
64octant `x = m` closes the budget. -/
65⧉theorem closure_count (m x : Nat) : 2 * x ≤ 2 * m ↔ x ≤ m := by
66  constructor
67  · intro h
68    match Nat.lt_or_ge m x with
69    | .inr hle => exact hle
70    | .inl hlt =>
71      have : m * 2 < x * 2 := FRC.Nat.mul_lt_mul_of_lt_of_pos hlt (Nat.zero_lt_succ 1)
72      rw [Nat.mul_comm m 2, Nat.mul_comm x 2] at this
73      exact absurd (Nat.lt_of_lt_of_le this h) (Nat.lt_irrefl _)
74  · intro h; exact Nat.mul_le_mul_left 2 h

83/-- Literals multiply as their values: `a · b = ab` on every shell. -/
84⧉theorem lit_mul (a b : Nat) : (OfNat.ofNat a : Shell p) * OfNat.ofNat b = (OfNat.ofNat (a * b) : Shell p) :=
85  Shell.ext (by
86    show (a % p * (b % p)) % p = (a * b) % p
87    exact (FRC.Nat.mul_mod a b p Pos.pos).symm)

89/-- `a + 1 + (1 − a) = 2` on every shell. -/
90⧉theorem add_one_add_one_neg (a : Shell p) : a + 1 + (1 + -a) = 2 := by
91  rw [Shell.add_assoc, ← Shell.add_assoc 1 1 (-a), Shell.add_comm (1 + 1) (-a),
92    ← Shell.add_assoc a (-a) (1 + 1), Shell.add_neg a, Shell.zero_add]
93  exact Frame.two_eq_one_add_one.symm

95/-- On a frame of capacity `κ = 2m`: `g^{4m} = −1` (the half-period) and `g^{8m} = 1`. -/
96⧉theorem even_capacity_powers (F : Frame p κ g) (m : Nat) (hm : κ = 2 * m) :
97    g ^ (4 * m) = -1 ∧ g ^ (8 * m) = 1 := by
98  have h4 : 4 * m = 2 * κ := by rw [hm, ← FRC.Nat.mul_assoc]
99  have h8 : 8 * m = p - 1 := by rw [F.n_eq, hm, ← FRC.Nat.mul_assoc]
100  exact ⟨by rw [h4]; exact F.half_period, by rw [h8]; exact F.pow_n⟩

102/-- 14:C6 — the octant character: on every frame `(τ; 0, 1, g)` of even capacity `κ = 2m`, `ζ = g^m` has
103`ζ⁴ = −1` and `ζ⁸ = 1` — the sector `C₈ ⊂ C_{4κ}` is realised and `ζ` is its residue `ζ₈`. -/
104⧉theorem octant_residue (F : Frame p κ g) (m : Nat) (hm : κ = 2 * m) :
105    (g ^ m) ^ 4 = -1 ∧ (g ^ m) ^ 8 = 1 := by
106  obtain ⟨h4, h8⟩ := even_capacity_powers F m hm
107  constructor
108  · rw [← Shell.pow_mul, Nat.mul_comm m 4]; exact h4
109  · rw [← Shell.pow_mul, Nat.mul_comm m 8]; exact h8

111/-- 14:C6, 14:X4 — the Tsirelson square: with `ζ = g^m` and `ζ' = g^{7m}` its inverse (`ζζ' = 1`),
112`(ζ + ζ')² = 2` and `(2(ζ + ζ'))² = 8` on every frame of capacity `κ = 2m`; hence `2` is a square on every
113such shell — the direction "octant ⇒ c-square" of X4's second face (the converse is
114`FRC.Entropy.octant_second_face` on Mathlib). -/
115⧉theorem tsirelson_square (F : Frame p κ g) (m : Nat) (hm : κ = 2 * m) :
116    g ^ m * g ^ (7 * m) = 1 ∧ (g ^ m + g ^ (7 * m)) ^ 2 = 2 ∧
117    (2 * (g ^ m + g ^ (7 * m))) ^ 2 = 8 := by
118  obtain ⟨h4, h8⟩ := even_capacity_powers F m hm
119  have hz : g ^ m * g ^ (7 * m) = 1 := by
120    rw [← Shell.pow_add, show m + 7 * m = 8 * m by
121      rw [Nat.mul_comm 8 m, Nat.mul_succ, Nat.mul_comm m 7, Nat.add_comm]]
122    exact h8
123  -- `ζ'² = −ζ²`: `g^{14m} = g^{8m} g^{4m} g^{2m} = (−1) g^{2m}`
124  have h14 : 7 * m + 7 * m = 8 * m + (4 * m + (m + m)) := by
125    calc 7 * m + 7 * m = (7 + 7) * m := (FRC.Nat.add_mul 7 7 m).symm
126      _ = (8 + (4 + (1 + 1))) * m := rfl
127      _ = 8 * m + (4 * m + (1 * m + 1 * m)) := by
128        rw [FRC.Nat.add_mul, FRC.Nat.add_mul, FRC.Nat.add_mul]
129      _ = 8 * m + (4 * m + (m + m)) := by rw [Nat.one_mul]
130  have hz' : g ^ (7 * m) * g ^ (7 * m) = -(g ^ m * g ^ m) := by
131    rw [← Shell.pow_add, h14, Shell.pow_add, Shell.pow_add, Shell.pow_add, h8, h4, Shell.one_mul,
132      Shell.neg_one_mul]
133  -- `(ζ + ζ')² = ζ² + ζζ' + ζ'ζ + ζ'² = ζ² + 1 + (1 − ζ²) = 2`
134  have hsq : (g ^ m + g ^ (7 * m)) ^ 2 = 2 := by
135    rw [Shell.pow_two, Shell.left_distrib, Shell.right_distrib, Shell.right_distrib, hz,
136      Shell.mul_comm (g ^ (7 * m)) (g ^ m), hz, hz']
137    exact add_one_add_one_neg _
138  refine ⟨hz, hsq, ?_⟩
139  rw [Shell.mul_pow, hsq, Shell.pow_two, lit_mul 2 2, lit_mul 4 2]
140
141/-- 14:X4 — one direction of the second face at tier 0: on every frame of even capacity `2` is a square,
142`(ζ + ζ⁻¹)² = 2`; the converse (`2` a square ⇒ even capacity) is on Mathlib, its instances decided in
143`octant17`. -/
144⧉theorem two_is_square (F : Frame p κ g) (m : Nat) (hm : κ = 2 * m) : ∃ r : Shell p, r * r = 2 :=
145  ⟨g ^ m + g ^ (7 * m), by rw [← Shell.pow_two]; exact (tsirelson_square F m hm).2.1⟩
146

152/-- 14:B10 — the nesting bound, the necessity direction: a shell `q` registered whole by a Subject `p` of the
153Carrier `Ω` (`q² < p`, `p² < Ω`) has `q⁴ < Ω`; for `q = 12N + 1` this bounds the hydrogen count `N`; a Subject
154between the two is exhibited on the laboratory Carrier (`lab_nesting`, `p = 1373`). -/
155⧉theorem nesting_bound (q p Ω : Nat) (hq : q * q < p) (hp : p * p < Ω) : q * q * (q * q) < Ω := by
156  have hp0 : 0 < p := FRC.Nat.pos_of_lt hq
157  have h1 : q * q * (q * q) < p * p := by
158    match Nat.lt_or_ge 0 (q * q) with
159    | .inl hpos =>
160      have ha : q * q * (q * q) < p * (q * q) := FRC.Nat.mul_lt_mul_of_lt_of_pos hq hpos
161      have hb : q * q * p < p * p := FRC.Nat.mul_lt_mul_of_lt_of_pos hq hp0
162      rw [Nat.mul_comm (q * q) p] at hb
163      exact Nat.lt_trans ha hb
164    | .inr hz =>
165      have h0 : q * q = 0 := Nat.le_antisymm hz (Nat.zero_le _)
166      rw [h0]; exact Nat.mul_pos hp0 hp0
167  exact Nat.lt_trans h1 hp

169/-- Trial division up to `B` decides primality when `n < (B + 1)²`: a divisor `d ≥ 2` of `n` with
170`d > B` has a cofactor `q = n/d` with `2 ≤ q ≤ B`, itself a divisor. -/
171⧉theorem isPrime_of_bounded (n B : Nat) (h2 : 2 ≤ n) (hB : n < (B + 1) * (B + 1))
172    (hd : ∀ d, d < B + 1 → 2 ≤ d → n % d ≠ 0) : Dimensions.isPrime n := by
173  refine ⟨h2, fun d hdn hd2 hmod => ?_⟩
174  have hd0 : 0 < d := Nat.lt_of_lt_of_le (Nat.zero_lt_succ 1) hd2
175  obtain ⟨q, hq⟩ := FRC.Nat.mod_spec d hd0 n
176  rw [hmod, Nat.add_zero] at hq
177  match Nat.lt_or_ge d (B + 1) with
178  | .inl hlt => exact hd d hlt hd2 hmod
179  | .inr hge =>
180    -- the cofactor `q`: `n = d q`, `q ≥ 2` (else `n = 0` or `n = d`), and `q ≤ B` (else `d q ≥ (B+1)²`)
181    have hq2 : 2 ≤ q := by
182      match q with
183      | 0 => rw [Nat.mul_zero] at hq; exact absurd (hq ▸ h2) (Nat.not_succ_le_zero 1)
184      | 1 => rw [Nat.mul_one] at hq; exact absurd (hq ▸ hdn) (Nat.lt_irrefl d)
185      | q + 2 => exact Nat.le_add_left 2 q
186    have hqB : q < B + 1 := by
187      match Nat.lt_or_ge q (B + 1) with
188      | .inl h => exact h
189      | .inr hqge =>
190        have : (B + 1) * (B + 1) ≤ d * q := Nat.mul_le_mul hge hqge
191        exact absurd (Nat.lt_of_lt_of_le hB this) (hq ▸ Nat.lt_irrefl n)
192    have hq0 : 0 < q := Nat.lt_of_lt_of_le (Nat.zero_lt_succ 1) hq2
193    have hmodq : n % q = 0 :=
194      FRC.Nat.mod_unique hq0 (by rw [hq, Nat.mul_comm d q, Nat.add_zero])
195    exact hd q hqB hq2 hmodq
196
197end nesting
