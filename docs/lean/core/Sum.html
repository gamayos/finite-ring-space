3/-!
4# FrcCore.Sum — finite sums on the shell, the geometric sum, the Fourier inversion
5
6Sums over `l < n` are defined by structural recursion (`sumRange f (n+1) = sumRange f n + f n`): no
7`Finset`, no quotient, no function extensionality — a congruence lemma (`sum_congr`) replaces `funext`.
8The geometric sum gives the principal-root identity (2:F1, Prop. 6.1 of 2-geometry) and the entrywise
9inversion of the shell Fourier matrix `W k j = g^{jk}`: `Σ_l W k l · (−g^{−lj}) = [k = j]` (2:F3,
10Prop. 6.3; 6:B5 in matrix form). No axioms.
11-/

18/-- `sumRange f n = f 0 + f 1 + ⋯ + f (n−1)`. -/
19⧉def sumRange (f : Nat → Shell p) : Nat → Shell p
20  | 0 => 0
21  | n + 1 => sumRange f n + f n
22
23⧉theorem sumRange_zero (f : Nat → Shell p) : sumRange f 0 = 0 := rfl
24⧉theorem sumRange_succ (f : Nat → Shell p) (n : Nat) : sumRange f (n + 1) = sumRange f n + f n := rfl
25
26⧉theorem sum_congr {f h : Nat → Shell p} (n : Nat) (e : ∀ l, l < n → f l = h l) :
27    sumRange f n = sumRange h n := by
28  induction n with
29  | zero => rfl
30  | succ n ih =>
31    rw [sumRange_succ, sumRange_succ, ih (fun l hl => e l (Nat.lt_succ_of_lt hl)),
32      e n (Nat.lt_succ_self n)]
33
34⧉theorem sum_add (f h : Nat → Shell p) (n : Nat) :
35    sumRange (fun l => f l + h l) n = sumRange f n + sumRange h n := by
36  induction n with
37  | zero => rw [sumRange_zero, sumRange_zero, sumRange_zero, add_zero]
38  | succ n ih =>
39    rw [sumRange_succ, sumRange_succ, sumRange_succ, ih, add_add_add_comm]
40
41⧉theorem sum_mul_right (f : Nat → Shell p) (c : Shell p) (n : Nat) :
42    sumRange (fun l => f l * c) n = sumRange f n * c := by
43  induction n with
44  | zero => rw [sumRange_zero, sumRange_zero, zero_mul]
45  | succ n ih => rw [sumRange_succ, sumRange_succ, ih, right_distrib]

47⧉theorem sum_mul_left (f : Nat → Shell p) (c : Shell p) (n : Nat) :
48    sumRange (fun l => c * f l) n = c * sumRange f n := by
49  induction n with
50  | zero => rw [sumRange_zero, sumRange_zero, mul_zero]
51  | succ n ih => rw [sumRange_succ, sumRange_succ, ih, left_distrib]
52
53⧉theorem sum_neg (f : Nat → Shell p) (n : Nat) : sumRange (fun l => -(f l)) n = -(sumRange f n) := by
54  induction n with
55  | zero => rw [sumRange_zero, sumRange_zero, neg_zero]
56  | succ n ih => rw [sumRange_succ, sumRange_succ, ih, neg_add_rev]
57
58⧉theorem sum_const (c : Shell p) (n : Nat) : sumRange (fun _ => c) n = ofNat n * c := by
59  induction n with
60  | zero => rw [sumRange_zero]; exact (zero_mul c).symm
61  | succ n ih =>
62    rw [sumRange_succ, ih]
63    have : (ofNat (n + 1) : Shell p) = ofNat n + 1 := ext (by
64      rw [val_ofNat, val_add, val_ofNat, val_one, FRC.Nat.mod_add_mod _ _ _ hp, FRC.Nat.add_mod_mod _ _ _ hp])
65    rw [this, right_distrib, one_mul]

67/-- The telescoping geometric sum: `(Σ_{l<n} x^l)·(x − 1) = x^n − 1`. -/
68⧉theorem geom_sum_mul (x : Shell p) (n : Nat) :
69    sumRange (fun l => x ^ l) n * (x + -1) = x ^ n + -1 := by
70  induction n with
71  | zero => rw [sumRange_zero, zero_mul, pow_zero, add_neg]
72  | succ n ih =>
73    rw [sumRange_succ, right_distrib, ih, left_distrib, ← mul_neg, mul_one, ← pow_succ]
74    rw [add_add_add_comm, add_comm (x ^ n) (x ^ (n + 1)), add_comm (-1) (-(x ^ n)), add_assoc,
75      ← add_assoc (x ^ n), add_neg, zero_add]

80/-- `Σ_{l<n} x^l = 0` when `x^n = 1` and `x ≠ 1` (the field has no zero divisors). -/
81⧉theorem geom_sum_eq_zero (F : Frame p κ g) {x : Shell p} (n : Nat) (hn : x ^ n = 1) (hx : x ≠ 1) :
82    sumRange (fun l => x ^ l) n = 0 := by
83  have h := geom_sum_mul x n
84  rw [hn, add_neg] at h
85  match F.mul_eq_zero h with
86  | .inl e => exact e
87  | .inr e => exact absurd (by
88      calc x = x + 0 := (add_zero x).symm
89        _ = x + (-1 + 1) := by rw [neg_add]
90        _ = (x + -1) + 1 := (add_assoc _ _ _).symm
91        _ = 1 := by rw [e, zero_add]) hx

93⧉theorem ofNat_n (F : Frame p κ g) : (ofNat (p - 1) : Shell p) = -1 := by

96    FRC.Nat.mod_eq_of_lt (Nat.sub_lt Pos.pos (Nat.zero_lt_succ 0))]

98/-- 2:F1 (Prop. 6.1) — `g` is a principal root of unity: `g^n = 1`, `Σ_{j<n} (g^k)^j = 0` for
99`0 < k < n`, and `n = p − 1` reads as `−1` in the shell. -/
100⧉theorem principal_root (F : Frame p κ g) (k : Nat) (hk0 : 0 < k) (hk : k < p - 1) :
101    g ^ (p - 1) = 1 ∧ sumRange (fun j => (g ^ k) ^ j) (p - 1) = 0 ∧ (ofNat (p - 1) : Shell p) = -1 := by
102  refine ⟨F.pow_n, ?_, F.ofNat_n⟩
103  apply F.geom_sum_eq_zero
104  · rw [pow_mul_comm, F.pow_n, one_pow]
105  · exact F.prim.2 k hk hk0

107/-- `g^{k + (n − j)} = 1` exactly when `k = j`, for `k, j < n`. -/
108⧉theorem pow_shift_eq_one_iff (F : Frame p κ g) {k j : Nat} (hk : k < p - 1) (hj : j < p - 1) :
109    g ^ (k + (p - 1 - j)) = 1 ↔ k = j := by
110  have hn := F.n_pos
111  have hjn : p - 1 - j + j = p - 1 := FRC.Nat.sub_add_cancel (Nat.le_of_lt hj)
112  constructor
113  · intro h
114    have hm := F.mod_eq_zero_of_pow_eq_one h
115    match Nat.lt_or_ge k j with
116    | .inl hlt =>
117      have hlt' : k + (p - 1 - j) < p - 1 := by
118        have := Nat.add_lt_add_right hlt (p - 1 - j)
119        rw [Nat.add_comm j, hjn] at this
120        exact this
121      rw [FRC.Nat.mod_eq_of_lt hlt'] at hm
122      have hpos : 0 < p - 1 - j := by
123        refine Nat.lt_of_add_lt_add_right (n := j) ?_
124        rw [Nat.zero_add, hjn]; exact hj
125      exact absurd hm (Nat.ne_of_gt (Nat.lt_of_lt_of_le hpos (Nat.le_add_left _ _)))
126    | .inr hge =>
127      have e : k + (p - 1 - j) = (p - 1) * 1 + (k - j) := by
128        rw [Nat.mul_one]
129        calc k + (p - 1 - j) = (j + (k - j)) + (p - 1 - j) := by rw [FRC.Nat.add_sub_of_le hge]
130          _ = (p - 1 - j + j) + (k - j) := by
131              rw [Nat.add_comm j (k - j), Nat.add_assoc, Nat.add_comm (k - j), Nat.add_comm j (p - 1 - j)]
132          _ = p - 1 + (k - j) := by rw [hjn]
133      rw [e, FRC.Nat.add_mul_mod_self_left _ _ _ hn,
134        FRC.Nat.mod_eq_of_lt (Nat.lt_of_le_of_lt (Nat.sub_le k j) hk)] at hm
135      have := FRC.Nat.add_sub_of_le hge
136      rw [hm, Nat.add_zero] at this
137      exact this.symm
138  · intro e
139    rw [e, FRC.Nat.add_sub_of_le (Nat.le_of_lt hj), F.pow_n]

141/-- 2:F3 (Prop. 6.3), 6:B5 — the inversion of the shell Fourier matrix, entrywise: with
142`W k j = g^{jk}` and `W' l j = −g^{(n−j)l}` (`= −g^{−lj}`), `Σ_{l<n} W k l · W' l j = [k = j]`. -/
143⧉theorem dft_inverse (F : Frame p κ g) {k j : Nat} (hk : k < p - 1) (hj : j < p - 1) :
144    sumRange (fun l => g ^ (l * k) * -(g ^ ((p - 1 - j) * l))) (p - 1) = if k = j then 1 else 0 := by
145  have hsum : sumRange (fun l => g ^ (l * k) * -(g ^ ((p - 1 - j) * l))) (p - 1)
146      = -(sumRange (fun l => (g ^ (k + (p - 1 - j))) ^ l) (p - 1)) := by
147    rw [← sum_neg]
148    apply sum_congr
149    intro l _
150    rw [← mul_neg, ← pow_add, ← pow_mul, Nat.mul_comm l k, ← FRC.Nat.add_mul]
151  rw [hsum]
152  exact match Nat.decEq k j with
153    | .isTrue e => by
154        rw [if_pos e]
155        have h1 : g ^ (k + (p - 1 - j)) = 1 := (F.pow_shift_eq_one_iff hk hj).2 e
156        have : sumRange (fun l => (g ^ (k + (p - 1 - j))) ^ l) (p - 1) = ofNat (p - 1) * 1 := by
157          rw [← sum_const]; apply sum_congr; intro l _; rw [h1, one_pow]
158        rw [this, F.ofNat_n, mul_one, neg_neg]
159    | .isFalse e => by
160        rw [if_neg e]
161        have h1 : g ^ (k + (p - 1 - j)) ≠ 1 := fun h => e ((F.pow_shift_eq_one_iff hk hj).1 h)
162        rw [F.geom_sum_eq_zero _ (by rw [pow_mul_comm, F.pow_n, one_pow]) h1, neg_zero]

164/-- `g^{k + j} = 1` exactly when `(k + j) % n = 0`, i.e. `j` is the reversal `−k` of `k` (`k, j < n`). -/

168/-- 6:B5 (the shell Fourier matrix squares to the reversal, entrywise): with `W k j = g^{jk}` and
169`J k j = [(k + j) % n = 0]`, `Σ_{l<n} W k l · W l j = −J k j` on every shell (`n = p − 1 ≡ −1`). -/
170⧉theorem W_sq (F : Frame p κ g) (k j : Nat) :
171    sumRange (fun l => g ^ (l * k) * g ^ (j * l)) (p - 1)
172      = -(if (k + j) % (p - 1) = 0 then 1 else 0) := by
173  have hsum : sumRange (fun l => g ^ (l * k) * g ^ (j * l)) (p - 1)
174      = sumRange (fun l => (g ^ (k + j)) ^ l) (p - 1) := by
175    apply sum_congr; intro l _
176    rw [← pow_add, ← pow_mul, Nat.mul_comm l k, ← FRC.Nat.add_mul]
177  rw [hsum]
178  exact match Nat.decEq ((k + j) % (p - 1)) 0 with
179    | .isTrue e => by
180        rw [if_pos e]
181        have h1 : g ^ (k + j) = 1 := F.pow_eq_one_of_mod e
182        have : sumRange (fun l => (g ^ (k + j)) ^ l) (p - 1) = ofNat (p - 1) * 1 := by
183          rw [← sum_const]; apply sum_congr; intro l _; rw [h1, one_pow]
184        rw [this, F.ofNat_n, mul_one]
185    | .isFalse e => by
186        rw [if_neg e, neg_zero]
187        exact F.geom_sum_eq_zero _ (by rw [pow_mul_comm, F.pow_n, one_pow]) (fun h => e (F.mod_eq_zero_of_pow_eq_one h))
