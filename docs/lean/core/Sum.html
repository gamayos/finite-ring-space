4/-!
5# FrcCore.Sum — finite sums on the shell, the geometric sum, the Fourier inversion
6
7Sums over `l < n` are defined by structural recursion (`sumRange f (n+1) = sumRange f n + f n`): no
8`Finset`, no quotient, no function extensionality — a congruence lemma (`sum_congr`) replaces `funext`.
9The geometric sum gives the principal-root identity (2:F1, Prop. 6.1 of 2-geometry) and the entrywise
10inversion of the shell Fourier matrix `W k j = g^{jk}`: `Σ_l W k l · (−g^{−lj}) = [k = j]` (2:F3,
11Prop. 6.3; 6:B5 in matrix form). No axioms.
12-/

19/-- `sumRange f n = f 0 + f 1 + ⋯ + f (n−1)`. -/
20⧉def sumRange (f : Nat → Shell p) : Nat → Shell p
21  | 0 => 0
22  | n + 1 => sumRange f n + f n

24⧉theorem sumRange_zero (f : Nat → Shell p) : sumRange f 0 = 0 := rfl
25⧉theorem sumRange_succ (f : Nat → Shell p) (n : Nat) : sumRange f (n + 1) = sumRange f n + f n := rfl

27⧉theorem sum_congr {f h : Nat → Shell p} (n : Nat) (e : ∀ l, l < n → f l = h l) :
28    sumRange f n = sumRange h n := by
29  induction n with
30  | zero => rfl
31  | succ n ih =>
32    rw [sumRange_succ, sumRange_succ, ih (fun l hl => e l (Nat.lt_succ_of_lt hl)),
33      e n (Nat.lt_succ_self n)]

40    rw [sumRange_succ, sumRange_succ, sumRange_succ, ih, add_add_add_comm]

59⧉theorem sum_const (c : Shell p) (n : Nat) : sumRange (fun _ => c) n = ofNat n * c := by

73/-- A sum with a single nonzero term. -/
74⧉theorem sum_eq_single {f : Nat → Shell p} {l₀ : Nat} : ∀ {n : Nat}, l₀ < n → (∀ l, l < n → l ≠ l₀ → f l = 0) →

87/-- The telescoping geometric sum: `(Σ_{l<n} x^l)·(x − 1) = x^n − 1`. -/
88⧉theorem geom_sum_mul (x : Shell p) (n : Nat) :
89    sumRange (fun l => x ^ l) n * (x + -1) = x ^ n + -1 := by
90  induction n with
91  | zero => rw [sumRange_zero, zero_mul, pow_zero, add_neg]
92  | succ n ih =>
93    rw [sumRange_succ, right_distrib, ih, left_distrib, ← mul_neg, mul_one, ← pow_succ]
94    rw [add_add_add_comm, add_comm (x ^ n) (x ^ (n + 1)), add_comm (-1) (-(x ^ n)), add_assoc,
95      ← add_assoc (x ^ n), add_neg, zero_add]

119/-- 2:F1 (Prop. 6.1) — `g` is a principal root of unity: `g^n = 1`, `Σ_{j<n} (g^k)^j = 0` for

128/-- `g^{k + (n − j)} = 1` exactly when `k = j`, for `k, j < n`. -/
129⧉theorem pow_shift_eq_one_iff (F : Frame p κ g) {k j : Nat} (hk : k < p - 1) (hj : j < p - 1) :
130    g ^ (k + (p - 1 - j)) = 1 ↔ k = j := by
131  have hn := F.n_pos
132  have hjn : p - 1 - j + j = p - 1 := FRC.Nat.sub_add_cancel (Nat.le_of_lt hj)
133  constructor
134  · intro h
135    have hm := F.mod_eq_zero_of_pow_eq_one h
136    match Nat.lt_or_ge k j with
137    | .inl hlt =>
138      have hlt' : k + (p - 1 - j) < p - 1 := by
139        have := Nat.add_lt_add_right hlt (p - 1 - j)
140        rw [Nat.add_comm j, hjn] at this
141        exact this
142      rw [FRC.Nat.mod_eq_of_lt hlt'] at hm
143      have hpos : 0 < p - 1 - j := by
144        refine Nat.lt_of_add_lt_add_right (n := j) ?_
145        rw [Nat.zero_add, hjn]; exact hj
146      exact absurd hm (Nat.ne_of_gt (Nat.lt_of_lt_of_le hpos (Nat.le_add_left _ _)))
147    | .inr hge =>
148      have e : k + (p - 1 - j) = (p - 1) * 1 + (k - j) := by
149        rw [Nat.mul_one]
150        calc k + (p - 1 - j) = (j + (k - j)) + (p - 1 - j) := by rw [FRC.Nat.add_sub_of_le hge]
151          _ = (p - 1 - j + j) + (k - j) := by
152              rw [Nat.add_comm j (k - j), Nat.add_assoc, Nat.add_comm (k - j), Nat.add_comm j (p - 1 - j)]
153          _ = p - 1 + (k - j) := by rw [hjn]
154      rw [e, FRC.Nat.add_mul_mod_self_left _ _ _ hn,
155        FRC.Nat.mod_eq_of_lt (Nat.lt_of_le_of_lt (Nat.sub_le k j) hk)] at hm
156      have := FRC.Nat.add_sub_of_le hge
157      rw [hm, Nat.add_zero] at this
158      exact this.symm
159  · intro e
160    rw [e, FRC.Nat.add_sub_of_le (Nat.le_of_lt hj), F.pow_n]

162/-- 2:F3 (Prop. 6.3), 6:B5 — the inversion of the shell Fourier matrix, entrywise: with
163`W k j = g^{jk}` and `W' l j = −g^{(n−j)l}` (`= −g^{−lj}`), `Σ_{l<n} W k l · W' l j = [k = j]`. -/
164⧉theorem dft_inverse (F : Frame p κ g) {k j : Nat} (hk : k < p - 1) (hj : j < p - 1) :
165    sumRange (fun l => g ^ (l * k) * -(g ^ ((p - 1 - j) * l))) (p - 1) = if k = j then 1 else 0 := by
166  have hsum : sumRange (fun l => g ^ (l * k) * -(g ^ ((p - 1 - j) * l))) (p - 1)
167      = -(sumRange (fun l => (g ^ (k + (p - 1 - j))) ^ l) (p - 1)) := by
168    rw [← sum_neg]
169    apply sum_congr
170    intro l _
171    rw [← mul_neg, ← pow_add, ← pow_mul, Nat.mul_comm l k, ← FRC.Nat.add_mul]
172  rw [hsum]
173  exact match Nat.decEq k j with
174    | .isTrue e => by
175        rw [ite_eq_left e]
176        have h1 : g ^ (k + (p - 1 - j)) = 1 := (F.pow_shift_eq_one_iff hk hj).2 e
177        have : sumRange (fun l => (g ^ (k + (p - 1 - j))) ^ l) (p - 1) = ofNat (p - 1) * 1 := by
178          rw [← sum_const]; apply sum_congr; intro l _; rw [h1, one_pow]
179        rw [this, F.ofNat_n, mul_one, neg_neg]
180    | .isFalse e => by
181        rw [ite_eq_right e]
182        have h1 : g ^ (k + (p - 1 - j)) ≠ 1 := fun h => e ((F.pow_shift_eq_one_iff hk hj).1 h)
183        rw [F.geom_sum_eq_zero _ (by rw [pow_mul_comm, F.pow_n, one_pow]) h1, neg_zero]

189/-- 6:B5 (the shell Fourier matrix squares to the reversal, entrywise): with `W k j = g^{jk}` and
190`J k j = [(k + j) % n = 0]`, `Σ_{l<n} W k l · W l j = −J k j` on every shell (`n = p − 1 ≡ −1`). -/
191⧉theorem W_sq (F : Frame p κ g) (k j : Nat) :

212/-- The reversal `rev n k = (n − k) % n`: the index `l` with `(k + l) % n = 0`. -/
213⧉def rev (n k : Nat) : Nat := (n - k) % n
214
215⧉theorem rev_lt {n : Nat} (hn : 0 < n) (k : Nat) : rev n k < n := Nat.mod_lt _ hn
216
217⧉theorem rev_zero (n : Nat) (hn : 0 < n) : rev n 0 = 0 := by
218  unfold rev; rw [Nat.sub_zero]; exact FRC.Nat.mod_self n hn
219
220⧉theorem rev_of_pos {n k : Nat} (hk : k < n) (hk0 : 0 < k) : rev n k = n - k := by
221  unfold rev; exact FRC.Nat.mod_eq_of_lt (Nat.sub_lt (Nat.lt_of_lt_of_le hk0 (Nat.le_of_lt hk)) hk0)

223⧉theorem rev_add_mod {n k : Nat} (hk : k < n) : (rev n k + k) % n = 0 := by
224  have hn : 0 < n := Nat.lt_of_le_of_lt (Nat.zero_le k) hk
225  exact match Nat.decEq k 0 with
226    | isTrue e => by rw [e, rev_zero n hn]; rfl
227    | isFalse e => by
228        rw [rev_of_pos hk (Nat.pos_of_ne_zero e), FRC.Nat.sub_add_cancel (Nat.le_of_lt hk)]
229        exact FRC.Nat.mod_self n hn

231⧉theorem rev_rev {n k : Nat} (hk : k < n) : rev n (rev n k) = k := by
232  have hn : 0 < n := Nat.lt_of_le_of_lt (Nat.zero_le k) hk
233  exact match Nat.decEq k 0 with
234    | isTrue e => by rw [e, rev_zero n hn, rev_zero n hn]
235    | isFalse e => by
236        have hk0 := Nat.pos_of_ne_zero e
237        rw [rev_of_pos hk hk0]
238        have hnk : n - k < n := Nat.sub_lt (Nat.lt_of_lt_of_le hk0 (Nat.le_of_lt hk)) hk0
239        have hnk0 : 0 < n - k := by
240          refine Nat.lt_of_add_lt_add_right (n := k) ?_
241          rw [Nat.zero_add, FRC.Nat.sub_add_cancel (Nat.le_of_lt hk)]; exact hk
242        rw [rev_of_pos hnk hnk0]
243        calc n - (n - k) = (n - k + k) - (n - k) := by rw [FRC.Nat.sub_add_cancel (Nat.le_of_lt hk)]
244          _ = k := FRC.Nat.add_sub_cancel_left _ _

246/-- `(k + l) % n = 0` exactly when `l` is the reversal of `k` (`k, l < n`). -/
247⧉theorem add_mod_eq_zero_iff {n k l : Nat} (hk : k < n) (hl : l < n) : (k + l) % n = 0 ↔ l = rev n k := by
248  have hn : 0 < n := Nat.lt_of_le_of_lt (Nat.zero_le k) hk
249  constructor
250  · intro h
251    exact match Nat.decEq k 0 with
252      | isTrue e => by
253          rw [e, Nat.zero_add, FRC.Nat.mod_eq_of_lt hl] at h
254          rw [e, rev_zero n hn, h]
255      | isFalse e => by
256          have hk0 := Nat.pos_of_ne_zero e
257          rw [rev_of_pos hk hk0]
258          exact match Nat.lt_or_ge l (n - k) with
259            | Or.inl hlt => by
260                have : k + l < n := by
261                  have := Nat.add_lt_add_left hlt k
262                  rw [FRC.Nat.add_sub_of_le (Nat.le_of_lt hk)] at this; exact this
263                rw [FRC.Nat.mod_eq_of_lt this] at h
264                exact absurd h (Nat.ne_of_gt (Nat.lt_of_lt_of_le hk0 (Nat.le_add_right k l)))
265            | Or.inr hge => by
266                have e1 : k + l = n * 1 + (l - (n - k)) := by
267                  rw [Nat.mul_one]
268                  calc k + l = k + ((n - k) + (l - (n - k))) := by rw [FRC.Nat.add_sub_of_le hge]
269                    _ = (k + (n - k)) + (l - (n - k)) := (Nat.add_assoc _ _ _).symm
270                    _ = n + (l - (n - k)) := by rw [FRC.Nat.add_sub_of_le (Nat.le_of_lt hk)]
271                rw [e1, FRC.Nat.add_mul_mod_self_left _ _ _ hn,
272                  FRC.Nat.mod_eq_of_lt (Nat.lt_of_le_of_lt (Nat.sub_le _ _) hl)] at h
273                have := FRC.Nat.add_sub_of_le hge
274                rw [h, Nat.add_zero] at this
275                exact this.symm
276  · intro e; rw [e, Nat.add_comm]; exact rev_add_mod hk

278/-- The shell Fourier matrix, `W k j = g^{jk}` (6:B5's convention). -/
279⧉def W (g : Shell p) (k j : Nat) : Shell p := g ^ (j * k)

281/-- The reversal matrix `J k j = [(k + j) % n = 0]`. -/
282⧉def J (n k j : Nat) : Shell p := if (k + j) % n = 0 then 1 else 0

284/-- The quarter-turn transform `F = i·W`, `i = −g^κ`. -/
285⧉def Fmat (g : Shell p) (κ k j : Nat) : Shell p := quarterTurn g κ * W g k j

322/-- 6:B7 (`W J = J W`, entrywise). -/
323⧉theorem W_J_comm (F : Frame p κ g) {k j : Nat} (hk : k < p - 1) (hj : j < p - 1) :
324    sumRange (fun l => W g k l * J (p - 1) l j) (p - 1) = sumRange (fun l => J (p - 1) k l * W g l j) (p - 1) := by
325  have hn := F.n_pos
326  have hrj : rev (p - 1) j < p - 1 := rev_lt hn j
327  have hrk : rev (p - 1) k < p - 1 := rev_lt hn k
328  rw [sum_eq_single hrj (fun l hl hne => by
329        rw [J_eq hl hj, ite_eq_right (fun e => hne (by rw [e, rev_rev hl])), mul_zero])]
330  rw [sum_eq_single hrk (fun l hl hne => by rw [J_eq hk hl, ite_eq_right hne, zero_mul])]
331  rw [J_eq hrj hj, ite_eq_left (rev_rev hj).symm, mul_one, J_eq hk hrk, ite_eq_left rfl, one_mul]
