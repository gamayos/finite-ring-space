3/-!
4# FrcCore.Sum — finite sums on the shell, the geometric sum, the Fourier inversion
5
6Sums over `l < n` are defined by structural recursion (`sumRange f (n+1) = sumRange f n + f n`): no
7`Finset`, no quotient, no function extensionality — a congruence lemma (`sum_congr`) replaces `funext`.
8The geometric sum gives the principal-root identity (2:F1, Prop. 6.1 of 2-geometry) and the entrywise
9inversion of the shell Fourier matrix `W k j = g^{jk}`: `Σ_l W k l · (−g^{−lj}) = [k = j]` (2:F3,
10Prop. 6.3; 6:B5 in matrix form). No axioms.
11-/

18/-- `sumRange f n = f 0 + f 1 + ⋯ + f (n−1)`. -/
19⧉def sumRange (f : Nat → Shell p) : Nat → Shell p
20  | 0 => 0
21  | n + 1 => sumRange f n + f n

23⧉theorem sumRange_zero (f : Nat → Shell p) : sumRange f 0 = 0 := rfl
24⧉theorem sumRange_succ (f : Nat → Shell p) (n : Nat) : sumRange f (n + 1) = sumRange f n + f n := rfl

26⧉theorem sum_congr {f h : Nat → Shell p} (n : Nat) (e : ∀ l, l < n → f l = h l) :
27    sumRange f n = sumRange h n := by
28  induction n with
29  | zero => rfl
30  | succ n ih =>
31    rw [sumRange_succ, sumRange_succ, ih (fun l hl => e l (Nat.lt_succ_of_lt hl)),
32      e n (Nat.lt_succ_self n)]

58⧉theorem sum_const (c : Shell p) (n : Nat) : sumRange (fun _ => c) n = ofNat n * c := by

72/-- A sum with a single nonzero term. -/
73⧉theorem sum_eq_single {f : Nat → Shell p} {l₀ : Nat} : ∀ {n : Nat}, l₀ < n → (∀ l, l < n → l ≠ l₀ → f l = 0) →

86/-- The telescoping geometric sum: `(Σ_{l<n} x^l)·(x − 1) = x^n − 1`. -/
87⧉theorem geom_sum_mul (x : Shell p) (n : Nat) :
88    sumRange (fun l => x ^ l) n * (x + -1) = x ^ n + -1 := by
89  induction n with
90  | zero => rw [sumRange_zero, zero_mul, pow_zero, add_neg]
91  | succ n ih =>
92    rw [sumRange_succ, right_distrib, ih, left_distrib, ← mul_neg, mul_one, ← pow_succ]
93    rw [add_add_add_comm, add_comm (x ^ n) (x ^ (n + 1)), add_comm (-1) (-(x ^ n)), add_assoc,
94      ← add_assoc (x ^ n), add_neg, zero_add]

118/-- 2:F1 (Prop. 6.1) — `g` is a principal root of unity: `g^n = 1`, `Σ_{j<n} (g^k)^j = 0` for

120⧉theorem principal_root (F : Frame p κ g) (k : Nat) (hk0 : 0 < k) (hk : k < p - 1) :

127/-- `g^{k + (n − j)} = 1` exactly when `k = j`, for `k, j < n`. -/
128⧉theorem pow_shift_eq_one_iff (F : Frame p κ g) {k j : Nat} (hk : k < p - 1) (hj : j < p - 1) :
129    g ^ (k + (p - 1 - j)) = 1 ↔ k = j := by
130  have hn := F.n_pos
131  have hjn : p - 1 - j + j = p - 1 := FRC.Nat.sub_add_cancel (Nat.le_of_lt hj)
132  constructor
133  · intro h
134    have hm := F.mod_eq_zero_of_pow_eq_one h
135    match Nat.lt_or_ge k j with
136    | .inl hlt =>
137      have hlt' : k + (p - 1 - j) < p - 1 := by
138        have := Nat.add_lt_add_right hlt (p - 1 - j)
139        rw [Nat.add_comm j, hjn] at this
140        exact this
141      rw [FRC.Nat.mod_eq_of_lt hlt'] at hm
142      have hpos : 0 < p - 1 - j := by
143        refine Nat.lt_of_add_lt_add_right (n := j) ?_
144        rw [Nat.zero_add, hjn]; exact hj
145      exact absurd hm (Nat.ne_of_gt (Nat.lt_of_lt_of_le hpos (Nat.le_add_left _ _)))
146    | .inr hge =>
147      have e : k + (p - 1 - j) = (p - 1) * 1 + (k - j) := by
148        rw [Nat.mul_one]
149        calc k + (p - 1 - j) = (j + (k - j)) + (p - 1 - j) := by rw [FRC.Nat.add_sub_of_le hge]
150          _ = (p - 1 - j + j) + (k - j) := by
151              rw [Nat.add_comm j (k - j), Nat.add_assoc, Nat.add_comm (k - j), Nat.add_comm j (p - 1 - j)]
152          _ = p - 1 + (k - j) := by rw [hjn]
153      rw [e, FRC.Nat.add_mul_mod_self_left _ _ _ hn,
154        FRC.Nat.mod_eq_of_lt (Nat.lt_of_le_of_lt (Nat.sub_le k j) hk)] at hm
155      have := FRC.Nat.add_sub_of_le hge
156      rw [hm, Nat.add_zero] at this
157      exact this.symm
158  · intro e
159    rw [e, FRC.Nat.add_sub_of_le (Nat.le_of_lt hj), F.pow_n]

161/-- 2:F3 (Prop. 6.3), 6:B5 — the inversion of the shell Fourier matrix, entrywise: with
162`W k j = g^{jk}` and `W' l j = −g^{(n−j)l}` (`= −g^{−lj}`), `Σ_{l<n} W k l · W' l j = [k = j]`. -/
163⧉theorem dft_inverse (F : Frame p κ g) {k j : Nat} (hk : k < p - 1) (hj : j < p - 1) :
164    sumRange (fun l => g ^ (l * k) * -(g ^ ((p - 1 - j) * l))) (p - 1) = if k = j then 1 else 0 := by
165  have hsum : sumRange (fun l => g ^ (l * k) * -(g ^ ((p - 1 - j) * l))) (p - 1)
166      = -(sumRange (fun l => (g ^ (k + (p - 1 - j))) ^ l) (p - 1)) := by
167    rw [← sum_neg]
168    apply sum_congr
169    intro l _
170    rw [← mul_neg, ← pow_add, ← pow_mul, Nat.mul_comm l k, ← FRC.Nat.add_mul]
171  rw [hsum]
172  exact match Nat.decEq k j with
173    | .isTrue e => by
174        rw [if_pos e]
175        have h1 : g ^ (k + (p - 1 - j)) = 1 := (F.pow_shift_eq_one_iff hk hj).2 e
176        have : sumRange (fun l => (g ^ (k + (p - 1 - j))) ^ l) (p - 1) = ofNat (p - 1) * 1 := by
177          rw [← sum_const]; apply sum_congr; intro l _; rw [h1, one_pow]
178        rw [this, F.ofNat_n, mul_one, neg_neg]
179    | .isFalse e => by
180        rw [if_neg e]
181        have h1 : g ^ (k + (p - 1 - j)) ≠ 1 := fun h => e ((F.pow_shift_eq_one_iff hk hj).1 h)
182        rw [F.geom_sum_eq_zero _ (by rw [pow_mul_comm, F.pow_n, one_pow]) h1, neg_zero]

211/-- The reversal `rev n k = (n − k) % n`: the index `l` with `(k + l) % n = 0`. -/
212⧉def rev (n k : Nat) : Nat := (n - k) % n

214⧉theorem rev_lt {n : Nat} (hn : 0 < n) (k : Nat) : rev n k < n := Nat.mod_lt _ hn

216⧉theorem rev_zero (n : Nat) (hn : 0 < n) : rev n 0 = 0 := by
217  unfold rev; rw [Nat.sub_zero]; exact FRC.Nat.mod_self n hn

219⧉theorem rev_of_pos {n k : Nat} (hk : k < n) (hk0 : 0 < k) : rev n k = n - k := by
220  unfold rev; exact FRC.Nat.mod_eq_of_lt (Nat.sub_lt (Nat.lt_of_lt_of_le hk0 (Nat.le_of_lt hk)) hk0)

222⧉theorem rev_add_mod {n k : Nat} (hk : k < n) : (rev n k + k) % n = 0 := by
223  have hn : 0 < n := Nat.lt_of_le_of_lt (Nat.zero_le k) hk
224  exact match Nat.decEq k 0 with
225    | isTrue e => by rw [e, rev_zero n hn]; rfl
226    | isFalse e => by
227        rw [rev_of_pos hk (Nat.pos_of_ne_zero e), FRC.Nat.sub_add_cancel (Nat.le_of_lt hk)]
228        exact FRC.Nat.mod_self n hn

230⧉theorem rev_rev {n k : Nat} (hk : k < n) : rev n (rev n k) = k := by
231  have hn : 0 < n := Nat.lt_of_le_of_lt (Nat.zero_le k) hk
232  exact match Nat.decEq k 0 with
233    | isTrue e => by rw [e, rev_zero n hn, rev_zero n hn]
234    | isFalse e => by
235        have hk0 := Nat.pos_of_ne_zero e
236        rw [rev_of_pos hk hk0]
237        have hnk : n - k < n := Nat.sub_lt (Nat.lt_of_lt_of_le hk0 (Nat.le_of_lt hk)) hk0
238        have hnk0 : 0 < n - k := by
239          refine Nat.lt_of_add_lt_add_right (n := k) ?_
240          rw [Nat.zero_add, FRC.Nat.sub_add_cancel (Nat.le_of_lt hk)]; exact hk
241        rw [rev_of_pos hnk hnk0]
242        calc n - (n - k) = (n - k + k) - (n - k) := by rw [FRC.Nat.sub_add_cancel (Nat.le_of_lt hk)]
243          _ = k := FRC.Nat.add_sub_cancel_left _ _

245/-- `(k + l) % n = 0` exactly when `l` is the reversal of `k` (`k, l < n`). -/
246⧉theorem add_mod_eq_zero_iff {n k l : Nat} (hk : k < n) (hl : l < n) : (k + l) % n = 0 ↔ l = rev n k := by
247  have hn : 0 < n := Nat.lt_of_le_of_lt (Nat.zero_le k) hk
248  constructor
249  · intro h
250    exact match Nat.decEq k 0 with
251      | isTrue e => by
252          rw [e, Nat.zero_add, FRC.Nat.mod_eq_of_lt hl] at h
253          rw [e, rev_zero n hn, h]
254      | isFalse e => by
255          have hk0 := Nat.pos_of_ne_zero e
256          rw [rev_of_pos hk hk0]
257          exact match Nat.lt_or_ge l (n - k) with
258            | Or.inl hlt => by
259                have : k + l < n := by
260                  have := Nat.add_lt_add_left hlt k
261                  rw [FRC.Nat.add_sub_of_le (Nat.le_of_lt hk)] at this; exact this
262                rw [FRC.Nat.mod_eq_of_lt this] at h
263                exact absurd h (Nat.ne_of_gt (Nat.lt_of_lt_of_le hk0 (Nat.le_add_right k l)))
264            | Or.inr hge => by
265                have e1 : k + l = n * 1 + (l - (n - k)) := by
266                  rw [Nat.mul_one]
267                  calc k + l = k + ((n - k) + (l - (n - k))) := by rw [FRC.Nat.add_sub_of_le hge]
268                    _ = (k + (n - k)) + (l - (n - k)) := (Nat.add_assoc _ _ _).symm
269                    _ = n + (l - (n - k)) := by rw [FRC.Nat.add_sub_of_le (Nat.le_of_lt hk)]
270                rw [e1, FRC.Nat.add_mul_mod_self_left _ _ _ hn,
271                  FRC.Nat.mod_eq_of_lt (Nat.lt_of_le_of_lt (Nat.sub_le _ _) hl)] at h
272                have := FRC.Nat.add_sub_of_le hge
273                rw [h, Nat.add_zero] at this
274                exact this.symm
275  · intro e; rw [e, Nat.add_comm]; exact rev_add_mod hk

277/-- The shell Fourier matrix, `W k j = g^{jk}` (6:B5's convention). -/
278⧉def W (g : Shell p) (k j : Nat) : Shell p := g ^ (j * k)

280/-- The reversal matrix `J k j = [(k + j) % n = 0]`. -/
281⧉def J (n k j : Nat) : Shell p := if (k + j) % n = 0 then 1 else 0

283/-- The quarter-turn transform `F = i·W`, `i = −g^κ`. -/
284⧉def Fmat (g : Shell p) (κ k j : Nat) : Shell p := quarterTurn g κ * W g k j

321/-- 6:B7 (`W J = J W`, entrywise). -/
322⧉theorem W_J_comm (F : Frame p κ g) {k j : Nat} (hk : k < p - 1) (hj : j < p - 1) :
323    sumRange (fun l => W g k l * J (p - 1) l j) (p - 1) = sumRange (fun l => J (p - 1) k l * W g l j) (p - 1) := by
324  have hn := F.n_pos
325  have hrj : rev (p - 1) j < p - 1 := rev_lt hn j
326  have hrk : rev (p - 1) k < p - 1 := rev_lt hn k
327  rw [sum_eq_single hrj (fun l hl hne => by
328        rw [J_eq hl hj, if_neg (fun e => hne (by rw [e, rev_rev hl])), mul_zero])]
329  rw [sum_eq_single hrk (fun l hl hne => by rw [J_eq hk hl, if_neg hne, zero_mul])]
330  rw [J_eq hrj hj, if_pos (rev_rev hj).symm, mul_one, J_eq hk hrk, if_pos rfl, one_mul]
331  unfold W
