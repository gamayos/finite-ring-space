122/-- `∀ m < n, P m`, decided by recursion on `n` — Lean's own instance carries `propext`; this one nothing. -/
123⧉def decForallLT (P : Nat → Prop) [DecidablePred P] : (n : Nat) → Decidable (∀ m, m < n → P m)
124  | 0 => isTrue (fun m hm => absurd hm (Nat.not_lt_zero m))
125  | n + 1 =>
126    match decForallLT P n with
127    | isFalse h => isFalse (fun hall => h (fun m hm => hall m (Nat.lt_succ_of_lt hm)))
128    | isTrue h =>
129      if hn : P n then
130        isTrue (fun m hm =>
131          match Nat.lt_or_ge m n with
132          | .inl hlt => h m hlt
133          | .inr hge => (Nat.le_antisymm (Nat.le_of_lt_succ hm) hge) ▸ hn)
134      else isFalse (fun hall => hn (hall n (Nat.lt_succ_self n)))
135
136instance instDecForallLT (P : Nat → Prop) [DecidablePred P] (n : Nat) : Decidable (∀ m, m < n → P m) :=
137  decForallLT P n
138
139/-- `∀ x : Shell p, P x`, decided through the representatives. -/
140instance instDecForallShell {p : Nat} [Pos p] (P : Shell p → Prop) [DecidablePred P] :
141    Decidable (∀ x, P x) :=
142  match decForallLT (fun v => P (ofNat v)) p with
143  | isTrue h => isTrue (fun x => ofNat_val x ▸ h x.val x.lt)
144  | isFalse h => isFalse (fun hall => h (fun v _ => hall (ofNat v)))

146/-- `∀ x : Dom p n, P x`, decided componentwise. -/
147instance instDecForallDom {p n : Nat} [Pos p] [Pos n] (P : Dom p n → Prop) [DecidablePred P] :
148    Decidable (∀ x, P x) :=
149  match (inferInstance : Decidable (∀ r : Shell p, ∀ s : Shell n, P ⟨r, s⟩)) with
150  | isTrue h => isTrue (fun x => match x with | ⟨r, s⟩ => h r s)
151  | isFalse h => isFalse (fun hall => h (fun r s => hall ⟨r, s⟩))
