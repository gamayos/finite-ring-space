2# FrcCore.Nat — the arithmetic the shells stand on, from first principles
3
4No Mathlib, no `simp`, no `omega`: every theorem here is checked by `#print axioms` to depend on no
5axiom at all. Lean's own core library proves most of these with `propext` for convenience (see
6`reports/lean-axioms-probe-20260917/core_probe.log`); they are re-derived here by induction from the
7axiom-free primitives (`Nat.rec`, `Nat.add_comm`, `Nat.mul_comm`, `Nat.left_distrib`, `Nat.mod_lt`,
8`Nat.le.dest`, `ite_eq_left`, `ite_eq_right`, `Decidable.em`).
9
10The division algorithm is obtained from the definition of `Nat.mod` in `Init.Prelude` (the wrapper
11around `Nat.modCore`, itself a fuel recursion): `mod_eq_of_lt`, `mod_eq_sub_mod`, `mod_spec`
12(`x = p·q + x % p`) and `mod_unique` (the remainder is determined by any such decomposition). Everything
13about residues mod `p` follows from those four.

20⧉theorem add_left_cancel {a b c : Nat} (h : a + b = a + c) : b = c := by
21  induction a with
22  | zero => rw [Nat.zero_add, Nat.zero_add] at h; exact h
23  | succ a ih =>
24    apply ih
25    rw [Nat.succ_add, Nat.succ_add] at h
26    exact Nat.succ.inj h

28⧉theorem add_right_cancel {a b c : Nat} (h : a + c = b + c) : a = b :=
29  add_left_cancel (a := c) (by rw [Nat.add_comm c a, Nat.add_comm c b]; exact h)

108/-- The fuel of `Nat.modCore.go` does not matter once it exceeds the dividend. -/
109⧉theorem modCore_go_fuel {y : Nat} (hy : 0 < y) : ∀ (f1 f2 x : Nat) (h1 : x < f1) (h2 : x < f2),
110    Nat.modCore.go y hy f1 x h1 = Nat.modCore.go y hy f2 x h2 := by
111  intro f1
112  induction f1 with
113  | zero => intro f2 x h1; exact absurd h1 (Nat.not_lt_zero x)
114  | succ f1 ih =>
115    intro f2 x h1 h2
116    cases f2 with
117    | zero => exact absurd h2 (Nat.not_lt_zero x)
118    | succ f2 =>
119      show (if h : y ≤ x then Nat.modCore.go y hy f1 (x - y) _ else x)
120          = (if h : y ≤ x then Nat.modCore.go y hy f2 (x - y) _ else x)
121      exact match Nat.decLe y x with
122        | .isTrue h => by rw [dite_eq_left h, dite_eq_left h]; exact ih f2 (x - y) _ _
123        | .isFalse h => by rw [dite_eq_right h, dite_eq_right h]
124
125⧉theorem modCore_eq' (x y : Nat) (hy : 0 < y) :
126    Nat.modCore x y = if y ≤ x then Nat.modCore (x - y) y else x := by
127  unfold Nat.modCore
128  rw [dite_eq_left hy]
129  show (if h : y ≤ x then Nat.modCore.go y hy x (x - y) _ else x) = _
130  exact match Nat.decLe y x with
131    | .isTrue h => by
132        rw [dite_eq_left h, ite_eq_left h, dite_eq_left hy]
133        exact modCore_go_fuel hy x (x - y + 1) (x - y) _ _
134    | .isFalse h => by rw [dite_eq_right h, ite_eq_right h]
135
136⧉theorem modCore_eq_mod' (n m : Nat) (hm : 0 < m) : Nat.modCore n m = n % m := by
137  cases n with
138  | zero =>
139    show Nat.modCore 0 m = 0
140    rw [modCore_eq' 0 m hm, ite_eq_right (Nat.not_le_of_lt hm)]
141  | succ n =>
142    show Nat.modCore (n + 1) m = ite (m ≤ n + 1) (Nat.modCore (n + 1) m) (n + 1)
143    exact match Nat.decLe m (n + 1) with
144      | .isTrue h => by rw [ite_eq_left h]
145      | .isFalse h => by rw [ite_eq_right h, modCore_eq' _ _ hm, ite_eq_right h]
146
147⧉theorem mod_eq_of_lt {x y : Nat} (h : x < y) : x % y = x := by
148  cases x with
149  | zero => rfl
150  | succ n =>
151    show ite (y ≤ n + 1) (Nat.modCore (n + 1) y) (n + 1) = n + 1
152    rw [ite_eq_right (Nat.not_le_of_lt h)]

154⧉theorem mod_eq_sub_mod {x y : Nat} (hy : 0 < y) (h : y ≤ x) : x % y = (x - y) % y := by
155  rw [← modCore_eq_mod' x y hy, modCore_eq' x y hy, ite_eq_left h, modCore_eq_mod' _ _ hy]

157/-- The division algorithm: `x = p·q + x % p` for some `q`. -/
158⧉theorem mod_spec (p : Nat) (hp : 0 < p) : ∀ x : Nat, ∃ q, x = p * q + x % p := by
159  intro x
160  induction x using Nat.strongRecOn with
161  | _ x ih =>
162    exact match Nat.decLe p x with
163      | .isTrue h =>
164          match ih (x - p) (Nat.sub_lt (Nat.lt_of_lt_of_le hp h) hp) with
165          | ⟨q, hq⟩ => ⟨q + 1, by
166              rw [mod_eq_sub_mod hp h, Nat.mul_succ, Nat.add_assoc, Nat.add_comm p, ← Nat.add_assoc,
167                ← hq, sub_add_cancel h]⟩
168      | .isFalse h => ⟨0, by rw [mod_eq_of_lt (Nat.lt_of_not_le h), Nat.mul_zero, Nat.zero_add]⟩
169
170/-- Uniqueness of the decomposition: the quotients agree. -/
171⧉theorem quot_unique {p q q' r r' : Nat} (hr : r < p) (hr' : r' < p) (h : p * q + r = p * q' + r') :
172    q = q' := by
173  have key : ∀ {q q' r r' : Nat}, r < p → r' < p → p * q + r = p * q' + r' → ¬ q < q' := by
174    intro q q' r r' hr hr' h hlt
175    have h1 : p * (q + 1) ≤ p * q' := Nat.mul_le_mul_left p hlt
176    have h2 : p * q + r < p * q + p := Nat.add_lt_add_left hr _
177    rw [Nat.mul_succ] at h1
178    have h3 : p * q + r < p * q' + r' := Nat.lt_of_lt_of_le h2 (Nat.le_trans h1 (Nat.le_add_right _ _))
179    rw [h] at h3
180    exact Nat.lt_irrefl _ h3
181  exact match Nat.lt_or_ge q q' with
182    | .inl hlt => absurd hlt (key hr hr' h)
183    | .inr hge => match Nat.lt_or_ge q' q with
184      | .inl hlt => absurd hlt (key hr' hr h.symm)
185      | .inr hge' => Nat.le_antisymm hge' hge

187/-- The remainder is determined: if `x = p·q + r` with `r < p`, then `r = x % p`. -/
188⧉theorem mod_unique {p x q r : Nat} (hr : r < p) (h : x = p * q + r) : x % p = r := by
189  have hp : 0 < p := pos_of_lt hr
190  match mod_spec p hp x with
191  | ⟨q', hq'⟩ =>
192    have hqq : q' = q := quot_unique (Nat.mod_lt x hp) hr (by rw [← hq', ← h])
193    rw [hqq] at hq'
194    rw [h] at hq'
195    rw [h]
196    exact (add_left_cancel hq').symm
197
198⧉theorem mod_mod (x p : Nat) (hp : 0 < p) : x % p % p = x % p :=
199  mod_eq_of_lt (Nat.mod_lt x hp)

201⧉theorem add_mod (a b p : Nat) (hp : 0 < p) : (a + b) % p = (a % p + b % p) % p := by
202  match mod_spec p hp a, mod_spec p hp b, mod_spec p hp (a % p + b % p) with
203  | ⟨qa, ha⟩, ⟨qb, hb⟩, ⟨qc, hc⟩ =>
204    apply mod_unique (q := qa + qb + qc) (Nat.mod_lt _ hp)
205    rw [Nat.left_distrib, Nat.left_distrib, Nat.add_assoc, ← hc]
206    rw [Nat.add_assoc, Nat.add_left_comm (p * qb), ← Nat.add_assoc, ← ha, ← hb]

208⧉theorem add_mul_mod_self_left (x k p : Nat) (hp : 0 < p) : (p * k + x) % p = x % p := by
209  match mod_spec p hp x with
210  | ⟨q, hq⟩ =>
211    apply mod_unique (q := k + q) (Nat.mod_lt x hp)
212    rw [Nat.left_distrib, Nat.add_assoc, ← hq]

214⧉theorem mul_mod_left' (a b p : Nat) (hp : 0 < p) : (a * b) % p = (a % p * b) % p := by
215  match mod_spec p hp a with
216  | ⟨qa, ha⟩ =>
217    have e : a * b = p * (qa * b) + a % p * b := by
218      rw [← mul_assoc, ← add_mul, ← ha]
219    rw [e, add_mul_mod_self_left _ _ _ hp]

221⧉theorem mul_mod (a b p : Nat) (hp : 0 < p) : (a * b) % p = (a % p * (b % p)) % p := by
222  rw [mul_mod_left' a b p hp, Nat.mul_comm (a % p) b, mul_mod_left' b (a % p) p hp,
223    Nat.mul_comm (b % p) (a % p)]
