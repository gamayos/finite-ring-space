1/-!
2# FrcCore.Nat — the arithmetic the shells stand on, from first principles
3
4No Mathlib, no `simp`, no `omega`: every theorem here is checked by `#print axioms` to depend on no
5axiom at all. Lean's own core library proves most of these with `propext` for convenience (see
6`reports/lean-axioms-probe-20260917/core_probe.log`); they are re-derived here by induction from the
7axiom-free primitives (`Nat.rec`, `Nat.add_comm`, `Nat.mul_comm`, `Nat.left_distrib`, `Nat.mod_lt`,
8`Nat.le.dest`, `if_pos`, `if_neg`, `Decidable.em`).
9
10The division algorithm is obtained from the definition of `Nat.mod` in `Init.Prelude` (the wrapper
11around `Nat.modCore`, itself a fuel recursion): `mod_eq_of_lt`, `mod_eq_sub_mod`, `mod_spec`
12(`x = p·q + x % p`) and `mod_unique` (the remainder is determined by any such decomposition). Everything
13about residues mod `p` follows from those four.
14-/

20⧉theorem add_left_cancel {a b c : Nat} (h : a + b = a + c) : b = c := by
21  induction a with
22  | zero => rw [Nat.zero_add, Nat.zero_add] at h; exact h
23  | succ a ih =>
24    apply ih
25    rw [Nat.succ_add, Nat.succ_add] at h
26    exact Nat.succ.inj h

28⧉theorem add_right_cancel {a b c : Nat} (h : a + c = b + c) : a = b :=
29  add_left_cancel (a := c) (by rw [Nat.add_comm c a, Nat.add_comm c b]; exact h)

80/-- The fuel of `Nat.modCore.go` does not matter once it exceeds the dividend. -/
81⧉theorem modCore_go_fuel {y : Nat} (hy : 0 < y) : ∀ (f1 f2 x : Nat) (h1 : x < f1) (h2 : x < f2),
82    Nat.modCore.go y hy f1 x h1 = Nat.modCore.go y hy f2 x h2 := by
83  intro f1
84  induction f1 with
85  | zero => intro f2 x h1; exact absurd h1 (Nat.not_lt_zero x)
86  | succ f1 ih =>
87    intro f2 x h1 h2
88    cases f2 with
89    | zero => exact absurd h2 (Nat.not_lt_zero x)
90    | succ f2 =>
91      show (if h : y ≤ x then Nat.modCore.go y hy f1 (x - y) _ else x)
92          = (if h : y ≤ x then Nat.modCore.go y hy f2 (x - y) _ else x)
93      exact match Nat.decLe y x with
94        | .isTrue h => by rw [dif_pos h, dif_pos h]; exact ih f2 (x - y) _ _
95        | .isFalse h => by rw [dif_neg h, dif_neg h]
96
97⧉theorem modCore_eq' (x y : Nat) (hy : 0 < y) :
98    Nat.modCore x y = if y ≤ x then Nat.modCore (x - y) y else x := by
99  unfold Nat.modCore
100  rw [dif_pos hy]
101  show (if h : y ≤ x then Nat.modCore.go y hy x (x - y) _ else x) = _
102  exact match Nat.decLe y x with
103    | .isTrue h => by
104        rw [dif_pos h, if_pos h, dif_pos hy]
105        exact modCore_go_fuel hy x (x - y + 1) (x - y) _ _
106    | .isFalse h => by rw [dif_neg h, if_neg h]

108⧉theorem modCore_eq_mod' (n m : Nat) (hm : 0 < m) : Nat.modCore n m = n % m := by
109  cases n with
110  | zero =>
111    show Nat.modCore 0 m = 0
112    rw [modCore_eq' 0 m hm, if_neg (Nat.not_le_of_lt hm)]
113  | succ n =>
114    show Nat.modCore (n + 1) m = ite (m ≤ n + 1) (Nat.modCore (n + 1) m) (n + 1)
115    exact match Nat.decLe m (n + 1) with
116      | .isTrue h => by rw [if_pos h]
117      | .isFalse h => by rw [if_neg h, modCore_eq' _ _ hm, if_neg h]

119⧉theorem mod_eq_of_lt {x y : Nat} (h : x < y) : x % y = x := by
120  cases x with
121  | zero => rfl
122  | succ n =>
123    show ite (y ≤ n + 1) (Nat.modCore (n + 1) y) (n + 1) = n + 1
124    rw [if_neg (Nat.not_le_of_lt h)]

129/-- The division algorithm: `x = p·q + x % p` for some `q`. -/
130⧉theorem mod_spec (p : Nat) (hp : 0 < p) : ∀ x : Nat, ∃ q, x = p * q + x % p := by
131  intro x
132  induction x using Nat.strongRecOn with
133  | _ x ih =>
134    exact match Nat.decLe p x with
135      | .isTrue h =>
136          match ih (x - p) (Nat.sub_lt (Nat.lt_of_lt_of_le hp h) hp) with
137          | ⟨q, hq⟩ => ⟨q + 1, by
138              rw [mod_eq_sub_mod hp h, Nat.mul_succ, Nat.add_assoc, Nat.add_comm p, ← Nat.add_assoc,
139                ← hq, sub_add_cancel h]⟩
140      | .isFalse h => ⟨0, by rw [mod_eq_of_lt (Nat.lt_of_not_le h), Nat.mul_zero, Nat.zero_add]⟩

142/-- Uniqueness of the decomposition: the quotients agree. -/
143⧉theorem quot_unique {p q q' r r' : Nat} (hr : r < p) (hr' : r' < p) (h : p * q + r = p * q' + r') :
144    q = q' := by
145  have key : ∀ {q q' r r' : Nat}, r < p → r' < p → p * q + r = p * q' + r' → ¬ q < q' := by
146    intro q q' r r' hr hr' h hlt
147    have h1 : p * (q + 1) ≤ p * q' := Nat.mul_le_mul_left p hlt
148    have h2 : p * q + r < p * q + p := Nat.add_lt_add_left hr _
149    rw [Nat.mul_succ] at h1
150    have h3 : p * q + r < p * q' + r' := Nat.lt_of_lt_of_le h2 (Nat.le_trans h1 (Nat.le_add_right _ _))
151    rw [h] at h3
152    exact Nat.lt_irrefl _ h3
153  exact match Nat.lt_or_ge q q' with
154    | .inl hlt => absurd hlt (key hr hr' h)
155    | .inr hge => match Nat.lt_or_ge q' q with
156      | .inl hlt => absurd hlt (key hr' hr h.symm)
157      | .inr hge' => Nat.le_antisymm hge' hge

159/-- The remainder is determined: if `x = p·q + r` with `r < p`, then `r = x % p`. -/
160⧉theorem mod_unique {p x q r : Nat} (hr : r < p) (h : x = p * q + r) : x % p = r := by
161  have hp : 0 < p := pos_of_lt hr
162  match mod_spec p hp x with
163  | ⟨q', hq'⟩ =>
164    have hqq : q' = q := quot_unique (Nat.mod_lt x hp) hr (by rw [← hq', ← h])
165    rw [hqq] at hq'
166    rw [h] at hq'
167    rw [h]
168    exact (add_left_cancel hq').symm

173⧉theorem add_mod (a b p : Nat) (hp : 0 < p) : (a + b) % p = (a % p + b % p) % p := by
174  match mod_spec p hp a, mod_spec p hp b, mod_spec p hp (a % p + b % p) with
175  | ⟨qa, ha⟩, ⟨qb, hb⟩, ⟨qc, hc⟩ =>
176    apply mod_unique (q := qa + qb + qc) (Nat.mod_lt _ hp)
177    rw [Nat.left_distrib, Nat.left_distrib, Nat.add_assoc, ← hc]
178    rw [Nat.add_assoc, Nat.add_left_comm (p * qb), ← Nat.add_assoc, ← ha, ← hb]

180⧉theorem add_mul_mod_self_left (x k p : Nat) (hp : 0 < p) : (p * k + x) % p = x % p := by

186⧉theorem mul_mod_left' (a b p : Nat) (hp : 0 < p) : (a * b) % p = (a % p * b) % p := by
187  match mod_spec p hp a with
188  | ⟨qa, ha⟩ =>
189    have e : a * b = p * (qa * b) + a % p * b := by
190      rw [← mul_assoc, ← add_mul, ← ha]
191    rw [e, add_mul_mod_self_left _ _ _ hp]

193⧉theorem mul_mod (a b p : Nat) (hp : 0 < p) : (a * b) % p = (a % p * (b % p)) % p := by
194  rw [mul_mod_left' a b p hp, Nat.mul_comm (a % p) b, mul_mod_left' b (a % p) p hp,
