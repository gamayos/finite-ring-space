3/-!
4# FrcCore.Quaternion — the framed quaternions and the window (1:G5, the finitary clause)
5
6A signed window integer is a pair of naturals `(u, v)` read as `u − v` in the shell (no `Int`: the core's
7integers are the shell's own readings).  A framed quaternion is four such pairs; the Hamilton product is
8computed on the pairs, and `read_mul` says the product of the readings is the reading of the product — the
9composition is exact.  `quaternion_window` adds the bounds: with coordinates in the window `H` and a shell
10`p > 8H²`, the product's coordinates lie in the window `4H²`, and the shell reading determines the integer
11product among the window's quaternions.  No axioms.
12-/

20/-- A signed window integer: `(u, v)` stands for `u − v`. -/
21⧉abbrev SInt := Nat × Nat

23⧉def sread (x : SInt) : Shell p := ofNat x.1 + -(ofNat x.2)
24⧉def sadd (x y : SInt) : SInt := (x.1 + y.1, x.2 + y.2)
25⧉def sneg (x : SInt) : SInt := (x.2, x.1)
26⧉def smul (x y : SInt) : SInt := (x.1 * y.1 + x.2 * y.2, x.1 * y.2 + x.2 * y.1)
27/-- The canonical pair of the same value: one component zero. -/
28⧉def snorm (x : SInt) : SInt := (x.1 - x.2, x.2 - x.1)

30⧉theorem prod_ext {x y : SInt} (h1 : x.1 = y.1) (h2 : x.2 = y.2) : x = y :=
31  match x, y, h1, h2 with
32  | (_, _), (_, _), h1, h2 => by cases h1; cases h2; rfl

34⧉theorem sread_add (x y : SInt) : (sread (sadd x y) : Shell p) = sread x + sread y := by
35  unfold sread sadd
36  show ofNat (x.1 + y.1) + -(ofNat (x.2 + y.2)) = ofNat x.1 + -(ofNat x.2) + (ofNat y.1 + -(ofNat y.2))
37  rw [← ofNat_add, ← ofNat_add, neg_add_rev, add_assoc, add_assoc, ← add_assoc (-(ofNat x.2)),
38    add_comm (-(ofNat x.2)) (ofNat y.1), add_assoc]

40⧉theorem sread_neg (x : SInt) : (sread (sneg x) : Shell p) = -(sread x) := by
41  unfold sread sneg
42  show ofNat x.2 + -(ofNat x.1) = -(ofNat x.1 + -(ofNat x.2))
43  rw [neg_add_rev, neg_neg, add_comm]

45⧉theorem sread_mul (x y : SInt) : (sread (smul x y) : Shell p) = sread x * sread y := by
46  unfold sread smul
47  show ofNat (x.1 * y.1 + x.2 * y.2) + -(ofNat (x.1 * y.2 + x.2 * y.1)) =
48    (ofNat x.1 + -(ofNat x.2)) * (ofNat y.1 + -(ofNat y.2))
49  rw [right_distrib, left_distrib, left_distrib, ← mul_neg, ← neg_mul, neg_mul_neg, ← ofNat_add, ← ofNat_add,
50    ← ofNat_mul, ← ofNat_mul, ← ofNat_mul, ← ofNat_mul, neg_add_rev]
51  -- a + d + (-b + -c) = a + -b + (-c + d)
52  rw [add_assoc, add_assoc]
53  refine congrArg (ofNat x.1 * ofNat y.1 + ·) ?_
54  rw [← add_assoc, add_comm (ofNat x.2 * ofNat y.2), add_assoc, add_comm (ofNat x.2 * ofNat y.2)]

56⧉theorem sread_norm (x : SInt) : (sread (snorm x) : Shell p) = sread x := by
57  unfold sread snorm
58  match Nat.decLe x.2 x.1 with
59  | .isTrue h =>
60    have e : x.2 - x.1 = 0 := FRC.Nat.sub_eq_zero_of_le h
61    show ofNat (x.1 - x.2) + -(ofNat (x.2 - x.1)) = ofNat x.1 + -(ofNat x.2)
62    rw [e]
63    have e0 : (ofNat 0 : Shell p) = 0 := rfl
64    rw [e0, neg_zero, add_zero]
65    have e1 : x.1 = (x.1 - x.2) + x.2 := (FRC.Nat.sub_add_cancel h).symm
66    calc ofNat (x.1 - x.2) = ofNat (x.1 - x.2) + 0 := (add_zero _).symm
67      _ = ofNat (x.1 - x.2) + (ofNat x.2 + -(ofNat x.2)) := by rw [add_neg]
68      _ = ofNat (x.1 - x.2 + x.2) + -(ofNat x.2) := by rw [← add_assoc, ofNat_add]
69      _ = ofNat x.1 + -(ofNat x.2) := by rw [← e1]
70  | .isFalse h =>
71    have h' : x.1 ≤ x.2 := Nat.le_of_lt (Nat.lt_of_not_le h)
72    have e : x.1 - x.2 = 0 := FRC.Nat.sub_eq_zero_of_le h'
73    show ofNat (x.1 - x.2) + -(ofNat (x.2 - x.1)) = ofNat x.1 + -(ofNat x.2)
74    rw [e]
75    have e0 : (ofNat 0 : Shell p) = 0 := rfl
76    rw [e0, zero_add]
77    have e1 : x.2 = (x.2 - x.1) + x.1 := (FRC.Nat.sub_add_cancel h').symm
78    calc (-(ofNat (x.2 - x.1)) : Shell p) = (0 : Shell p) + -(ofNat (x.2 - x.1)) := (zero_add _).symm
79      _ = (ofNat x.1 + -(ofNat x.1)) + -(ofNat (x.2 - x.1)) := by rw [add_neg]
80      _ = ofNat x.1 + -((ofNat (x.2 - x.1) : Shell p) + ofNat x.1) := by
81          rw [add_assoc, neg_add_rev, add_comm (-(ofNat x.1))]
82      _ = ofNat x.1 + -(ofNat x.2) := by rw [ofNat_add, ← e1]

84/-- The window bound: both components at most `H`, one of them zero. -/
85⧉def SBound (H : Nat) (x : SInt) : Prop := x.1 ≤ H ∧ x.2 ≤ H ∧ (x.1 = 0 ∨ x.2 = 0)

87⧉theorem sneg_bound {H : Nat} {x : SInt} (h : SBound H x) : SBound H (sneg x) :=

90⧉theorem smul_bound {H : Nat} {x y : SInt} (hx : SBound H x) (hy : SBound H y) : SBound (H * H) (smul x y) := by
91  unfold smul
92  have b11 := Nat.mul_le_mul hx.1 hy.1
93  have b22 := Nat.mul_le_mul hx.2.1 hy.2.1
94  have b12 := Nat.mul_le_mul hx.1 hy.2.1
95  have b21 := Nat.mul_le_mul hx.2.1 hy.1
96  match hx.2.2, hy.2.2 with
97  | Or.inl e, Or.inl f =>
98    refine ⟨?_, ?_, Or.inr ?_⟩
99    · show x.1 * y.1 + x.2 * y.2 ≤ H * H
100      rw [e, Nat.zero_mul, Nat.zero_add]; exact b22
101    · show x.1 * y.2 + x.2 * y.1 ≤ H * H
102      rw [e, f, Nat.zero_mul, Nat.mul_zero]; exact Nat.zero_le _
103    · show x.1 * y.2 + x.2 * y.1 = 0
104      rw [e, f, Nat.zero_mul, Nat.mul_zero]
105  | Or.inl e, Or.inr f =>
106    refine ⟨?_, ?_, Or.inl ?_⟩
107    · show x.1 * y.1 + x.2 * y.2 ≤ H * H
108      rw [e, f, Nat.zero_mul, Nat.mul_zero]; exact Nat.zero_le _
109    · show x.1 * y.2 + x.2 * y.1 ≤ H * H
110      rw [e, Nat.zero_mul, Nat.zero_add]; exact b21
111    · show x.1 * y.1 + x.2 * y.2 = 0
112      rw [e, f, Nat.zero_mul, Nat.mul_zero]
113  | Or.inr e, Or.inl f =>
114    refine ⟨?_, ?_, Or.inl ?_⟩
115    · show x.1 * y.1 + x.2 * y.2 ≤ H * H
116      rw [e, f, Nat.mul_zero, Nat.zero_mul]; exact Nat.zero_le _
117    · show x.1 * y.2 + x.2 * y.1 ≤ H * H
118      rw [e, Nat.zero_mul, Nat.add_zero]; exact b12
119    · show x.1 * y.1 + x.2 * y.2 = 0
120      rw [e, f, Nat.mul_zero, Nat.zero_mul]
121  | Or.inr e, Or.inr f =>
122    refine ⟨?_, ?_, Or.inr ?_⟩
123    · show x.1 * y.1 + x.2 * y.2 ≤ H * H
124      rw [e, Nat.zero_mul, Nat.add_zero]; exact b11
125    · show x.1 * y.2 + x.2 * y.1 ≤ H * H
126      rw [e, f, Nat.mul_zero, Nat.zero_mul]; exact Nat.zero_le _
127    · show x.1 * y.2 + x.2 * y.1 = 0
128      rw [e, f, Nat.mul_zero, Nat.zero_mul]

130/-- Four window terms of size `M` add to a pair with both components at most `4M`. -/
131⧉theorem sum4_bound {M : Nat} {t1 t2 t3 t4 : SInt} (h1 : SBound M t1) (h2 : SBound M t2) (h3 : SBound M t3)
132    (h4 : SBound M t4) :
133    (sadd (sadd t1 t2) (sadd t3 t4)).1 ≤ 4 * M ∧ (sadd (sadd t1 t2) (sadd t3 t4)).2 ≤ 4 * M := by
134  unfold sadd
135  rw [four_mul_eq, Nat.add_assoc (M + M) M M]
136  exact ⟨Nat.add_le_add (Nat.add_le_add h1.1 h2.1) (Nat.add_le_add h3.1 h4.1),
137    Nat.add_le_add (Nat.add_le_add h1.2.1 h2.2.1) (Nat.add_le_add h3.2.1 h4.2.1)⟩

139⧉theorem snorm_bound {M : Nat} {x : SInt} (h1 : x.1 ≤ M) (h2 : x.2 ≤ M) : SBound M (snorm x) := by
140  unfold snorm
141  refine ⟨Nat.le_trans (Nat.sub_le _ _) h1, Nat.le_trans (Nat.sub_le _ _) h2, ?_⟩
142  match Nat.decLe x.2 x.1 with
143  | .isTrue h => exact Or.inr (FRC.Nat.sub_eq_zero_of_le h)
144  | .isFalse h => exact Or.inl (FRC.Nat.sub_eq_zero_of_le (Nat.le_of_lt (Nat.lt_of_not_le h)))

146/-- The signed window law: two canonical pairs of size `M`, `2M < p`, with the same reading are equal. -/
147⧉theorem sread_inj {M : Nat} (hM : 2 * M < p) {x y : SInt} (hx : SBound M x) (hy : SBound M y)
148    (h : (sread x : Shell p) = sread y) : x = y := by
149  unfold sread at h
150  have h' : (ofNat (x.1 + y.2) : Shell p) = ofNat (y.1 + x.2) := by
151    rw [← ofNat_add, ← ofNat_add]
152    calc ofNat x.1 + ofNat y.2 = ofNat x.1 + -(ofNat x.2) + (ofNat x.2 + ofNat y.2) := by
153          rw [add_assoc, ← add_assoc (-(ofNat x.2)), neg_add, zero_add]
154      _ = ofNat y.1 + -(ofNat y.2) + (ofNat x.2 + ofNat y.2) := by rw [h]
155      _ = ofNat y.1 + ofNat x.2 := by
156          rw [add_assoc, add_comm (ofNat x.2), ← add_assoc (-(ofNat y.2)), neg_add, zero_add]
157  have hv := val_injective h'
158  rw [val_ofNat, val_ofNat] at hv
159  have l1 : x.1 + y.2 < p := Nat.lt_of_le_of_lt (Nat.add_le_add hx.1 hy.2.1) (Nat.two_mul M ▸ hM)
160  have l2 : y.1 + x.2 < p := Nat.lt_of_le_of_lt (Nat.add_le_add hy.1 hx.2.1) (Nat.two_mul M ▸ hM)
161  rw [FRC.Nat.mod_eq_of_lt l1, FRC.Nat.mod_eq_of_lt l2] at hv
162  -- hv : x.1 + y.2 = y.1 + x.2
163  match hx.2.2, hy.2.2 with
164  | Or.inl e, Or.inl f =>
165    rw [e, f, Nat.zero_add, Nat.zero_add] at hv
166    exact prod_ext (e.trans f.symm) hv.symm
167  | Or.inl e, Or.inr f =>
168    rw [e, f, Nat.zero_add] at hv
169    -- hv : 0 = y.1 + x.2
170    have hy1 : y.1 = 0 := Nat.eq_zero_of_add_eq_zero_right hv.symm
171    have hx2 : x.2 = 0 := Nat.eq_zero_of_add_eq_zero_left hv.symm
172    exact prod_ext (e.trans hy1.symm) (hx2.trans f.symm)
173  | Or.inr e, Or.inl f =>
174    rw [e, f, Nat.zero_add] at hv
175    -- hv : x.1 + y.2 = 0
176    have hx1 : x.1 = 0 := Nat.eq_zero_of_add_eq_zero_right hv
177    have hy2 : y.2 = 0 := Nat.eq_zero_of_add_eq_zero_left hv
178    exact prod_ext (hx1.trans f.symm) (e.trans hy2.symm)
179  | Or.inr e, Or.inr f =>
180    rw [e, f, Nat.add_zero, Nat.add_zero] at hv
181    exact prod_ext hv (e.trans f.symm)

190/-- A quaternion over the shell. -/
191⧉structure Quat (p : Nat) [Pos p] where

197/-- The Hamilton product on the shell. -/
198⧉def Quat.mul (q r : Quat p) : Quat p :=
199  ⟨(q.a * r.a + -(q.b * r.b)) + (-(q.c * r.c) + -(q.d * r.d)),
200   (q.a * r.b + q.b * r.a) + (q.c * r.d + -(q.d * r.c)),
201   (q.a * r.c + -(q.b * r.d)) + (q.c * r.a + q.d * r.b),
202   (q.a * r.d + q.b * r.c) + (-(q.c * r.b) + q.d * r.a)⟩
203
204/-- The Hamilton product on the window pairs, the same formula. -/
205⧉def IQuat.mul (q r : IQuat) : IQuat :=
206  ⟨sadd (sadd (smul q.a r.a) (sneg (smul q.b r.b))) (sadd (sneg (smul q.c r.c)) (sneg (smul q.d r.d))),
207   sadd (sadd (smul q.a r.b) (smul q.b r.a)) (sadd (smul q.c r.d) (sneg (smul q.d r.c))),
208   sadd (sadd (smul q.a r.c) (sneg (smul q.b r.d))) (sadd (smul q.c r.a) (smul q.d r.b)),
209   sadd (sadd (smul q.a r.d) (smul q.b r.c)) (sadd (sneg (smul q.c r.b)) (smul q.d r.a))⟩

211⧉def IQuat.read (q : IQuat) : Quat p := ⟨sread q.a, sread q.b, sread q.c, sread q.d⟩
212⧉def IQuat.norm (q : IQuat) : IQuat := ⟨snorm q.a, snorm q.b, snorm q.c, snorm q.d⟩
213⧉def QBound (H : Nat) (q : IQuat) : Prop := SBound H q.a ∧ SBound H q.b ∧ SBound H q.c ∧ SBound H q.d

230⧉theorem norm_bound {H : Nat} {q r : IQuat} (hq : QBound H q) (hr : QBound H r) :
231    QBound (4 * (H * H)) (IQuat.mul q r).norm := by
232  have m := fun {x y : SInt} (hx : SBound H x) (hy : SBound H y) => smul_bound hx hy
233  refine ⟨?_, ?_, ?_, ?_⟩
234  · have := sum4_bound (m hq.1 hr.1) (sneg_bound (m hq.2.1 hr.2.1)) (sneg_bound (m hq.2.2.1 hr.2.2.1))
235      (sneg_bound (m hq.2.2.2 hr.2.2.2))
236    exact snorm_bound this.1 this.2
237  · have := sum4_bound (m hq.1 hr.2.1) (m hq.2.1 hr.1) (m hq.2.2.1 hr.2.2.2) (sneg_bound (m hq.2.2.2 hr.2.2.1))
238    exact snorm_bound this.1 this.2

240    exact snorm_bound this.1 this.2
241  · have := sum4_bound (m hq.1 hr.2.2.2) (m hq.2.1 hr.2.2.1) (sneg_bound (m hq.2.2.1 hr.2.1)) (m hq.2.2.2 hr.1)

244/-- 1:G5, the finitary clause — framed quaternions with window coordinates `≤ H`, read in a shell `p > 8H²`,
245compose exactly: the product of the readings is the reading of the integer product; that product's
246coordinates lie in the window `4H²`; and the reading determines it among the window's quaternions. -/
247⧉theorem quaternion_window {H : Nat} (hH : 8 * (H * H) < p) {q r : IQuat} (hq : QBound H q) (hr : QBound H r) :
248    ((IQuat.mul q r).read : Quat p) = Quat.mul q.read r.read ∧
249    QBound (4 * (H * H)) (IQuat.mul q r).norm ∧
250    ∀ s : IQuat, QBound (4 * (H * H)) s → (s.read : Quat p) = (IQuat.mul q r).read → s = (IQuat.mul q r).norm := by
251  refine ⟨read_mul q r, norm_bound hq hr, ?_⟩
252  intro s hs he
253  have hn := norm_bound hq hr
254  have hM : 2 * (4 * (H * H)) < p := by rw [← Nat.mul_assoc]; exact hH
255  rw [← read_norm (IQuat.mul q r)] at he
256  unfold IQuat.read at he
257  exact IQuat.ext' (sread_inj hM hs.1 hn.1 (congrArg Quat.a he)) (sread_inj hM hs.2.1 hn.2.1 (congrArg Quat.b he))
258    (sread_inj hM hs.2.2.1 hn.2.2.1 (congrArg Quat.c he)) (sread_inj hM hs.2.2.2 hn.2.2.2 (congrArg Quat.d he))
