4/-!
5# FrcCore.Frame — the frame `(τ; 0, 1, g)` and the Euclidean datum, from first principles
6
7The shell of capacity `κ` has modulus `p = 4κ + 1`; its frame carries the drive `g` (00:A8, 00:C1).
8One decidable predicate states what the frame's generator is: `IsPrimitive g n` (`g^n = 1`, no positive
9power below `n` is `1`). That `g` then *generates* — every nonzero residue is a power of `g`
10(`Generates g n`, also decidable) — is proved by the pigeonhole (`FrcCore.Pigeonhole`): the `n` powers are
11distinct nonzero residues and there are `n` of those. Primality of `p` is neither assumed nor used; the
12classical equivalence ("a primitive root of order `p − 1` exists iff `p` is prime") is a theorem for later.
13
14From primitivity alone: inverses (`exists_inv`), no zero divisors (`mul_eq_zero`), the square roots
15of one (`sq_eq_one`), the half-period `g^{2κ} = −1` (2:D1, 00:C1), the quarter-turn `i = −g^κ` with
16`i² = −1` (1:B3, 2:D2), its orientation classes under `g ↦ g^u` (2:D5), and the Euler identity
17`(g^i)^{i·2κ} = (−1)^i` (2:D6, 00:C14). No axioms.
18-/

25/-- 00:A8 — the drive generator is primitive of order `n`: `g^n = 1` and `g^l ≠ 1` for `0 < l < n`. -/
26def IsPrimitive (g : Shell p) (n : Nat) : Prop :=
27  g ^ n = 1 ∧ ∀ l, l < n → 0 < l → g ^ l ≠ 1
28
29instance (g : Shell p) (n : Nat) : Decidable (IsPrimitive g n) := by
30  unfold IsPrimitive; exact inferInstance
31
32/-- 00:A8 — the drive generates the shell: every nonzero residue `v < p` is `g^m` for some `m < n`. -/
33def Generates (g : Shell p) (n : Nat) : Prop :=
34  ∀ v, v < p → 0 < v → ∃ m, m < n ∧ (g ^ m).val = v
35
36/-- Bounded existence `∃ m < n, P m`, decided by search — Lean's own `Nat.decidableExistsLT` carries
37`propext` and `Quot.sound`; this one carries nothing. -/
38def decExistsLT (P : Nat → Prop) [DecidablePred P] : (n : Nat) → Decidable (∃ m, m < n ∧ P m)
39  | 0 => isFalse (fun ⟨m, hm, _⟩ => Nat.not_lt_zero m hm)
40  | n + 1 =>
41    match decExistsLT P n with
42    | isTrue h => isTrue (match h with | ⟨m, hm, hp⟩ => ⟨m, Nat.lt_succ_of_lt hm, hp⟩)
43    | isFalse hno =>
44      if h : P n then isTrue ⟨n, Nat.lt_succ_self n, h⟩
45      else isFalse (fun ⟨m, hm, hp⟩ =>
46        match Nat.lt_or_ge m n with
47        | .inl hlt => hno ⟨m, hlt, hp⟩
48        | .inr hge => h ((Nat.le_antisymm (Nat.le_of_lt_succ hm) hge) ▸ hp))
49
50instance (g : Shell p) (n : Nat) : Decidable (Generates g n) := by
51  unfold Generates
52  have : ∀ v, Decidable (∃ m, m < n ∧ (g ^ m).val = v) := fun v => decExistsLT (fun m => (g ^ m).val = v) n
53  exact inferInstance

55/-- 00:C1 — the frame `(τ; 0, 1, g)` on the shell of capacity `κ`: `p = 4κ + 1`, `κ` positive, and the
56drive `g` a primitive generator of the `p − 1` nonzero residues. -/
57structure Frame (p : Nat) [Pos p] (κ : Nat) (g : Shell p) : Prop where
58  cap : p = 4 * κ + 1
59  cap_pos : 0 < κ
60  prim : IsPrimitive g (p - 1)

80/-- Powers of the drive are periodic with period `p − 1`. -/
81theorem pow_mod (F : Frame p κ g) (l : Nat) : g ^ l = g ^ (l % (p - 1)) := by
82  match FRC.Nat.mod_spec (p - 1) F.n_pos l with
83  | ⟨q, hq⟩ =>
84    calc g ^ l = g ^ ((p - 1) * q + l % (p - 1)) := by rw [← hq]
85      _ = (g ^ (p - 1)) ^ q * g ^ (l % (p - 1)) := by rw [pow_add, pow_mul]
86      _ = g ^ (l % (p - 1)) := by rw [F.pow_n, one_pow, one_mul]

88theorem pow_eq_one_of_mod (F : Frame p κ g) {l : Nat} (h : l % (p - 1) = 0) : g ^ l = 1 := by
89  rw [F.pow_mod, h, pow_zero]

91theorem mod_eq_zero_of_pow_eq_one (F : Frame p κ g) {l : Nat} (h : g ^ l = 1) : l % (p - 1) = 0 := by

97theorem pow_inj (F : Frame p κ g) {i j : Nat} (hi : i < p - 1) (hj : j < p - 1) (h : g ^ i = g ^ j) :
98    i = j := by
99  have key : ∀ {i j : Nat}, i ≤ j → j < p - 1 → g ^ i = g ^ j → i = j := by
100    intro i j hij hj h
101    have hji : j - i < p - 1 := Nat.lt_of_le_of_lt (Nat.sub_le j i) hj
102    have e : g ^ j = g ^ i * g ^ (j - i) := by rw [← pow_add, FRC.Nat.add_sub_of_le hij]
103    have hinv : g ^ i * g ^ (p - 1 - i) = 1 := by
104      rw [← pow_add, FRC.Nat.add_sub_of_le (Nat.le_of_lt (Nat.lt_of_le_of_lt hij hj)), F.pow_n]
105    have h1 : g ^ (j - i) = 1 := by
106      calc g ^ (j - i) = 1 * g ^ (j - i) := (one_mul _).symm
107        _ = g ^ (p - 1 - i) * g ^ i * g ^ (j - i) := by rw [mul_comm (g ^ (p - 1 - i)), hinv]
108        _ = g ^ (p - 1 - i) * g ^ j := by rw [mul_assoc, ← e]
109        _ = g ^ (p - 1 - i) * g ^ i := by rw [h]
110        _ = 1 := by rw [mul_comm, hinv]
111    have h2 := F.mod_eq_zero_of_pow_eq_one h1
112    rw [FRC.Nat.mod_eq_of_lt hji] at h2
113    have : j = i + (j - i) := (FRC.Nat.add_sub_of_le hij).symm
114    rw [h2, Nat.add_zero] at this
115    exact this.symm
116  exact match Nat.lt_or_ge i j with
117    | .inl hlt => key (Nat.le_of_lt hlt) hj h
118    | .inr hge => (key hge hi h.symm).symm

129/-- No power of the drive is zero: `g^m · g^{(n−1)m} = g^{nm} = 1`. -/
130theorem pow_ne_zero (F : Frame p κ g) (m : Nat) : g ^ m ≠ 0 := fun h0 => by
131  have hn := F.n_pos
132  have e : m + (p - 1 - 1) * m = (p - 1) * m := by
133    calc m + (p - 1 - 1) * m = 1 * m + (p - 1 - 1) * m := by rw [Nat.one_mul]
134      _ = (1 + (p - 1 - 1)) * m := (FRC.Nat.add_mul _ _ _).symm
135      _ = (p - 1) * m := by rw [FRC.Nat.add_sub_of_le hn]
136  have : g ^ m * g ^ ((p - 1 - 1) * m) = 1 := by
137    rw [← pow_add, e, pow_mul, F.pow_n, one_pow]
138  rw [h0, zero_mul] at this
139  exact F.one_ne_zero this.symm

141/-- The representatives of `g^0, …, g^{n−1}`, as a list (latest first). -/
142def powList (g : Shell p) : Nat → List Nat
143  | 0 => []
144  | m + 1 => (g ^ m).val :: powList g m

146theorem powList_length (g : Shell p) (n : Nat) : (powList g n).length = n := by
147  induction n with
148  | zero => rfl
149  | succ n ih => show (powList g n).length + 1 = n + 1; rw [ih]

151theorem mem_powList {g : Shell p} {v : Nat} : ∀ {n : Nat}, Pigeonhole.mem v (powList g n) → ∃ m, m < n ∧ (g ^ m).val = v
152  | 0, h => absurd h id
153  | n + 1, h => match h with
154    | Or.inl e => ⟨n, Nat.lt_succ_self n, e.symm⟩
155    | Or.inr h' => match mem_powList h' with
156      | ⟨m, hm, e⟩ => ⟨m, Nat.lt_succ_of_lt hm, e⟩
157
158theorem powList_nodup (F : Frame p κ g) : ∀ {n : Nat}, n ≤ p - 1 → Pigeonhole.NoDup (powList g n)
159  | 0, _ => trivial
160  | n + 1, hn => ⟨fun h => match mem_powList h with
161      | ⟨m, hm, e⟩ =>
162        have : m = n := F.pow_inj (Nat.lt_trans hm hn) hn (ext e)
163        Nat.lt_irrefl n (this ▸ hm),
164    powList_nodup F (Nat.le_of_lt hn)⟩

166/-- 00:A8 — the drive generates: every nonzero residue is a power `g^m`, `m < p − 1` (the pigeonhole). -/
167theorem generates (F : Frame p κ g) : Generates g (p - 1) := by
168  intro v hv hv0
169  have hb : ∀ e, Pigeonhole.mem e (powList g (p - 1)) → 1 ≤ e ∧ e ≤ p - 1 := fun e he =>
170    match mem_powList he with
171    | ⟨m, _, hm⟩ =>
172      ⟨Nat.pos_of_ne_zero (fun h0 => F.pow_ne_zero m (ext (by rw [hm, h0]; rfl))),
173       by rw [← hm]; exact Nat.le_of_lt_succ (Nat.lt_of_lt_of_le (g ^ m).lt (Nat.le_of_eq (FRC.Nat.sub_add_cancel Pos.pos).symm))⟩
174  exact mem_powList (Pigeonhole.mem_of_nodup_of_length (p - 1) (powList g (p - 1)) (F.powList_nodup (Nat.le_refl _))
175    hb (powList_length g (p - 1)) v hv0 (Nat.le_of_lt_succ (Nat.lt_of_lt_of_le hv (Nat.le_of_eq (FRC.Nat.sub_add_cancel Pos.pos).symm))))

177/-- Every nonzero residue is a power of the drive, on residues. -/
178theorem eq_pow_of_ne_zero (F : Frame p κ g) {x : Shell p} (hx : x ≠ 0) :
179    ∃ m, m < p - 1 ∧ g ^ m = x := by

184theorem exists_inv (F : Frame p κ g) {x : Shell p} (hx : x ≠ 0) : ∃ y, x * y = 1 := by

187    refine ⟨g ^ (p - 1 - m), ?_⟩
188    rw [← e, ← pow_add, FRC.Nat.add_sub_of_le (Nat.le_of_lt hm), F.pow_n]

190theorem mul_eq_zero (F : Frame p κ g) {a b : Shell p} (h : a * b = 0) : a = 0 ∨ b = 0 :=
191  match Shell.instDecidableEq a 0 with
192  | .isTrue ha => .inl ha
193  | .isFalse ha => .inr (by
194      match F.exists_inv ha with
195      | ⟨y, hy⟩ =>
196        calc b = 1 * b := (one_mul b).symm
197          _ = y * a * b := by rw [mul_comm y a, hy]
198          _ = y * (a * b) := mul_assoc _ _ _
199          _ = 0 := by rw [h, mul_zero])

201theorem mul_ne_zero (F : Frame p κ g) {a b : Shell p} (ha : a ≠ 0) (hb : b ≠ 0) : a * b ≠ 0 :=
202  fun h => match F.mul_eq_zero h with
203    | .inl e => ha e
204    | .inr e => hb e

206/-- Cancellation: `a * b = a * c` with `a ≠ 0` gives `b = c`. -/
207theorem mul_left_cancel (F : Frame p κ g) {a b c : Shell p} (ha : a ≠ 0) (h : a * b = a * c) : b = c := by
208  have : a * (b + -c) = 0 := by rw [left_distrib, ← mul_neg, h, add_neg]
209  match F.mul_eq_zero this with
210  | .inl e => exact absurd e ha
211  | .inr e =>
212    calc b = b + 0 := (add_zero b).symm
213      _ = b + (-c + c) := by rw [neg_add]
214      _ = (b + -c) + c := (add_assoc _ _ _).symm
215      _ = c := by rw [e, zero_add]

217/-- The square roots of one are `±1`. -/
218theorem sq_eq_one (F : Frame p κ g) {x : Shell p} (h : x * x = 1) : x = 1 ∨ x = -1 := by
219  have e : (x + -1) * (x + 1) = 0 := by
220    rw [right_distrib, left_distrib, left_distrib, h, mul_one, neg_one_mul, neg_one_mul]
221    rw [add_assoc, ← add_assoc x (-x), add_neg, zero_add, add_neg]
222  match F.mul_eq_zero e with
223  | .inl e1 => exact .inl (by
224      calc x = x + 0 := (add_zero x).symm
225        _ = x + (-1 + 1) := by rw [neg_add]
226        _ = (x + -1) + 1 := (add_assoc _ _ _).symm
227        _ = 1 := by rw [e1, zero_add])
228  | .inr e2 => exact .inr (eq_neg_of_add_eq_zero e2)

238theorem two_kappa_lt (F : Frame p κ g) : 2 * κ < p - 1 := by

248/-- 2:D1, 13:H2, 00:C1 — the half-period: `g^{2κ} = −1` for the drive of every frame (the half-turn tautology
249behind `χ(−1) = e^{iπ}`). -/
250theorem half_period (F : Frame p κ g) : g ^ (2 * κ) = -1 := by
251  have hsq : g ^ (2 * κ) * g ^ (2 * κ) = 1 := by rw [← pow_add, F.four_kappa, F.pow_n]

253  | .inr e => exact e

261/-- 2:D2 — `g^κ` has order four: `(g^κ)^2 = −1` and `(g^κ)^4 = 1`. -/
262theorem quarter_turn_order (F : Frame p κ g) : (g ^ κ) ^ 2 = -1 ∧ (g ^ κ) ^ 4 = 1 := by
263  have h2 : (g ^ κ) ^ 2 = -1 := by rw [← pow_mul, Nat.mul_comm, F.half_period]
264  refine ⟨h2, ?_⟩
265  show (g ^ κ) ^ (2 * 2) = 1
266  rw [pow_mul, h2, neg_pow_two, one_pow]

268/-- 2:D5, 6:B3, 13:B2 — the orientation classes: for `g' = g^u`, the quarter-turn `−g'^κ` is `−g^κ` when
269`u ≡ 1 (mod 4)` and `−(−g^κ)` when `u ≡ 3 (mod 4)`. -/
270theorem orientation_class (F : Frame p κ g) (u : Nat) :
271    (u % 4 = 1 → -((g ^ u) ^ κ) = -(g ^ κ)) ∧ (u % 4 = 3 → -((g ^ u) ^ κ) = -(-(g ^ κ))) := by
272  have h4 : (g ^ κ) ^ 4 = 1 := (F.quarter_turn_order).2
273  have h2 : (g ^ κ) ^ 2 = -1 := (F.quarter_turn_order).1
274  have key : (g ^ u) ^ κ = (g ^ κ) ^ u := pow_mul_comm g u κ
275  have hu : (g ^ κ) ^ u = (g ^ κ) ^ (u % 4) := by
276    match FRC.Nat.mod_spec 4 (Nat.zero_lt_succ 3) u with
277    | ⟨q, hq⟩ =>
278      calc (g ^ κ) ^ u = (g ^ κ) ^ (4 * q + u % 4) := by rw [← hq]
279        _ = ((g ^ κ) ^ 4) ^ q * (g ^ κ) ^ (u % 4) := by rw [pow_add, pow_mul]
280        _ = (g ^ κ) ^ (u % 4) := by rw [h4, one_pow, one_mul]
281  constructor
282  · intro h1; rw [key, hu, h1, pow_one]
283  · intro h3
284    rw [key, hu, h3]
285    show -((g ^ κ) ^ (2 + 1)) = -(-(g ^ κ))
286    rw [pow_add, h2, pow_one, neg_one_mul]

288theorem sq_mod_two (i : Nat) : (i * i) % 2 = i % 2 := by
289  rw [FRC.Nat.mul_mod i i 2 (Nat.zero_lt_succ 1)]
290  have := Nat.mod_lt i (Nat.zero_lt_succ 1)
291  match i % 2, this with
292  | 0, _ => rfl
293  | 1, _ => rfl
294  | k + 2, hk => exact absurd hk (Nat.not_lt_of_le (Nat.le_add_left 2 k))

296/-- 2:D6, 6:B2, 00:C14 — the Euler identity on the shell: with `e = g^i` and `π = 2κ`,
297`(g^i)^{i·2κ} = (−1)^i` for every natural reading `i` of the quarter-turn — `−1` exactly when `i` is odd. -/
298theorem euler_identity (F : Frame p κ g) (i : Nat) :
299    (g ^ i) ^ (i * (2 * κ)) = if i % 2 = 0 then 1 else -1 := by
300  have e1 : (g ^ i) ^ (i * (2 * κ)) = (g ^ (2 * κ)) ^ (i * i) := by
301    rw [← pow_mul, ← pow_mul]
302    show g ^ (i * (i * (2 * κ))) = g ^ (2 * κ * (i * i))
303    rw [← FRC.Nat.mul_assoc, Nat.mul_comm (i * i)]
304  rw [e1, F.half_period, neg_one_pow, sq_mod_two]

306/-- 00:C1, 13:J2, 13:I4 — the web closes: `2π ≡ −1` on every shell (`4κ = p − 1`): the height-two pinning
307`2π_A + 1 ≡ 0`, which holds in every fibre of a composite lift. -/
