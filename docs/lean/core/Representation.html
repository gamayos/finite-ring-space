19/-- A bijection as data: the map, an inverse, and the two inverse laws. -/
20⧉structure Bij (Z W : Type) where
21  toFun : Z → W
22  inv : W → Z
23  left : ∀ z, inv (toFun z) = z
24  right : ∀ w, toFun (inv w) = w
25
26/-- 4:B3 (Definition 2) — `E : X → W` is adequate for the observation map `g : Z → X` when `E ∘ g` is a
27bijection `φ : Z → W` (carried as data), `E (g z) = φ z`. -/
28⧉structure Adequate {Z X W : Type} (g : Z → X) (E : X → W) where
29  φ : Bij Z W
30  eq : ∀ z, E (g z) = φ.toFun z

34/-- 4:B4 — adequacy makes the observation map injective on the latent domain: `g z₁ = g z₂ → z₁ = z₂`. -/
35⧉theorem adequate_inj {g : Z → X} {E : X → W} (A : Adequate g E) {z₁ z₂ : Z} (h : g z₁ = g z₂) : z₁ = z₂ := by
36  have e : A.φ.toFun z₁ = A.φ.toFun z₂ := by rw [← A.eq, ← A.eq, h]
37  rw [← A.φ.left z₁, ← A.φ.left z₂, e]
38
39/-- 4:B5 (Lemma 2) — the transition `ψ = φ₂ ∘ φ₁⁻¹` between two adequate representations of one observation map:
40`E₂ (g z) = ψ (E₁ (g z))` for every latent state. -/
41⧉def transition {g : Z → X} {E₁ : X → W₁} {E₂ : X → W₂} (A₁ : Adequate g E₁) (A₂ : Adequate g E₂) : W₁ → W₂ :=
42  fun w => A₂.φ.toFun (A₁.φ.inv w)
43
44/-- 4:B5 — `E₂ ∘ g = ψ ∘ E₁ ∘ g` pointwise. -/
45⧉theorem transition_spec {g : Z → X} {E₁ : X → W₁} {E₂ : X → W₂} (A₁ : Adequate g E₁) (A₂ : Adequate g E₂) (z : Z) :
46    E₂ (g z) = transition A₁ A₂ (E₁ (g z)) := by
47  unfold transition; rw [A₁.eq, A₁.φ.left, A₂.eq]
48
49/-- 4:B5 — the transition is a bijection `W₁ → W₂`: its inverse is `φ₁ ∘ φ₂⁻¹`. -/
50⧉def transitionBij {g : Z → X} {E₁ : X → W₁} {E₂ : X → W₂} (A₁ : Adequate g E₁) (A₂ : Adequate g E₂) : Bij W₁ W₂ where
51  toFun := transition A₁ A₂
52  inv := fun w => A₁.φ.toFun (A₂.φ.inv w)
53  left := fun w => by show A₁.φ.toFun (A₂.φ.inv (A₂.φ.toFun (A₁.φ.inv w))) = w; rw [A₂.φ.left, A₁.φ.right]
54  right := fun w => by show A₂.φ.toFun (A₁.φ.inv (A₁.φ.toFun (A₂.φ.inv w))) = w; rw [A₁.φ.left, A₂.φ.right]
55
56/-- 4:B5 (the uniqueness clause) — any `ψ'` with `E₂ (g z) = ψ' (E₁ (g z))` for every `z` agrees with the
57transition on all of `W₁`, because `E₁ ∘ g` is onto `W₁`. -/
58⧉theorem transition_unique {g : Z → X} {E₁ : X → W₁} {E₂ : X → W₂} (A₁ : Adequate g E₁) (A₂ : Adequate g E₂)
59    (ψ' : W₁ → W₂) (h : ∀ z, E₂ (g z) = ψ' (E₁ (g z))) (w : W₁) : ψ' w = transition A₁ A₂ w := by
60  rw [← A₁.φ.right w, ← A₁.eq, ← h]; exact transition_spec A₁ A₂ _

62/-- 4:B6, 4:C2 (the canonical embedding into the shell) — with `incl : Z → U` the inclusion of the latent domain,
63`ι := incl ∘ φ⁻¹ : W → U` is injective when `incl` is, and `ι (E (g z)) = incl z`: the chart `ψ = ι ∘ φ` is the
64inclusion itself. -/
65⧉def iota {g : Z → X} {E : X → W} (A : Adequate g E) (incl : Z → U) : W → U := fun w => incl (A.φ.inv w)

67/-- 4:B6 — `ι` is injective when the inclusion is. -/
68⧉theorem iota_inj {g : Z → X} {E : X → W} (A : Adequate g E) {incl : Z → U}
69    (hincl : ∀ z₁ z₂, incl z₁ = incl z₂ → z₁ = z₂) {w₁ w₂ : W} (h : iota A incl w₁ = iota A incl w₂) : w₁ = w₂ := by
70  rw [← A.φ.right w₁, ← A.φ.right w₂, hincl _ _ h]

72/-- 4:C2 (Theorem 1 (iii)) — in the canonical parametrisation every chart is the inclusion: `ι (E (g z)) = incl z`. -/
73⧉theorem canonical_chart {g : Z → X} {E : X → W} (A : Adequate g E) (incl : Z → U) (z : Z) :
74    iota A incl (E (g z)) = incl z := by
75  unfold iota; rw [A.eq, A.φ.left]

77/-- 4:C1 (Theorem 1 (i)) — the chart `ψ = ι ∘ E ∘ g` is injective for any injective `ι`. -/
78⧉theorem chart_inj {g : Z → X} {E : X → W} (A : Adequate g E) {ι : W → U} (hι : ∀ w₁ w₂, ι w₁ = ι w₂ → w₁ = w₂)
79    {z₁ z₂ : Z} (h : ι (E (g z₁)) = ι (E (g z₂))) : z₁ = z₂ := by
80  have e : A.φ.toFun z₁ = A.φ.toFun z₂ := by rw [← A.eq, ← A.eq]; exact hι _ _ h
81  rw [← A.φ.left z₁, ← A.φ.left z₂, e]

83/-- 4:C1 (Theorem 1 (ii)) — the transition `Ψ_{m→n} = ψ_n ∘ ψ_m⁻¹` is well defined and bijective between the images:
84two latent states have the same `ψ_m`-image exactly when they have the same `ψ_n`-image, so the pairs
85`(ψ_m z, ψ_n z)` are the graph of a bijection `ψ_m(Z) → ψ_n(Z)`. -/
86⧉theorem transition_graph {ψm ψn : Z → U} (hm : ∀ z₁ z₂, ψm z₁ = ψm z₂ → z₁ = z₂)
87    (hn : ∀ z₁ z₂, ψn z₁ = ψn z₂ → z₁ = z₂) (z₁ z₂ : Z) : ψm z₁ = ψm z₂ ↔ ψn z₁ = ψn z₂ :=
88  ⟨fun h => by rw [hm _ _ h], fun h => by rw [hn _ _ h]⟩

91⧉def lift {g : Z → X} {E : X → W} (A : Adequate g E) : X → Z := fun x => A.φ.inv (E x)

93/-- 4:C3 — the lift recovers the latent state, `L (g z) = z`. -/
94⧉theorem lift_g {g : Z → X} {E : X → W} (A : Adequate g E) (z : Z) : lift A (g z) = z := by

97/-- 4:C3 — `L x` is the unique `z` with `E x = φ z`. -/
98⧉theorem lift_unique {g : Z → X} {E : X → W} (A : Adequate g E) (x : X) (z : Z) :
99    E x = A.φ.toFun z ↔ lift A x = z := by

102  · intro h; unfold lift at h; rw [← h, A.φ.right]

104/-- 4:C4 (Corollary 2) — cross-modal consistency: the lifts of two modalities' observations of one latent state
105agree, `L_m (g_m z) = L_n (g_n z)`, hence so do their canonical charts `ψ ∘ L_m ∘ g_m = ψ = ψ ∘ L_n ∘ g_n`. -/
106⧉theorem consistency {Xm Xn Wm Wn : Type} {gm : Z → Xm} {gn : Z → Xn} {Em : Xm → Wm} {En : Xn → Wn}
107    (Am : Adequate gm Em) (An : Adequate gn En) (incl : Z → U) (z : Z) :
108    lift Am (gm z) = lift An (gn z) ∧ incl (lift Am (gm z)) = incl z := by
109  rw [lift_g, lift_g]; exact ⟨rfl, rfl⟩

111/-! ### 4:E3 — Gödel collapse into one residue, the two-prime instance -/

113⧉theorem three_pow_mod_two : ∀ b : Nat, 3 ^ b % 2 = 1
114  | 0 => rfl
115  | b + 1 => by
116      rw [Nat.pow_succ, FRC.Nat.mul_mod _ _ 2 (Nat.zero_lt_succ 1), three_pow_mod_two b]
117
118⧉theorem two_mul_mod_two (x : Nat) : 2 * x % 2 = 0 := by
119  rw [FRC.Nat.mul_mod_left' 2 x 2 (Nat.zero_lt_succ 1)]
120  show 0 * x % 2 = 0
121  rw [Nat.zero_mul]
122
123⧉theorem three_pow_inj : ∀ b d : Nat, 3 ^ b = 3 ^ d → b = d
124  | 0, 0, _ => rfl
125  | 0, d + 1, h => by
126      have h1 : 1 * 3 ≤ 3 ^ d * 3 := Nat.mul_le_mul_right 3 (Nat.pow_pos (Nat.zero_lt_succ 2))
127      have h' : 1 = 3 ^ d * 3 := h
128      rw [← h'] at h1
129      exact absurd h1 (by decide)
130  | b + 1, 0, h => by
131      have h1 : 1 * 3 ≤ 3 ^ b * 3 := Nat.mul_le_mul_right 3 (Nat.pow_pos (Nat.zero_lt_succ 2))
132      have h' : 3 ^ b * 3 = 1 := h
133      rw [h'] at h1
134      exact absurd h1 (by decide)
135  | b + 1, d + 1, h =>
136      congrArg Nat.succ (three_pow_inj b d (Nat.eq_of_mul_eq_mul_right (Nat.zero_lt_succ 2) h))

138/-- 4:E3 (Gödel encoding, two primes) — `2^a 3^b = 2^c 3^d` forces `a = c` and `b = d`: the code is injective, by
139parity and cancellation; the general case over `d` distinct primes is `FrcLedger.Representation.godel_injective`. -/
140⧉theorem godel2_inj : ∀ a c b d : Nat, 2 ^ a * 3 ^ b = 2 ^ c * 3 ^ d → a = c ∧ b = d
141  | 0, 0, b, d, h => ⟨rfl, three_pow_inj b d (by rw [Nat.pow_zero, Nat.one_mul, Nat.one_mul] at h; exact h)⟩
142  | 0, c + 1, b, d, h => by
143      have hl : 2 ^ 0 * 3 ^ b % 2 = 1 := by rw [Nat.pow_zero, Nat.one_mul]; exact three_pow_mod_two b
144      have hr : 2 ^ (c + 1) * 3 ^ d % 2 = 0 := by
145        rw [Nat.pow_succ, FRC.Nat.mul_assoc, FRC.Nat.mul_left_comm]; exact two_mul_mod_two _
146      rw [h] at hl; rw [hl] at hr; exact absurd hr (by decide)
147  | a + 1, 0, b, d, h => by
148      have hr : 2 ^ 0 * 3 ^ d % 2 = 1 := by rw [Nat.pow_zero, Nat.one_mul]; exact three_pow_mod_two d
149      have hl : 2 ^ (a + 1) * 3 ^ b % 2 = 0 := by
150        rw [Nat.pow_succ, FRC.Nat.mul_assoc, FRC.Nat.mul_left_comm]; exact two_mul_mod_two _
151      rw [h] at hl; rw [hl] at hr; exact absurd hr (by decide)
152  | a + 1, c + 1, b, d, h => by
153      have h' : 2 * (2 ^ a * 3 ^ b) = 2 * (2 ^ c * 3 ^ d) := by
154        rw [← FRC.Nat.mul_left_comm, ← FRC.Nat.mul_assoc, ← Nat.pow_succ, h, Nat.pow_succ, FRC.Nat.mul_assoc,
155          FRC.Nat.mul_left_comm]
156      have ih := godel2_inj a c b d (Nat.eq_of_mul_eq_mul_left (Nat.zero_lt_succ 1) h')
157      exact ⟨congrArg Nat.succ ih.1, ih.2⟩

159/-- 4:E3 (the reduction into the shell) — an injective integer code with values below `q` stays injective
160modulo `q`: `f a % q = f b % q` forces `f a = f b`. -/
161⧉theorem mod_injective {α : Type} (f : α → Nat) (q : Nat) (hq : ∀ a, f a < q) {a b : α}
162    (h : f a % q = f b % q) : f a = f b := by
163  rw [FRC.Nat.mod_eq_of_lt (hq a), FRC.Nat.mod_eq_of_lt (hq b)] at h; exact h

173/-- 4:E1 — the character chart of the shell, `z ↦ (g^{kz})_{k}`: the finite shadow of the Fourier embedding
174`z ↦ (cos 2πkz/p, sin 2πkz/p)_k`. -/
175⧉def charChart (g : Shell p) (z k : Nat) : Shell p := g ^ (k * z)
176
177/-- 4:E1 (equivariance) — translation by `a` multiplies coordinate `k` by `g^{ka}`: the shell's translations act on
178the chart coordinatewise, as the rotations act on the Fourier embedding. -/
179⧉theorem charChart_shift (g : Shell p) (z a k : Nat) : charChart g (z + a) k = g ^ (k * a) * charChart g z k := by
180  unfold charChart; rw [Nat.mul_add, pow_add, mul_comm]
181
182/-- 4:E1 (constant norm) — the conjugate norm `Σ_{k<n} g^{kz} g^{k(n−z)}` of the chart equals `n = p − 1` for every
183`z ≤ n`: every point of the chart lies on one sphere, exactly. -/
184⧉theorem charChart_norm {κ : Nat} {g : Shell p} (F : Frame p κ g) (z : Nat) (hz : z ≤ p - 1) :
185    sumRange (fun k => charChart g z k * charChart g (p - 1 - z) k) (p - 1) = ofNat (p - 1) := by
186  have e : ∀ k, k < p - 1 → charChart g z k * charChart g (p - 1 - z) k = (fun _ => (1 : Shell p)) k := by
187    intro k _
188    show g ^ (k * z) * g ^ (k * (p - 1 - z)) = 1
189    rw [← pow_add, ← Nat.mul_add, FRC.Nat.add_sub_of_le hz, Nat.mul_comm, pow_mul, F.pow_n, one_pow]
190  rw [sum_congr (p - 1) e, sum_const, mul_one]
