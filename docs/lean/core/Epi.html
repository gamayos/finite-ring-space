19/-! ### The factorial, Kurepa's left factorial and the derangement numbers -/
20
21/-- The factorial. -/
22⧉def fact : Nat → Nat
23  | 0 => 1
24  | n + 1 => (n + 1) * fact n
25
26/-- The derangement numbers `!n` by the natural recurrence `!(n+2) = (n+1)(!n + !(n+1))`. -/
27⧉def dr : Nat → Nat
28  | 0 => 1
29  | 1 => 0
30  | n + 2 => (n + 1) * (dr n + dr (n + 1))
31
32⧉theorem dr_zero : dr 0 = 1 := rfl
33⧉theorem dr_one : dr 1 = 0 := rfl
34⧉theorem dr_add_two (n : Nat) : dr (n + 2) = (n + 1) * (dr n + dr (n + 1)) := rfl
35
36/-- Kurepa's left factorial `!p = Σ_{k<p} k!`. -/
37⧉def kur : Nat → Nat
38  | 0 => 0
39  | n + 1 => kur n + fact n
40
41/-- The central binomial coefficient `C(2n, n) = (2n)!/(n!·n!)`. -/
42⧉def cb (n : Nat) : Nat := fact (2 * n) / (fact n * fact n)

48/-- The factorial reduced on the shell, `k! mod p`, computed one factor at a time. -/
49⧉def factS : Nat → Shell p
50  | 0 => 1
51  | n + 1 => factS n * ofNat (n + 1)
52
53⧉theorem factS_eq : ∀ n, (factS n : Shell p) = ofNat (fact n)
54  | 0 => rfl
55  | n + 1 => by
56    show factS n * ofNat (n + 1) = ofNat ((n + 1) * fact n)
57    rw [factS_eq n, ofNat_mul, Nat.mul_comm]
58
59/-- `Σ_{k<n} k!` on the shell. -/
60⧉def kurS (n : Nat) : Shell p := sumRange (fun k => factS k) n
61
62/-- `Σ_{k<n} (−1)^k k!` on the shell (the alternating factorial sum `A(n)`). -/
63⧉def altS (n : Nat) : Shell p := sumRange (fun k => (-1) ^ k * factS k) n
64
65⧉theorem kurS_eq : ∀ n, (kurS n : Shell p) = ofNat (kur n)
66  | 0 => rfl
67  | n + 1 => by
68    show sumRange (fun k => factS k) n + factS n = ofNat (kur n + fact n)
69    rw [← ofNat_add, ← factS_eq]; exact congrArg (· + factS n) (kurS_eq n)

71/-- The inverse by bounded search: the `y < n` with `x·y = 1`, or `0` when there is none. -/
72⧉def invSearch (x : Shell p) : Nat → Shell p
73  | 0 => 0
74  | n + 1 => if x * ofNat n = 1 then ofNat n else invSearch x n
75
76/-- The inverse on the shell (`0` for a non-unit). -/
77⧉def inv (x : Shell p) : Shell p := invSearch x p
78
79/-- The reading of `ε_n = n!/!n` on the shell (`0` at a blind scale). -/
80⧉def readE (n : Nat) : Shell p := ofNat (fact n) * inv (ofNat (dr n))
81
82/-- The reading of the Wallis member `w_n = 16^n/(n·C(2n,n)²)` on the shell. -/
83⧉def readW (n : Nat) : Shell p := ofNat (16 ^ n) * inv (ofNat n * ofNat (cb n) ^ 2)
84
85/-- The reading of the Wallis member `v_n = 2·16^n/((2n+1)·C(2n,n)²)` on the shell. -/
86⧉def readV (n : Nat) : Shell p := ofNat (2 * 16 ^ n) * inv (ofNat (2 * n + 1) * ofNat (cb n) ^ 2)

90⧉theorem neg_one_pow_succ (n : Nat) : (-1 : Shell p) ^ (n + 1) = -((-1) ^ n) := by
91  rw [pow_succ, ← mul_neg, mul_one]

93/-- 13:F3 — the signed recurrence on the shell: `!(n+1) = (n+1)·!n + (−1)^{n+1}`. -/
94⧉theorem dr_succ : ∀ n : Nat, (ofNat (dr (n + 1)) : Shell p) = ofNat (n + 1) * ofNat (dr n) + (-1) ^ (n + 1)
95  | 0 => by
96    show (ofNat 0 : Shell p) = ofNat 1 * ofNat 1 + (-1) ^ 1
97    rw [pow_one, ofNat_mul, Nat.mul_one]
98    show (0 : Shell p) = 1 + -1
99    rw [add_neg]
100  | n + 1 => by
101    have ih := dr_succ n
102    rw [dr_add_two, ← ofNat_mul, ← ofNat_add (dr n) (dr (n + 1)), left_distrib]
103    have e1 : (ofNat (n + 1) : Shell p) * ofNat (dr n) = ofNat (dr (n + 1)) + -((-1) ^ (n + 1)) := by
104      rw [ih, add_assoc, add_neg, add_zero]
105    rw [e1, neg_one_pow_succ (n + 1), ofNat_succ (n + 1), right_distrib, one_mul,
106      add_comm (ofNat (dr (n + 1)) + -(-1) ^ (n + 1)) (ofNat (n + 1) * ofNat (dr (n + 1))), ← add_assoc]

112⧉theorem ofNat_add_self (n : Nat) : (ofNat (n + p) : Shell p) = ofNat n := by

115⧉theorem four_kappa_add_one_mod_two (κ : Nat) : (4 * κ + 1) % 2 = 1 :=

118⧉theorem neg_one_pow_p {κ : Nat} (hκ : p = 4 * κ + 1) : (-1 : Shell p) ^ p = -1 := by
119  have h2 : p % 2 = 1 := by rw [hκ]; exact four_kappa_add_one_mod_two κ
120  rw [neg_one_pow, h2]; rfl

122/-- 13:F3 — antiperiodicity on every shell `p = 4κ + 1`: `!(n + p) ≡ −!n (mod p)` for every `n`, so
123`!n mod p` depends on `n mod 2p`; and the blind set contains `n = 1` (`!1 = 0`). -/
124⧉theorem dr_antiperiodic {κ : Nat} (hκ : p = 4 * κ + 1) :
125    (∀ n, (ofNat (dr (n + p)) : Shell p) = -(ofNat (dr n))) ∧ dr 1 = 0 := by
126  refine ⟨?_, rfl⟩
127  intro n
128  induction n with
129  | zero =>
130    have h : (ofNat (dr (0 + p)) : Shell p) = ofNat p * ofNat (dr (p - 1)) + (-1) ^ p := by
131      have e : 0 + p = (p - 1) + 1 := by rw [Nat.zero_add, hκ]; rfl
132      rw [e, dr_succ, ← e, Nat.zero_add]
133    rw [h, ofNat_self, zero_mul, zero_add, neg_one_pow_p hκ]
134    rfl
135  | succ n ih =>
136    have e : n + 1 + p = (n + p) + 1 := Nat.add_right_comm n 1 p
137    have e2 : n + p + 1 = (n + 1) + p := Nat.add_right_comm n p 1
138    rw [e, dr_succ, ih, e2, ofNat_add_self (n + 1), pow_add, neg_one_pow_p hκ, ← mul_neg, ← mul_neg,
139      mul_one, ← neg_add_rev, ← dr_succ]

141/-- 13:F7 — the fixed-shell tower of `e`, exact at every grade: `(mp)! ≡ 0` and `!(mp) ≡ (−1)^m` for `m ≥ 1`,
142so the member `q_m = ((mp)! + δ_m)/!(mp)` reads `e_p` — numerator `≡ !(mp)·e_p` — whenever `δ_m ≡ (−1)^m e_p`. -/
143⧉theorem tower_e_exact {κ : Nat} (hκ : p = 4 * κ + 1) (m : Nat) (e δ : Shell p) (hδ : δ = (-1) ^ (m + 1) * e) :
144    (ofNat (fact ((m + 1) * p)) : Shell p) = 0 ∧ (ofNat (dr ((m + 1) * p)) : Shell p) = (-1) ^ (m + 1) ∧
145    ofNat (fact ((m + 1) * p)) + δ = ofNat (dr ((m + 1) * p)) * e := by
146  have hfact : ∀ k, (ofNat (fact (k + p)) : Shell p) = 0 := by
147    intro k
148    induction k with
149    | zero =>
150      have e : 0 + p = (p - 1) + 1 := by rw [Nat.zero_add, hκ]; rfl
151      rw [e]
152      show (ofNat ((p - 1 + 1) * fact (p - 1)) : Shell p) = 0
153      rw [← e, Nat.zero_add, ← ofNat_mul, ofNat_self, zero_mul]
154    | succ k ih =>
155      rw [Nat.add_right_comm]
156      show (ofNat ((k + p + 1) * fact (k + p)) : Shell p) = 0
157      rw [← ofNat_mul, ih, mul_zero]
158  have hdr : ∀ m, (ofNat (dr (m * p)) : Shell p) = (-1) ^ m := by
159    intro m
160    induction m with
161    | zero => rw [Nat.zero_mul]; rfl
162    | succ m ih => rw [Nat.succ_mul, (dr_antiperiodic hκ).1, ih, neg_one_pow_succ]
163  have hf : (ofNat (fact ((m + 1) * p)) : Shell p) = 0 := by
164    rw [Nat.succ_mul]; exact hfact _
165  refine ⟨hf, hdr (m + 1), ?_⟩
166  rw [hf, zero_add, hδ, hdr]

175⧉theorem odd_of_coprime_four (F : Frame p κ g) {u : Nat} (hu : Coprime u (p - 1)) : u % 2 = 1 := by
176  match hu with
177  | ⟨a, _, hau⟩ =>
178    match FRC.Nat.mod_spec 2 (Nat.zero_lt_succ 1) u with
179    | ⟨t, ht⟩ =>
180      match FRC.Nat.mod_spec (p - 1) F.n_pos (a * u) with
181      | ⟨q, hq⟩ =>
182        rw [hau, F.n_eq] at hq
183        -- a * u = 4κq + 1 is odd; if u % 2 = 0 then a * u = 2 * (a * t) is even
184        match h2 : u % 2 with
185        | 0 =>
186          rw [h2, Nat.add_zero] at ht
187          have hodd : (a * u) % 2 = 1 := by
188            apply FRC.Nat.mod_unique (q := 2 * κ * q) (Nat.lt_succ_self 1)
189            rw [hq]
190            show 4 * κ * q + 1 = 2 * (2 * κ * q) + 1
191            rw [← FRC.Nat.mul_assoc, ← FRC.Nat.mul_assoc]
192          have heven : (a * u) % 2 = 0 := by
193            apply FRC.Nat.mod_unique (q := a * t) (Nat.zero_lt_succ 1)
194            rw [ht, Nat.add_zero, FRC.Nat.mul_left_comm]
195          exact (Nat.succ_ne_zero 0 (hodd.symm.trans heven)).elim
196        | 1 => rfl
197        | r + 2 =>
198          have := Nat.mod_lt u (Nat.zero_lt_succ 1)
199          rw [h2] at this
200          exact absurd (Nat.lt_of_succ_lt_succ (Nat.lt_of_succ_lt_succ this)) (Nat.not_lt_zero r)
201
202⧉theorem mod_four_of_odd {u : Nat} (h : u % 2 = 1) : u % 4 = 1 ∨ u % 4 = 3 := by
203  match FRC.Nat.mod_spec 4 (Nat.zero_lt_succ 3) u with
204  | ⟨q, hq⟩ =>
205    have hr := Nat.mod_lt u (Nat.zero_lt_succ 3)
206    match hr4 : u % 4, hr with
207    | 0, _ =>
208      rw [hr4, Nat.add_zero] at hq
209      have : u % 2 = 0 := FRC.Nat.mod_unique (q := 2 * q) (Nat.zero_lt_succ 1)
210        (by rw [hq, Nat.add_zero, ← FRC.Nat.mul_assoc])
211      exact (Nat.succ_ne_zero 0 (this.symm.trans h).symm).elim
212    | 1, _ => exact Or.inl rfl
213    | 2, _ =>
214      rw [hr4] at hq
215      have : u % 2 = 0 := FRC.Nat.mod_unique (q := 2 * q + 1) (Nat.zero_lt_succ 1)
216        (by rw [hq, Nat.add_zero, Nat.mul_succ, ← FRC.Nat.mul_assoc])
217      exact (Nat.succ_ne_zero 0 (this.symm.trans h).symm).elim
218    | 3, _ => exact Or.inr rfl
219    | r + 4, hlt =>
220      exact absurd (Nat.lt_of_succ_lt_succ (Nat.lt_of_succ_lt_succ (Nat.lt_of_succ_lt_succ
221        (Nat.lt_of_succ_lt_succ hlt)))) (Nat.not_lt_zero r)

223/-- 13:B2 — orientation transport: under `g' = g^u` with `u` coprime to `p − 1`, `u` is odd, so either
224`u ≡ 1 (mod 4)` and the re-derived quarter-turn `−g'^κ` is `−g^κ` (the same chirality), or `u ≡ 3 (mod 4)`
225and it is `−(−g^κ)` (the conjugate chart); there is no third case. -/
226⧉theorem orientation_transport (F : Frame p κ g) {u : Nat} (hu : Coprime u (p - 1)) :
227    (u % 4 = 1 ∨ u % 4 = 3) ∧
228    (u % 4 = 1 → quarterTurn (g ^ u) κ = quarterTurn g κ) ∧
229    (u % 4 = 3 → quarterTurn (g ^ u) κ = -(quarterTurn g κ)) :=
230  ⟨mod_four_of_odd (odd_of_coprime_four F hu), (F.orientation_class u).1, (F.orientation_class u).2⟩

232end transport

237/-- 13:J1 — the wrap-free window: two counts below `√p` (`a·a < p`, `b·b < p`) have a product below `p`, and,
238for `p ≥ 4`, a sum below `p`: the accessible smalls neither wrap under `+` nor under `×`. -/
239⧉theorem wrap_free {a b p : Nat} (ha : a * a < p) (hb : b * b < p) :
240    a * b < p ∧ (4 ≤ p → a + b < p) := by
241  have hprod : a * b < p := by
242    match Nat.lt_or_ge a b with
243    | Or.inl hab => exact Nat.lt_of_le_of_lt (Nat.mul_le_mul_right b (Nat.le_of_lt hab)) hb
244    | Or.inr hba => exact Nat.lt_of_le_of_lt (Nat.mul_le_mul_left a hba) ha
245  refine ⟨hprod, fun h4 => ?_⟩
246  -- with c = max: a + b ≤ c + c ≤ c·c < p when c ≥ 2; a + b ≤ 2 < 4 ≤ p when c ≤ 1
247  have key : ∀ c, c * c < p → c + c < p := by
248    intro c hc
249    match c with
250    | 0 => exact Nat.lt_of_lt_of_le (Nat.zero_lt_succ 3) h4
251    | 1 => exact Nat.lt_of_lt_of_le (Nat.lt_succ_self 3 |> Nat.lt_of_succ_lt) h4
252    | c + 2 =>
253      have : (c + 2) + (c + 2) ≤ (c + 2) * (c + 2) := by
254        rw [← Nat.two_mul]; exact Nat.mul_le_mul_right (c + 2) (Nat.le_add_left 2 c)
255      exact Nat.lt_of_le_of_lt this hc
256  match Nat.lt_or_ge a b with
257  | Or.inl hab => exact Nat.lt_of_le_of_lt (Nat.add_le_add_right (Nat.le_of_lt hab) b) (key b hb)
258  | Or.inr hba => exact Nat.lt_of_le_of_lt (Nat.add_le_add_left hba a) (key a ha)
259
260end window
