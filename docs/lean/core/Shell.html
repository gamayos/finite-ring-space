3/-!
4# FrcCore.Shell — the residues of a shell, from first principles
5
6`Shell p` is the set of residues mod `p`, carried by their canonical representatives `val < p` — no
7quotient type, so no `Quot.sound`; every identity below is an identity of representatives, decided by
8`Nat` arithmetic. The operations reduce by `%`, so `decide` computes on any concrete shell through the
9kernel's accelerated `Nat` arithmetic, and every theorem is checked to depend on no axiom.
10
11The ring laws are derived from `FrcCore.Nat`'s division algorithm (`mod_unique`); negation is
12characterised by `add_neg : a + -a = 0`, and everything about `-` follows from the uniqueness of
13additive inverses (`neg_unique`), never from `Nat` subtraction.
14-/
15

18/-- A residue mod `p`: its representative and the bound. -/
19⧉structure Shell (p : Nat) where
20  val : Nat
21  lt : val < p
22
23/-- `Pos p`: the modulus is positive. Instances for literals `n + 1` are found automatically. -/
24class Pos (p : Nat) : Prop where
25  pos : 0 < p
26
27instance (n : Nat) : Pos (n + 1) := ⟨Nat.zero_lt_succ n⟩

33⧉theorem ext {a b : Shell p} (h : a.val = b.val) : a = b := by
34  cases a; cases b; cases h; rfl
35
36⧉theorem val_injective {a b : Shell p} (h : a = b) : a.val = b.val := by cases h; rfl
37
38instance : DecidableEq (Shell p) := fun a b =>
39  if h : a.val = b.val then isTrue (ext h) else isFalse (fun e => h (val_injective e))

44/-- The residue of a natural number. -/
45⧉def ofNat (n : Nat) : Shell p := ⟨n % p, Nat.mod_lt n Pos.pos⟩

47instance (n : Nat) : OfNat (Shell p) n := ⟨ofNat n⟩
48instance : Add (Shell p) := ⟨fun a b => ofNat (a.val + b.val)⟩
49instance : Mul (Shell p) := ⟨fun a b => ofNat (a.val * b.val)⟩

52/-- Powers by structural recursion (`a ^ 0 = 1`, `a ^ (n+1) = a ^ n * a`). -/

54  | 0 => 1

132⧉theorem add_neg (a : Shell p) : a + -a = 0 :=
133  ext (by
134    rw [val_add, val_neg, FRC.Nat.add_mod_mod _ _ _ hp, FRC.Nat.add_sub_of_le (Nat.le_of_lt a.lt),
135      FRC.Nat.mod_self p hp]; rfl)

137⧉theorem neg_add (a : Shell p) : -a + a = 0 := by rw [add_comm]; exact add_neg a
138
139/-- Additive inverses are unique. -/
140⧉theorem neg_unique {a x y : Shell p} (hx : a + x = 0) (hy : a + y = 0) : x = y := by
141  calc x = x + 0 := (add_zero x).symm
142    _ = x + (a + y) := by rw [hy]
143    _ = (x + a) + y := (add_assoc _ _ _).symm
144    _ = (a + x) + y := by rw [add_comm x a]
145    _ = 0 + y := by rw [hx]
146    _ = y := zero_add y
147
148⧉theorem neg_eq_of_add_eq_zero {a b : Shell p} (h : a + b = 0) : -a = b :=
149  neg_unique (add_neg a) h

151⧉theorem neg_neg (a : Shell p) : - -a = a :=
152  neg_eq_of_add_eq_zero (neg_add a)
153
154⧉theorem neg_zero : (-0 : Shell p) = 0 :=
155  neg_eq_of_add_eq_zero (add_zero 0)

157⧉theorem neg_mul (a b : Shell p) : -(a * b) = -a * b :=

212/-- `(-1)^n` by the parity of `n`. -/
213⧉theorem neg_one_pow (n : Nat) : (-1 : Shell p) ^ n = if n % 2 = 0 then 1 else -1 := by
214  induction n with
215  | zero => rfl
216  | succ n ih =>
217    rw [pow_succ, ih]
218    exact match Nat.decEq (n % 2) 0 with
219      | .isTrue h => by
220          rw [ite_eq_left h, one_mul]
221          have : (n + 1) % 2 = 1 := by
222            rw [← FRC.Nat.mod_add_mod _ _ _ (Nat.zero_lt_succ 1), h]
223          rw [ite_eq_right (by rw [this]; exact fun e => Nat.noConfusion e)]
224      | .isFalse h => by
225          rw [ite_eq_right h, neg_mul_neg, one_mul]
226          have h1 : n % 2 = 1 := by
227            have := Nat.mod_lt n (Nat.zero_lt_succ 1)
228            exact match n % 2, this, h with
229              | 0, _, h => absurd rfl h
230              | 1, _, _ => rfl
231              | k + 2, hk, _ => absurd hk (Nat.not_lt_of_le (Nat.le_add_left 2 k))
232          have : (n + 1) % 2 = 0 := by
233            rw [← FRC.Nat.mod_add_mod _ _ _ (Nat.zero_lt_succ 1), h1]
234          rw [ite_eq_left this]
