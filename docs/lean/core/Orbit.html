6`Coprime u n` is taken in its invertible form — some `a < n` has `a·u ≡ 1 (mod n)` — which is decidable by
7search and, for `n ≥ 1`, the same as `gcd(u, n) = 1` (Bezout); it is what every proof uses. Theorem:
8`h` is primitive of order `n = p − 1` exactly when `h = g^u` with `u < n` coprime to `n`. No axioms.

14/-- `u` is invertible mod `n`: some `a < n` has `a·u % n = 1`. -/
15⧉def Coprime (u n : Nat) : Prop := ∃ a, a < n ∧ (a * u) % n = 1
16
17instance (u n : Nat) : Decidable (Coprime u n) := decExistsLT (fun a => (a * u) % n = 1) n

22/-- A power of the drive with an invertible exponent is again primitive. -/
23⧉theorem primitive_pow_of_coprime (F : Frame p κ g) {u : Nat} (hu : Coprime u (p - 1)) :
24    IsPrimitive (g ^ u) (p - 1) := by
25  have hn := F.n_pos
26  match hu with
27  | ⟨a, _, ha⟩ =>
28    refine ⟨by rw [pow_mul_comm, F.pow_n, one_pow], ?_⟩
29    intro l hl hl0 hpow
30    -- (g^u)^l = 1 gives (u·l) % n = 0; with a·u = n·q + 1, g^l = g^{l·a·u} = ((g^u)^l)^a = 1
31    have h1 : (u * l) % (p - 1) = 0 := F.mod_eq_zero_of_pow_eq_one (by rw [pow_mul]; exact hpow)
32    match FRC.Nat.mod_spec (p - 1) hn (a * u) with
33    | ⟨q, hq⟩ =>
34      rw [ha] at hq
35      have e : l * (a * u) = (p - 1) * (q * l) + l := by
36        rw [hq, Nat.left_distrib, Nat.mul_one, FRC.Nat.mul_left_comm, Nat.mul_comm l q]
37      have h2 : g ^ (l * (a * u)) = 1 := by
38        rw [show l * (a * u) = (u * l) * a by rw [Nat.mul_comm u l, FRC.Nat.mul_assoc, Nat.mul_comm a u]]
39        rw [pow_mul, F.pow_eq_one_of_mod h1, one_pow]
40      rw [e, pow_add, pow_mul, F.pow_n, one_pow, one_mul] at h2
41      have := F.mod_eq_zero_of_pow_eq_one h2
42      rw [FRC.Nat.mod_eq_of_lt hl] at this
43      exact Nat.lt_irrefl 0 (this ▸ hl0)

49/-- 2:B3 (Props. 2.7, 4.5), 1-algebra's torsor — the primitive generators form one orbit: `h` is primitive
50of order `p − 1` exactly when `h = g^u` for some `u < p − 1` coprime to `p − 1`. -/
51⧉theorem generator_orbit (F : Frame p κ g) (h : Shell p) :
52    IsPrimitive h (p - 1) ↔ ∃ u, u < p - 1 ∧ Coprime u (p - 1) ∧ h = g ^ u := by
53  have hn := F.n_pos
54  constructor
55  · intro hh
56    have Fh : Frame p κ h := F.of_primitive hh
57    have hh0 : h ≠ 0 := Fh.g_ne_zero
58    match F.eq_pow_of_ne_zero hh0 with
59    | ⟨u, hu, e⟩ =>
60      refine ⟨u, hu, ?_, e.symm⟩
61      -- g is a power of h: g = h^v; then g^{u v} = g, so u·v ≡ 1 (mod n)
62      match Fh.eq_pow_of_ne_zero F.g_ne_zero with
63      | ⟨v, hv, ev⟩ =>
64        refine ⟨v, hv, ?_⟩
65        have e1 : g ^ (v * u) = g ^ 1 := by
66          rw [pow_one, Nat.mul_comm v u, pow_mul, e, ev]
67        have h1n : 1 < p - 1 := by
68          rw [F.n_eq]; exact Nat.lt_of_lt_of_le (by decide : 1 < 4 * 1) (Nat.mul_le_mul_left 4 F.cap_pos)
69        have := F.pow_inj (Nat.mod_lt _ hn) h1n (by rw [← F.pow_mod, e1])
70        exact this
71  · intro ⟨u, _, hu, e⟩
72    rw [e]; exact F.primitive_pow_of_coprime hu

74/-- 1:G1, the finitary core (Fermat): every residue satisfies `x^p = x` — the polynomial `X^p − X` vanishes on
75the whole shell, which is what the root test of 1:G1 rests on. -/
76⧉theorem fermat (F : Frame p κ g) (x : Shell p) : x ^ p = x := by
77  have hp1 : x ^ p = x ^ (p - 1) * x :=
78    congrArg (fun k => x ^ k) (FRC.Nat.sub_add_cancel Pos.pos).symm
79  rw [hp1]
80  exact match Shell.instDecidableEq x 0 with
81    | isTrue e => by rw [e, mul_zero]
82    | isFalse e => by
83        match F.eq_pow_of_ne_zero e with
84        | ⟨m, _, em⟩ => rw [← em, pow_mul_comm, F.pow_n, one_pow, one_mul]
