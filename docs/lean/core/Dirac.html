42/-- 8:B1 — an element `a + b w` of the coefficient field, on components. -/
43structure Ext (p : Nat) [Pos p] (ν : Shell p) where
44  re : Shell p
45  im : Shell p

47namespace Ext
48variable {ν : Shell p}

50theorem ext' {z z' : Ext p ν} (h1 : z.re = z'.re) (h2 : z.im = z'.im) : z = z' := by
51  cases z; cases z'; cases h1; cases h2; rfl
52
53instance : DecidableEq (Ext p ν) := fun a b =>
54  if h1 : a.re = b.re then
55    if h2 : a.im = b.im then isTrue (ext' h1 h2) else isFalse (fun e => h2 (by cases e; rfl))
56  else isFalse (fun e => h1 (by cases e; rfl))

58/-- The base field inside `K`. -/
59def ofShell (a : Shell p) : Ext p ν := ⟨a, 0⟩
60/-- The adjoined root `w`, `w² = ν`. -/
61def w : Ext p ν := ⟨0, 1⟩
62instance (n : Nat) : OfNat (Ext p ν) n := ⟨ofShell (OfNat.ofNat n)⟩
63instance : Add (Ext p ν) := ⟨fun z z' => ⟨z.re + z'.re, z.im + z'.im⟩⟩
64instance : Neg (Ext p ν) := ⟨fun z => ⟨-z.re, -z.im⟩⟩
65instance : Mul (Ext p ν) := ⟨fun z z' => ⟨z.re * z'.re + ν * (z.im * z'.im), z.re * z'.im + z.im * z'.re⟩⟩
66/-- Frobenius conjugation `a + b w ↦ a − b w`. -/
67def conj (z : Ext p ν) : Ext p ν := ⟨z.re, -z.im⟩
68/-- The norm `N(z) = z z̄ = a² − ν b²`. -/
69def norm (z : Ext p ν) : Shell p := z.re * z.re + -(ν * (z.im * z.im))
70/-- Powers by structural recursion. -/
71def pow (z : Ext p ν) : Nat → Ext p ν
72  | 0 => 1
73  | n + 1 => pow z n * z
74instance : Pow (Ext p ν) Nat := ⟨pow⟩

76theorem mul_re (z z' : Ext p ν) : (z * z').re = z.re * z'.re + ν * (z.im * z'.im) := rfl
77theorem mul_im (z z' : Ext p ν) : (z * z').im = z.re * z'.im + z.im * z'.re := rfl
78theorem ofShell_re (a : Shell p) : (ofShell a : Ext p ν).re = a := rfl
79theorem ofShell_im (a : Shell p) : (ofShell a : Ext p ν).im = 0 := rfl
80theorem conj_re (z : Ext p ν) : (conj z).re = z.re := rfl
81theorem conj_im (z : Ext p ν) : (conj z).im = -z.im := rfl

83/-- 8:A2 — `w² = ν`. -/
84theorem w_sq : (w : Ext p ν) * w = ofShell ν :=
85  ext' (by show 0 * 0 + ν * (1 * 1) = ν; rw [Shell.zero_mul, Shell.one_mul, Shell.mul_one, Shell.zero_add])
86       (by show 0 * 1 + 1 * 0 = 0; rw [Shell.zero_mul, Shell.mul_zero, Shell.zero_add])

88/-- 8:A2 — the norm is the product with the conjugate: `z z̄ = N(z)`, a base-field element. -/
89theorem mul_conj (z : Ext p ν) : z * conj z = ofShell (norm z) :=
90  ext' (by show z.re * z.re + ν * (z.im * -z.im) = z.re * z.re + -(ν * (z.im * z.im))
91           rw [← Shell.mul_neg, ← Shell.mul_neg])
92       (by show z.re * -z.im + z.im * z.re = 0
93           rw [← Shell.mul_neg, Shell.mul_comm z.im z.re, Shell.neg_add])
94
95/-- 8:A2 — conjugation is multiplicative (the Frobenius involution is a ring map). -/
96theorem conj_mul (z z' : Ext p ν) : conj (z * z') = conj z * conj z' :=
97  ext' (by show z.re * z'.re + ν * (z.im * z'.im) = z.re * z'.re + ν * (-z.im * -z'.im)
98           rw [Shell.neg_mul_neg])
99       (by show -(z.re * z'.im + z.im * z'.re) = z.re * -z'.im + -z.im * z'.re
100           rw [Shell.neg_add_rev, ← Shell.mul_neg, ← Shell.neg_mul])

103norm by `g²`, a square. -/

105  unfold norm
106  rw [mul_re, mul_im, ofShell_re, ofShell_im, Shell.zero_mul, Shell.mul_zero, Shell.add_zero,

117theorem two_mul_mod (l : Nat) : (2 * l) % 2 = 0 := by

120theorem mod_two_of_mod_four_mul (m q : Nat) : (4 * q + m) % 2 = m % 2 := by
121  have : 4 * q + m = 2 * (2 * q) + m := by rw [← FRC.Nat.mul_assoc]
122  rw [this, FRC.Nat.add_mul_mod_self_left m (2 * q) 2 (Nat.zero_lt_succ 1)]

124theorem mod_n_mod_two (F : Frame p κ g) (m : Nat) : (m % (p - 1)) % 2 = m % 2 := by

129    calc (m % (p - 1)) % 2 = (4 * (κ * q) + m % (p - 1)) % 2 := (mod_two_of_mod_four_mul _ _).symm

132/-- 8:B5 — the square class is chronon parity: on every frame, `g^m` is a square exactly when the drive-step
133count `m` is even (Theorem `parity`). -/
134theorem parity_iff (F : Frame p κ g) (m : Nat) : (∃ y : Shell p, y * y = g ^ m) ↔ m % 2 = 0 := by
135  constructor
136  · rintro ⟨y, hy⟩
137    have hy0 : y ≠ 0 := fun h0 => F.pow_ne_zero m (by rw [← hy, h0, Shell.mul_zero])
138    obtain ⟨l, hl, hgl⟩ := F.eq_pow_of_ne_zero hy0
139    have h2 : g ^ (2 * l) = g ^ m := by
140      rw [Nat.mul_comm, Shell.pow_mul, Shell.pow_two, hgl]; exact hy
141    rw [F.pow_mod, F.pow_mod m] at h2
142    have h3 := F.pow_inj (Nat.mod_lt _ F.n_pos) (Nat.mod_lt _ F.n_pos) h2
143    rw [← mod_n_mod_two F m, ← h3, mod_n_mod_two F]
144    exact two_mul_mod l
145  · intro h
146    match FRC.Nat.mod_spec 2 (Nat.zero_lt_succ 1) m with
147    | ⟨q, hq⟩ =>
148      rw [h, Nat.add_zero] at hq
149      exact ⟨g ^ q, by rw [hq, Nat.mul_comm, Shell.pow_mul, Shell.pow_two]⟩

155theorem mod_two_cases (m : Nat) : m % 2 = 0 ∨ m % 2 = 1 := by

168/-- The inverse of the drive is `g^{p−2}`: every `y` with `g y = 1` is an odd power of `g`. -/
169theorem inv_drive_odd (F : Frame p κ g) {y : Shell p} (hy : g * y = 1) :
170    ∃ l, l < p - 1 ∧ g ^ l = y ∧ l % 2 = 1 := by
171  have hy0 : y ≠ 0 := fun h0 => F.one_ne_zero (by rw [← hy, h0, Shell.mul_zero])
172  obtain ⟨l, hl, hgl⟩ := F.eq_pow_of_ne_zero hy0
173  refine ⟨l, hl, hgl, ?_⟩
174  have h1 : g ^ (l + 1) = 1 := by rw [Shell.pow_succ, hgl, Shell.mul_comm]; exact hy
175  have h2 := F.mod_eq_zero_of_pow_eq_one h1
176  have h3 : l + 1 = p - 1 := by
177    match Nat.lt_or_ge (l + 1) (p - 1) with
178    | .inl hlt => exact absurd (by rw [FRC.Nat.mod_eq_of_lt hlt] at h2; exact h2) (Nat.succ_ne_zero l)
179    | .inr hge => exact Nat.le_antisymm (Nat.succ_le_of_lt hl) hge

181    rw [h3, F.n_eq, ← Nat.add_zero (4 * κ), mod_two_of_mod_four_mul]

243/-- 8:B4 — the factorisation of the coefficient: with `c² = 2⁻¹`, `ν = g = c²·(2g)` exactly on every shell,

275/-- 8:F2 — `4κ ≡ −1` on the shell (`4κ = p − 1`), so `κ = −4⁻¹` and the energy radius `1 + κ = 3·4⁻¹`. -/
276theorem four_kappa (F : Frame p κ g) {q : Shell p} (hq : 4 * q = 1) :
277    (4 : Shell p) * ofNat κ = -1 ∧ (ofNat κ : Shell p) = -q ∧ (1 : Shell p) + ofNat κ = 3 * q := by
278  have h4 : (4 : Shell p) * ofNat κ = -1 := by
279    have := F.two_pi
280    unfold Frame.halfPeriod at this
281    rw [← FRC.Nat.mul_assoc, ← Frame.ofNat_mul] at this
282    exact this
283  have hk : (ofNat κ : Shell p) = -q := by
284    calc (ofNat κ : Shell p) = 1 * ofNat κ := (Shell.one_mul _).symm
285      _ = 4 * q * ofNat κ := by rw [hq]
286      _ = q * (4 * ofNat κ) := by rw [Shell.mul_comm 4 q, Shell.mul_assoc]
287      _ = -q := by rw [h4, ← Shell.mul_neg, Shell.mul_one]
288  refine ⟨h4, hk, ?_⟩
289  have h3 : (4 : Shell p) = 3 + 1 :=
290    Shell.ext (by rw [Shell.val_add, Shell.val_lit, Shell.val_lit, Shell.val_one,
291      FRC.Nat.mod_add_mod _ _ _ Shell.hp, FRC.Nat.add_mod_mod _ _ _ Shell.hp])
292  rw [hk, ← hq, h3, Shell.right_distrib, Shell.one_mul, Shell.add_assoc, Shell.add_neg, Shell.add_zero]

302    exact this

305    _ = h * (2 * ofNat (2 * κ)) := by rw [Shell.mul_comm 2 h, Shell.mul_assoc]

321/-- 8:F3 — the exceptional eigenvalue: `k = 2κ` gives `g^{−2κ} = −1` (and `k = 0` gives `1`) — the
322intersection `N¹ ∩ μ_{p−1} = {±1}` of the norm-one torus with the phase cycle. -/
323theorem exceptional_eigen (F : Frame p κ g) {y : Shell p} (hy : g * y = 1) : y ^ (2 * κ) = -1 := by
324  have h1 : y ^ (2 * κ) * g ^ (2 * κ) = 1 := by
325    rw [← Shell.mul_pow, Shell.mul_comm y g, hy, Shell.one_pow]
326  rw [F.half_period] at h1
327  calc y ^ (2 * κ) = y ^ (2 * κ) * (-1 * -1) := by rw [Shell.neg_mul_neg, Shell.one_mul, Shell.mul_one]
328    _ = y ^ (2 * κ) * -1 * -1 := (Shell.mul_assoc _ _ _).symm
329    _ = -1 := by rw [h1, Shell.one_mul]
330
331/-- 8:F3 — isotropy: the character line `χ_k` is isotropic, `Σ_j χ_k(g^j)² = Σ_j g^{2jk} = 0`, unless
332`(p − 1) ∣ 2k`; on the two exceptional lines the sum is `p − 1 = −1 ≠ 0` (Theorem `zonal` (3)). -/
333theorem isotropy (F : Frame p κ g) (k : Nat) :
334    ((2 * k) % (p - 1) ≠ 0 → sumRange (fun j => g ^ (j * k) * g ^ (j * k)) (p - 1) = 0) ∧
335    ((2 * k) % (p - 1) = 0 → sumRange (fun j => g ^ (j * k) * g ^ (j * k)) (p - 1) = -1) := by
336  have e : ∀ j, g ^ (j * k) * g ^ (j * k) = (g ^ (2 * k)) ^ j := fun j => by
337    rw [← Shell.pow_add, ← Shell.pow_mul]
338    show g ^ (j * k + j * k) = g ^ (2 * k * j)
339    rw [← Nat.two_mul, Nat.mul_assoc 2 k j, Nat.mul_comm k j]
340  rw [sum_congr (p - 1) (fun j _ => e j)]
341  constructor
342  · intro hk
343    have hx : g ^ (2 * k) ≠ 1 := fun h1 => hk (F.mod_eq_zero_of_pow_eq_one h1)
344    have hn : (g ^ (2 * k)) ^ (p - 1) = 1 := by rw [Shell.pow_mul_comm, F.pow_n, Shell.one_pow]
345    exact Frame.geom_sum_eq_zero F (p - 1) hn hx
346  · intro hk
347    rw [sum_congr (p - 1) (fun j _ => by rw [F.pow_eq_one_of_mod hk, Shell.one_pow]),
348      sum_const, Shell.mul_one]
349    exact Frame.ofNat_n F
350
351/-- 8:F4 — sector separation: the inverse of the drive is primitive of order `p − 1` (`ord(g⁻¹) = p − 1`,
352so the free evolution is no Cayley step, whose scalar phases have order dividing `p + 1`), and the norm-one
353torus meets the base field in `{±1}` alone (`u² = 1 ⇒ u = ±1`). -/
354theorem sector_separation (F : Frame p κ g) {y : Shell p} (hy : g * y = 1) :
355    IsPrimitive y (p - 1) ∧ ∀ u : Shell p, u * u = 1 → u = 1 ∨ u = -1 := by
356  refine ⟨⟨?_, fun l hl hl0 h1 => ?_⟩, fun u hu => F.sq_eq_one hu⟩
357  · rw [← Shell.one_pow (p - 1), ← hy, Shell.mul_pow, F.pow_n, Shell.one_mul]
358  · have h2 : g ^ l = 1 := by
359      calc g ^ l = g ^ l * 1 := (Shell.mul_one _).symm
360        _ = g ^ l * y ^ l := by rw [h1]
361        _ = 1 := by rw [← Shell.mul_pow, hy, Shell.one_pow]
362    exact F.prim.2 l hl hl0 h2
363
364end drive

416/-- A matrix over `K` as a list of rows. -/

421/-- The `j`-th element of a list, `d` beyond its end (structural; Lean's `List.getD` carries `propext`). -/
422def nth {α : Type} (d : α) : List α → Nat → α
423  | [], _ => d
424  | a :: _, 0 => a
425  | _ :: l, n + 1 => nth d l n
426def dot (r c : List (Ext p ν)) : Ext p ν := (List.zipWith (· * ·) r c).foldl (· + ·) 0
427def col (A : Mat p ν) (j : Nat) : List (Ext p ν) := A.map fun r => nth 0 r j
428def transpose (A : Mat p ν) : Mat p ν := (List.range (A.headD []).length).map (col A)
429def mul (A B : Mat p ν) : Mat p ν :=
430  let Bt := transpose B
431  A.map fun r => Bt.map fun c => dot r c
432def add (A B : Mat p ν) : Mat p ν := List.zipWith (fun r s => List.zipWith (· + ·) r s) A B
433def neg (A : Mat p ν) : Mat p ν := A.map fun r => r.map (- ·)
434def smul (c : Ext p ν) (A : Mat p ν) : Mat p ν := A.map fun r => r.map (c * ·)
435def ident (n : Nat) : Mat p ν := (List.range n).map fun i => (List.range n).map fun j => if i = j then 1 else 0
436/-- Powers by squaring. -/
437def pow (A : Mat p ν) (n : Nat) : Nat → Mat p ν
438  | 0 => ident n
439  | e + 1 => mul (pow A n e) A
440/-- The conjugate transpose (Frobenius on the entries). -/
441def dagger (A : Mat p ν) : Mat p ν := (transpose A).map fun r => r.map Ext.conj

443instance : Mul (Mat p ν) := ⟨mul⟩
444instance : Add (Mat p ν) := ⟨add⟩
445instance : Neg (Mat p ν) := ⟨neg⟩

451/-- The gamma matrices of `(eq:gamma-definition)` over `K`, given the quarter-turn `i` of the shell:
452`γ⁰ = i w β`, `γ^k = i ρ^k` (with `i² = −1`, so `γ²`'s entries are `∓i² = ±1`). -/
453def gamma0 (i : Shell p) : Mat p ν :=
454  let iw : Ext p ν := ⟨0, i⟩
455  [[0, 0, iw, 0], [0, 0, 0, iw], [iw, 0, 0, 0], [0, iw, 0, 0]]
456def gamma1 (i : Shell p) : Mat p ν :=
457  let ii : Ext p ν := ⟨i, 0⟩
458  [[0, 0, 0, ii], [0, 0, ii, 0], [0, -ii, 0, 0], [-ii, 0, 0, 0]]
459def gamma2 (i : Shell p) : Mat p ν :=
460  let ii : Ext p ν := ⟨i, 0⟩
461  [[0, 0, 0, -(ii * ii)], [0, 0, ii * ii, 0], [0, ii * ii, 0, 0], [-(ii * ii), 0, 0, 0]]
462def gamma3 (i : Shell p) : Mat p ν :=
463  let ii : Ext p ν := ⟨i, 0⟩
464  [[0, 0, ii, 0], [0, 0, 0, -ii], [-ii, 0, 0, 0], [0, ii, 0, 0]]
465def gamma (i : Shell p) : Nat → Mat p ν
466  | 0 => gamma0 i
467  | 1 => gamma1 i
468  | 2 => gamma2 i
469  | _ => gamma3 i
470/-- `η = diag(−ν, 1, 1, 1)`. -/
471def eta (ν : Shell p) : Nat → Nat → Ext p ν
472  | 0, 0 => -(⟨ν, 0⟩ : Ext p ν)
473  | 1, 1 => 1
474  | 2, 2 => 1
475  | 3, 3 => 1
476  | _, _ => 0

478/-- 8:D2 — the Clifford relations `γ^μ γ^ν + γ^ν γ^μ = 2 η_{μν} I₄`, all sixteen. -/
479def clifford (ν : Shell p) (i : Shell p) : Bool :=
480  (List.range 4).all fun μ => (List.range 4).all fun μ' =>
481    decide (gamma i μ * gamma i μ' + gamma i μ' * gamma i μ = smul (2 * eta ν μ μ') (ident 4 : Mat p ν))

483/-- 8:D2 [value] — the Clifford relations hold on `𝔽₁₃` (`ν = 2`, `i = 5`), `𝔽₁₇` (`ν = 3`, `i = 4`) and
484`𝔽₅` (`ν = 2`, `i = 2`). -/
485theorem clifford_values : clifford (2 : Shell 13) 5 = true ∧ clifford (3 : Shell 17) 4 = true ∧
486    clifford (2 : Shell 5) 2 = true := by decide +kernel

488/-- The spin lift `S(x, y) = x I₄ + y γ⁰γ¹` and its inverse `δ⁻¹(x I₄ − y γ⁰γ¹)`, `δ = x² − ν y²`, with the
489inverse of `δ` supplied. -/
490def spinLift (i x y : Shell p) : Mat p ν :=
491  smul ⟨x, 0⟩ (ident 4) + smul ⟨y, 0⟩ (gamma0 i * gamma1 i)
492def spinLiftInv (i x y dinv : Shell p) : Mat p ν :=
493  smul ⟨dinv, 0⟩ (smul ⟨x, 0⟩ (ident 4) + smul ⟨-y, 0⟩ (gamma0 i * gamma1 i))
494
495/-- 8:D6, 8:D7, 8:D12 — the spin conjugation `S⁻¹γ⁰S = Aγ⁰ + νBγ¹`, `S⁻¹γ¹S = Bγ⁰ + Aγ¹`, `γ², γ³` fixed, and
496the transported gammas `Sγ⁰S⁻¹ = Aγ⁰ − νBγ¹`, `Sγ¹S⁻¹ = −Bγ⁰ + Aγ¹`, at the lift `(x, y)` with the entries
497`A`, `B` given. -/
498def spinConj (ν i x y dinv A B : Shell p) : Bool :=
499  let S : Mat p ν := spinLift i x y
500  let Si : Mat p ν := spinLiftInv i x y dinv
501  let a : Ext p ν := ⟨A, 0⟩
502  let b : Ext p ν := ⟨B, 0⟩
503  let n : Ext p ν := ⟨ν, 0⟩
504  decide (Si * S = ident 4) &&
505  decide (Si * gamma0 i * S = smul a (gamma0 i) + smul (n * b) (gamma1 i)) &&
506  decide (Si * gamma1 i * S = smul b (gamma0 i) + smul a (gamma1 i)) &&
507  decide (Si * gamma2 i * S = gamma2 i) && decide (Si * gamma3 i * S = gamma3 i) &&
508  decide (S * gamma0 i * Si = smul a (gamma0 i) + smul (-(n * b)) (gamma1 i)) &&
509  decide (S * gamma1 i * Si = smul (-b) (gamma0 i) + smul a (gamma1 i))
510
511/-- 8:D6, 8:D7, 8:D12 [value] — on `𝔽₁₃` (`ν = g = 2`, `x = y = 1`, `δ = −1`): `A = 10`, `B = 2`,
512`γ̂⁰ = 10γ⁰ − 4γ¹`, `γ̂¹ = −2γ⁰ + 10γ¹`; on `𝔽₁₇` (`ν = g = 3`, `δ = −2`, `δ⁻¹ = 8`): `A = 15`, `B = 1`,
513`γ̂⁰ = 15γ⁰ − 3γ¹`, `γ̂¹ = −γ⁰ + 15γ¹`. -/
514theorem spin_conjugation_values :
515    spinConj (2 : Shell 13) 5 1 1 12 10 2 = true ∧ spinConj (3 : Shell 17) 4 1 1 8 15 1 = true := by
516  decide +kernel

518/-- 8:D9 — the spinor twist `X = γ⁰γ¹γ³`: Hermitian for the conjugate transpose, `X² = ν I₄`, commuting with
519`γ⁰, γ¹, γ³`, anticommuting with `γ²`, and `(γ^μ)† X = −X γ^μ` for every `μ` (so `X⁻¹(γ^μ)†X = −γ^μ`); the
520component signs `(γ⁰)† = −γ⁰`, `(γ¹)† = −γ¹`, `(γ²)† = +γ²` (the obstruction), `(γ³)† = −γ³`; and the failure
521of the `γ⁰`-twist: `(γ¹)† γ⁰ ≠ −γ⁰ γ¹`. -/
522def spinorForm (ν i : Shell p) : Bool :=
523  let X : Mat p ν := gamma0 i * gamma1 i * gamma3 i
524  let n : Ext p ν := ⟨ν, 0⟩
525  decide (dagger X = X) && decide (X * X = smul n (ident 4)) &&
526  decide (X * gamma0 i = gamma0 i * X) && decide (X * gamma1 i = gamma1 i * X) &&
527  decide (X * gamma3 i = gamma3 i * X) && decide (X * gamma2 i = -(gamma2 i * X)) &&
528  (List.range 4).all (fun μ => decide (dagger (gamma i μ) * X = -(X * gamma i μ))) &&
529  decide ((dagger (gamma0 i) : Mat p ν) = -(gamma0 i)) && decide ((dagger (gamma1 i) : Mat p ν) = -(gamma1 i)) &&
530  decide ((dagger (gamma2 i) : Mat p ν) = gamma2 i) && decide ((dagger (gamma3 i) : Mat p ν) = -(gamma3 i)) &&
531  decide ((dagger (gamma1 i) : Mat p ν) * gamma0 i ≠ -(gamma0 i * gamma1 i))
532
533/-- 8:D9 [value] — the spinor form on `𝔽₅`, `𝔽₁₃` and `𝔽₁₇`. -/
534theorem spinor_form_values :
535    spinorForm (2 : Shell 5) 2 = true ∧ spinorForm (2 : Shell 13) 5 = true ∧ spinorForm (3 : Shell 17) 4 = true := by
536  decide +kernel

538/-- The inverse of a residue by bounded search (`0` when none). -/
539def invS (a : Shell p) : Shell p :=
540  match (List.range p).find? (fun b => decide (a * ofNat b = 1)) with
541  | some b => ofNat b
542  | none => 0

544/-- A circulant `p × p` matrix over `K` by its first row; the product of two circulants is the cyclic
545convolution of their first rows. -/
546def cyc (n : Nat) (a b : List (Ext p ν)) : List (Ext p ν) :=
547  (List.range n).map fun k =>
548    ((List.range n).map fun j => nth 0 a j * nth 0 b ((k + n - j) % n)).foldl (· + ·) 0
549def cycId (n : Nat) : List (Ext p ν) := (List.range n).map fun k => if k = 0 then 1 else 0
550def cycAdd (a b : List (Ext p ν)) : List (Ext p ν) := List.zipWith (· + ·) a b
551def cycSmul (c : Ext p ν) (a : List (Ext p ν)) : List (Ext p ν) := a.map (c * ·)
552def cycPow (n : Nat) (a : List (Ext p ν)) : Nat → List (Ext p ν)
553  | 0 => cycId n
554  | e + 1 => cyc n (cycPow n a e) a
555
556/-- `H = −Δ = 2I − T − T⁻¹` on `𝔽_p`, the kinetic Hamiltonian at unit stiffness, a circulant with first row
557`(2, −1, 0, …, 0, −1)`. -/
558def kineticRow (p : Nat) [Pos p] (ν : Shell p) : List (Ext p ν) :=
559  (List.range p).map fun j => if j = 0 then 2 else if j = 1 ∨ j + 1 = p then -1 else 0

561/-- The Cayley step `U = (I − αH)⁻¹(I + αH)` of a nilpotent circulant `H`, the inverse as the series
562`Σ_{j≤n} (αH)^j` (exact once `(αH)^n = 0`, which `cayley13` checks). -/
563def cayleyRow (α : Ext p ν) (H : List (Ext p ν)) : List (Ext p ν) :=
564  let N := cycSmul α H
565  let inv := (List.range p).foldl (fun acc _ => cycAdd (cycId p) (cyc p N acc)) (cycId p)
566  cyc p inv (cycAdd (cycId p) N)

568/-- 8:C6, 8:E4 — on `𝔽₁₃` with `ν = g = 2`: for `α = w` and `α = c = 6w` (`c² = 2⁻¹ = 7`, `(6w)² = 36·2 = 7`),
569`(αH)^{13} = 0` (so the series inverse is exact), `U^{13} = I` and `U ≠ I`: the order is exactly `13`, the
570translation torus `C_p` (Example `cayley-13`, Theorem `period-dichotomy` (1)). -/
571def cayley13 (α : Ext 13 2) : Bool :=
572  let H := kineticRow 13 2
573  let U := cayleyRow α H
574  decide (cycPow 13 (cycSmul α H) 13 = (List.range 13).map fun _ => (0 : Ext 13 2)) &&
575  decide (cycPow 13 U 13 = cycId 13) && decide (U ≠ cycId 13)

582/-- The scalar Cayley map `φ(λ) = (1 + wλ)/(1 − wλ)` (`α = w`) on `K`, by the conjugate and the searched
583inverse of the norm. -/
584def phi (ν : Shell p) (lam : Shell p) : Ext p ν :=

589/-- The order of `u` in the torus, by search up to `p + 1` (`0` if none). -/
590def orderIn (u : Ext p ν) : Nat :=

595/-- 8:E4 (2), 8:C5 — the potential propagator `U = diag(φ(V(x)))` for `V = id`: every phase `φ(λ)` has norm
596one and `φ(λ)^{p+1} = 1`, and the order of the diagonal is the least common multiple of the phase orders —
597`14 = p + 1` on `𝔽₁₃` and `18 = p + 1` on `𝔽₁₇`: the bound `p + 1` is attained. -/
598def potentialOrder (p : Nat) [Pos p] (ν : Shell p) : Nat :=
