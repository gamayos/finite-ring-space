51    exact hx ((mul_eq_zero.1 this).resolve_left hchar)

53  · intro h; apply hi; rw [sq]; nth_rewrite 2 [h]; rw [mul_neg, mul_inv_cancel₀ hx]

92/-- 1:B4 (Definition 2 corrected, Lemma 1): in the affine frame `(a, b)`, `φ x = a + b x`
93carries `+` and `·` to `⊕` and `⊗`, is a bijection, and the multiplicative unit of the
94relabelled field is `a + b`; `b` is the unit only when `a = 0`. -/
95⧉theorem affine_frame (a b : F) (hb : b ≠ 0) :
96    let φ : F → F := fun x => a + b * x
97    let oplus : F → F → F := fun x y => a + b * ((x - a) / b + (y - a) / b)
98    let otimes : F → F → F := fun x y => a + b * ((x - a) / b * ((y - a) / b))
99    (∀ x y, φ (x + y) = oplus (φ x) (φ y)) ∧
100    (∀ x y, φ (x * y) = otimes (φ x) (φ y)) ∧
101    Function.Bijective φ ∧
102    (∀ y, otimes (a + b) y = y) ∧
103    (a ≠ 0 → otimes b (a + b) ≠ a + b) := by
104  intro φ oplus otimes
105  refine ⟨?_, ?_, ?_, ?_, ?_⟩
106  · intro x y; simp only [φ, oplus]; field_simp; ring
107  · intro x y; simp only [φ, otimes]; field_simp; ring
108  · constructor
109    · intro x y h; simp only [φ] at h
110      exact mul_left_cancel₀ hb (add_left_cancel h)
111    · intro y; exact ⟨(y - a) / b, by simp only [φ]; field_simp; ring⟩
112  · intro y; simp only [otimes]; field_simp
113    ring
114  · intro ha h; simp only [otimes] at h
115    -- a + b·((b−a)/b · b/b) = a + b  ⇒  ab = 0  ⇒  a = 0
116    field_simp at h
117    have hab : a * b = 0 := by linear_combination -h
118    exact ha ((mul_eq_zero.1 hab).resolve_right hb)

128/-- 1:D2 (Definition 6 made precise): the framed-integer window. Integers of absolute value
129`≤ H` with `2H < card F` have distinct residues: the cast `ℤ → ZMod p` is injective on the
130window. Stated on `ZMod p`. -/
131⧉theorem framed_integer_window (p : ℕ) [NeZero p] (H : ℕ) (hH : 2 * H < p) (x y : ℤ)
132    (hx : |x| ≤ H) (hy : |y| ≤ H) (h : (x : ZMod p) = (y : ZMod p)) : x = y := by
133  have hd : (p : ℤ) ∣ x - y := (ZMod.intCast_eq_intCast_iff_dvd_sub y x p).1 h.symm
134  have hlt : |x - y| < (p : ℤ) := by
135    calc |x - y| ≤ |x| + |y| := abs_sub _ _
136      _ ≤ H + H := by omega
137      _ < p := by omega
138  have := Int.eq_zero_of_abs_lt_dvd hd hlt

148/-- 1:E2 (Proposition 5 reversed): adjoining a root of `X² + 1` to a field that already
149contains one gives zero divisors — `F[X]/(X²+1)` is not a field on the shell. With `u² = −1`,
150`(X + u)(X − u) = 0` in the quotient while both factors are nonzero. -/
151⧉theorem complex_chart_zero_divisor (u : F) (hu : u ^ 2 = -1) :
152    let R := AdjoinRoot ((X : F[X]) ^ 2 + 1)
153    ∃ α β : R, α ≠ 0 ∧ β ≠ 0 ∧ α * β = 0 := by
154  intro R
155  have hmonic : ((X : F[X]) ^ 2 + 1).Monic := by monicity!
156  have hdeg : ((X : F[X]) ^ 2 + 1).natDegree = 2 := by compute_degree!
157  refine ⟨AdjoinRoot.root _ + AdjoinRoot.of _ u, AdjoinRoot.root _ - AdjoinRoot.of _ u, ?_, ?_, ?_⟩
158  · -- X + u is not divisible by X² + 1
159    intro h
160    have h' : AdjoinRoot.mk ((X : F[X]) ^ 2 + 1) (X + C u) = 0 := by
161      rw [map_add, AdjoinRoot.mk_X, AdjoinRoot.mk_C]; exact h
162    rw [AdjoinRoot.mk_eq_zero] at h'
163    have hne : (X + C u : F[X]) ≠ 0 := X_add_C_ne_zero u
164    have := Polynomial.natDegree_le_of_dvd h' hne
165    rw [hdeg, natDegree_X_add_C] at this
166    omega
167  · intro h
168    have h' : AdjoinRoot.mk ((X : F[X]) ^ 2 + 1) (X - C u) = 0 := by
169      rw [map_sub, AdjoinRoot.mk_X, AdjoinRoot.mk_C]; exact h
170    rw [AdjoinRoot.mk_eq_zero] at h'
171    have hne : (X - C u : F[X]) ≠ 0 := X_sub_C_ne_zero u
172    have := Polynomial.natDegree_le_of_dvd h' hne
173    rw [hdeg, natDegree_X_sub_C] at this
174    omega
175  · -- (root + u)(root − u) = root² − u² = −1 − (−1) = 0
176    have hroot : (AdjoinRoot.root ((X : F[X]) ^ 2 + 1)) ^ 2 = -1 := by
177      have := AdjoinRoot.eval₂_root ((X : F[X]) ^ 2 + 1)
178      simp only [eval₂_add, eval₂_pow, eval₂_X, eval₂_one] at this
179      linear_combination this
180    have hu' : (AdjoinRoot.of ((X : F[X]) ^ 2 + 1) u) ^ 2 = -1 := by
181      rw [← map_pow, hu, map_neg, map_one]
182    linear_combination hroot - hu'
