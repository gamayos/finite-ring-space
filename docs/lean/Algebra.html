94/-- 1:B4 (Definition 2 corrected, Lemma 1): in the affine frame `(a, b)`, `φ x = a + b x`
95carries `+` and `·` to `⊕` and `⊗`, is a bijection, and the multiplicative unit of the
96relabelled field is `a + b`; `b` is the unit only when `a = 0`. -/
97⧉theorem affine_frame (a b : F) (hb : b ≠ 0) :
98    let φ : F → F := fun x => a + b * x
99    let oplus : F → F → F := fun x y => a + b * ((x - a) / b + (y - a) / b)
100    let otimes : F → F → F := fun x y => a + b * ((x - a) / b * ((y - a) / b))
101    (∀ x y, φ (x + y) = oplus (φ x) (φ y)) ∧
102    (∀ x y, φ (x * y) = otimes (φ x) (φ y)) ∧
103    Function.Bijective φ ∧
104    (∀ y, otimes (a + b) y = y) ∧
105    (a ≠ 0 → otimes b (a + b) ≠ a + b) := by
106  intro φ oplus otimes
107  refine ⟨?_, ?_, ?_, ?_, ?_⟩
108  · intro x y; simp only [φ, oplus]; field_simp; ring
109  · intro x y; simp only [φ, otimes]; field_simp; ring
110  · constructor
111    · intro x y h; simp only [φ] at h
112      exact mul_left_cancel₀ hb (add_left_cancel h)
113    · intro y; exact ⟨(y - a) / b, by simp only [φ]; field_simp; ring⟩
114  · intro y; simp only [otimes]; field_simp
115    ring
116  · intro ha h; simp only [otimes] at h
117    -- a + b·((b−a)/b · b/b) = a + b  ⇒  ab = 0  ⇒  a = 0
118    field_simp at h
119    have hab : a * b = 0 := by linear_combination -h
120    exact ha ((mul_eq_zero.1 hab).resolve_right hb)

153⧉theorem complex_chart_zero_divisor (u : F) (hu : u ^ 2 = -1) :

158  have hdeg : ((X : F[X]) ^ 2 + 1).natDegree = 2 := by compute_degree!

162    have h' : AdjoinRoot.mk ((X : F[X]) ^ 2 + 1) (X + C u) = 0 := by
163      rw [map_add, AdjoinRoot.mk_X, AdjoinRoot.mk_C]; exact h
